     6→/** Smooth 0..1 step. */
     7→function sstep(a: number, b: number, x: number): number {
     8→  const t = clamp((x - a) / (b - a), 0, 1);
     9→  return t * t * (3 - 2 * t);
    10→}

    12→function wrapIdx(i: number, n: number): number {

   125→/** How deep the dry-riverbed pit floor sits below road level at gap progress t (0..1). */
   126→const PIT_DEPTH = -9;
   127→export function pitDepthAt(gapT: number): number {
   128→  if (gapT < 0) return 0;
   129→  // flat landing floor under the cliff, then a smooth drivable climb back to road level
   130→  if (gapT < 0.12) return PIT_DEPTH;
   131→  if (gapT > 0.99) return 0;
   132→  const u = (gapT - 0.12) / 0.87; // 0..1 across the exit ramp
   133→  // smoothstep-blended ramp: gentle at both ends, steady grade in the middle
   134→  const s = u * u * (3 - 2 * u);
   135→  return PIT_DEPTH * (1 - s);
   136→}

   165→  private buildSpline() {
   166→    const pts = CONTROL.map((p) => new THREE.Vector3(p[0], p[1], p[2]));
   167→    const curve = new THREE.CatmullRomCurve3(pts, true, "catmullrom", 0.5);
   168→    const approxLen = curve.getLength();
   169→    const n = Math.max(600, Math.round(approxLen / this.step));
   170→    const spaced = curve.getSpacedPoints(n);
   171→    this.N = n;
   172→    this.length = n * this.step;
   173→    const raw = spaced.slice(0, n);
   174→
   175→    const dirs: THREE.Vector2[] = [];
   176→    for (let i = 0; i < n; i++) {
   177→      const a = raw[wrapIdx(i - 1, n)];
   178→      const b = raw[wrapIdx(i + 1, n)];
   179→      const d = new THREE.Vector2(b.x - a.x, b.z - a.z);
   180→      if (d.lengthSq() < 1e-8) d.set(0, -1);
   181→      dirs.push(d.normalize());
   182→    }
   183→    // curvature → banking (outer edge raised)
   184→    const banks = new Array<number>(n).fill(0);
   185→    for (let i = 0; i < n; i++) {
   186→      const d0 = dirs[wrapIdx(i - 2, n)];
   187→      const d1 = dirs[wrapIdx(i + 2, n)];
   188→      const crossY = d0.x * d1.y - d0.y * d1.x; // >0 = left turn
   189→      const dot = clamp(d0.dot(d1), -1, 1);
   190→      const dTheta = Math.atan2(crossY, dot);
   191→      banks[i] = clamp((dTheta / (4 * this.step)) * 46, -1, 1) * 3.0;
   192→    }
   193→    const sm = new Array<number>(n).fill(0);
   194→    for (let i = 0; i < n; i++) {
   195→      let acc = 0;
   196→      for (let k = -9; k <= 9; k++) acc += banks[wrapIdx(i + k, n)];
   197→      sm[i] = acc / 19;
   198→    }
   199→
   200→    for (let i = 0; i < n; i++) {
   201→      const p = raw[i];
   202→      const d = dirs[i];
   203→      const r = new THREE.Vector2(-d.y, d.x); // right = (-dz, dx)
   204→      this.samples.push({
   205→        x: p.x,
   206→        y: p.y,
   207→        z: p.z,
   208→        dx: d.x,
   209→        dz: d.y,
   210→        rx: r.x,
   211→        rz: r.y,
   212→        hw: 9,
   213→        shoulderL: 8,
   214→        shoulderR: 8,
   215→        bank: sm[i],
   216→        gap: false,
   217→        gapT: -1,
   218→        rampP: -1,
   219→        dualT: 0,
   220→        wallRideR: false,
   221→        wallRideL: false,
   222→        riverbed: false,
   223→        tunnel: false,
   224→        wallHL: 24,
   225→        wallHR: 24,
   226→      });
   227→    }
   228→  }
   229→
   230→  private idxNear(x: number, z: number): number {
   231→    let best = 0;
   232→    let bd = Infinity;
   233→    for (let i = 0; i < this.N; i++) {
   234→      const s = this.samples[i];
   235→      const d = (s.x - x) * (s.x - x) + (s.z - z) * (s.z - z);
   236→      if (d < bd) {
   237→        bd = d;
   238→        best = i;
   239→      }
   240→    }
   241→    return best;
   242→  }

   244→  private range(a: number, b: number): [number, number] {
   245→    return [
   246→      this.idxNear(CONTROL[a][0], CONTROL[a][2]),
   247→      this.idxNear(CONTROL[b][0], CONTROL[b][2]),
   248→    ];

   251→  private forEachInRange(r: [number, number], fn: (i: number, t: number) => void) {
   252→    const [a, b] = r;
   253→    const span = (b - a + this.N) % this.N;
   254→    for (let k = 0; k <= span; k++) fn(wrapIdx(a + k, this.N), span > 0 ? k / span : 1);
   255→  }

   261→  private buildZones() {
   262→    const S = this.samples;
   263→
   264→    // --- launch ramp before the gap (last 58% of approach → crest) ---
   265→    const ramp = this.range(2, 3);
   266→    this.forEachInRange(ramp, (i, t) => {
   267→      S[i].rampP = clamp((t - 0.42) / 0.58, 0, 1);
   268→    });
   269→
   270→    // --- the GAP over the dry riverbed canyon ---
   271→    // The pit is a smooth valley, not a box canyon: a short flat landing
   272→    // floor under the cliff, then a long drivable ramp that climbs back up
   273→    // to the landing road — undershoot the jump and you can still drive out.
   274→    const crestI = this.idxNear(CONTROL[3][0], CONTROL[3][2]);
   275→    const landI = this.idxNear(CONTROL[4][0], CONTROL[4][2]);
   276→    const gapSpan = (landI - crestI + this.N) % this.N;
   277→    const gapStart = crestI + 2;
   278→    const gapEnd = crestI + Math.round(gapSpan * 0.62);
   279→    for (let k = gapStart; k <= gapEnd; k++) {
   280→      const s = S[wrapIdx(k, this.N)];
   281→      s.gap = true;
   282→      s.gapT = (k - gapStart) / Math.max(1, gapEnd - gapStart);
   283→      s.shoulderL = 3.2;
   284→      s.shoulderR = 3.2;
   285→      s.wallHL = 30;
   286→      s.wallHR = 30;
   287→    }
   288→
   289→    // --- natural arch zone: taller walls ---
   290→    const archI = this.idxNear(CONTROL[6][0], CONTROL[6][2]);
   291→    for (let k = archI - 14; k <= archI + 14; k++) {
   292→      const s = S[wrapIdx(k, this.N)];
   293→      s.wallHL = 42;
   294→      s.wallHR = 42;
   295→    }
   296→
   297→    // --- wall-ride carousel: banked outer wall on the right ---
   298→    const carousel = this.range(8, 11);
   299→    this.forEachInRange(carousel, (i, t) => {
   300→      const edge = sstep(0, 0.14, t) * (1 - sstep(0.86, 1, t));
   301→      const s = S[i];
   302→      s.wallRideR = edge > 0.01;
   303→      s.shoulderR = 26 * edge + 8 * (1 - edge);
   304→      s.wallHR = 30;
   305→      s.bank = s.bank * (1 - edge) + (s.bank * 0.4 + 2.4) * edge;
   306→    });
   307→
   308→    // --- tunnel through the mesa ---
   309→    const tunnel = this.range(13, 16);
   310→    this.forEachInRange(tunnel, (i, t) => {
   311→      const edge = sstep(0, 0.05, t) * (1 - sstep(0.95, 1, t));
   312→      const s = S[i];
   313→      s.tunnel = edge > 0.01;
   314→      s.hw = 8.5 + 0.5 * (1 - edge);
   315→      s.shoulderL = 1.5 + 6.5 * (1 - edge);
   316→      s.shoulderR = 1.5 + 6.5 * (1 - edge);
   317→    });
   318→
   319→    // --- dual-line downhill: high terrace on the left ---
   320→    const dual = this.range(16, 18);
   321→    this.forEachInRange(dual, (i, t) => {
   322→      const s = S[i];
   323→      s.dualT = sstep(0, 0.12, t) * (1 - sstep(0.88, 1, t));
   324→      s.shoulderL = 9.5;
   325→    });
   326→
   327→    // --- riverbed: wide sandy floor ---
   328→    const river = this.range(19, 24);
   329→    this.forEachInRange(river, (i, t) => {
   330→      const edge = sstep(0, 0.06, t) * (1 - sstep(0.94, 1, t));
   331→      const s = S[i];
   332→      s.riverbed = true;
   333→      s.hw = 9 + 11 * edge;
   334→      s.shoulderL = 8 + 6 * edge;
   335→      s.shoulderR = 8 + 6 * edge;
   336→      s.wallHL = 20;
   337→      s.wallHR = 20;
   338→    });
   339→
   340→    // --- low walls near the start/finish for the vista ---
   341→    for (let k = -30; k <= 40; k++) {
   342→      const s = S[wrapIdx(k, this.N)];
   343→      s.wallHL = Math.min(s.wallHL, 14);
   344→      s.wallHR = Math.min(s.wallHR, 14);
   345→    }
   346→
   347→    // wall height noise
   348→    for (let i = 0; i < this.N; i++) {
   349→      const s = S[i];
   350→      const n =
   351→        Math.sin(i * 0.093) * 0.5 + Math.sin(i * 0.031 + 2.1) * 0.3 + Math.sin(i * 0.017 + 4) * 0.2;
   352→      s.wallHL = Math.max(10, s.wallHL + n * 7);
   353→      s.wallHR = Math.max(10, s.wallHR + n * 7 + Math.sin(i * 0.05) * 2);
   354→    }
   355→
   356→    // --- kicker bumps in the riverbed ---
   357→    const rB = this.idxNear(CONTROL[21][0], CONTROL[21][2]);
   358→    const rC = this.idxNear(CONTROL[22][0], CONTROL[22][2]);
   359→    const rD = this.idxNear(CONTROL[23][0], CONTROL[23][2]);
   360→    this.kickers = [
   361→      { s: (rB + 9) * this.step, lat: -8, r: 7, h: 2.1 },
   362→      { s: (rC + 10) * this.step, lat: 7, r: 7, h: 2.2 },
   363→      { s: (rD + 12) * this.step, lat: -5, r: 6.5, h: 1.9 },
   364→    ];
   365→
   366→    // --- checkpoint gates in racing order ---
   367→    const gateAnchors: { c: number; finish: boolean }[] = [
   368→      { c: 4, finish: false }, // gap landing
   369→      { c: 6, finish: false }, // arch
   370→      { c: 7, finish: false }, // climb
   371→      { c: 10, finish: false }, // carousel apex
   372→      { c: 13, finish: false }, // tunnel entry
   373→      { c: 16, finish: false }, // tunnel exit
   374→      { c: 18, finish: false }, // dual-line merge
   375→      { c: 22, finish: false }, // riverbed mid
   376→      { c: 27, finish: false }, // final arc
   377→      { c: 0, finish: true }, // start/finish
   378→    ];
   379→    for (const g of gateAnchors) {
   380→      const idx = this.idxNear(CONTROL[g.c][0], CONTROL[g.c][2]);
   381→      const s = S[idx];
   382→      this.gates.push({ idx, pos: new THREE.Vector3(s.x, s.y, s.z), isFinish: g.finish });
   383→      this.gatePts.push({ x: s.x, z: s.z, finish: g.finish });
   384→    }
   385→
   386→    for (let i = 0; i < this.N; i += 6) this.minimapPts.push({ x: S[i].x, z: S[i].z });
   387→
   388→    this.addObstacles();
   389→  }

   392→    const rnd = mulberry32(20240607);
   393→    // riverbed boulders
   394→    const rA = this.idxNear(CONTROL[20][0], CONTROL[20][2]);
   395→    const rB = this.idxNear(CONTROL[23][0], CONTROL[23][2]);
   396→    const span = (rB - rA + this.N) % this.N;
   397→    for (let k = 0; k < 8; k++) {
   398→      const idx = wrapIdx(rA + Math.round(span * (0.12 + 0.78 * rnd())), this.N);
   399→      const s = this.samples[idx];
   400→      const lat = (rnd() * 9 + 4.5) * (rnd() > 0.5 ? 1 : -1);
   401→      const r = 1.7 + rnd() * 0.9;
   402→      this.obstacles.push({
   403→        x: s.x + s.rx * lat,
   404→        z: s.z + s.rz * lat,
   405→        r,
   406→        top: this.heightAt(idx, lat) + r * 1.5,
   407→        lastNear: -1e9,
   408→      });
   409→    }
   410→    // rock fins on the dual-line low (right) side
   411→    const dA = this.idxNear(CONTROL[17][0], CONTROL[17][2]);
   412→    for (let k = 0; k < 3; k++) {
   413→      const idx = wrapIdx(dA + 8 + k * 14, this.N);
   414→      const s = this.samples[idx];
   415→      const lat = 5.5 + rnd() * 3;
   416→      this.obstacles.push({
   417→        x: s.x + s.rx * lat,
   418→        z: s.z + s.rz * lat,
   419→        r: 1.6,
   420→        top: this.heightAt(idx, lat) + 3.4,
   421→        lastNear: -1e9,
   422→      });
   423→    }
   424→    // arch pillars
   425→    const archI = this.idxNear(CONTROL[6][0], CONTROL[6][2]);

   443→        x: ls.x + ls.rx * lat,

   445→        r: 2.0,
   446→        top: this.heightAt(landI, lat) + 3,

   460→  groundAt(i: number, lat: number, out: GroundInfo): GroundInfo {
   461→    const s = this.sample(i);
   462→    const al = Math.abs(lat);
   463→    if (s.gap && al < s.hw + 3) {
   464→      out.h = s.y + pitDepthAt(s.gapT);
   465→      out.onRoad = false;
   466→      out.inPitZone = true;
   467→      return out;
   468→    }
   469→    out.inPitZone = false;
   470→    let h = s.y;
   471→    const u = clamp(lat / s.hw, -1, 1);
   472→    h += s.bank * u;
   473→    if (s.riverbed) h += 0.45 * u * u;
   474→    if (s.dualT > 0 && lat > 3) {
   475→      const t = sstep(3, 5, lat) * (1 - sstep(13, 15.5, lat));
   476→      h += 2.6 * s.dualT * t;
   477→    }
   478→    if (s.wallRideR && lat > 4) h += 13 * Math.pow(Math.min(1, (lat - 4) / 24), 2.4);
   479→    if (s.wallRideL && lat < -4) h += 13 * Math.pow(Math.min(1, (-lat - 4) / 24), 2.4);
   480→    if (s.rampP >= 0 && al < s.hw + 1) {
   481→      const latFade = 1 - sstep(4.5, s.hw, al);
   482→      h += 3.4 * sstep(0.1, 0.92, s.rampP) * latFade;
   483→    }
   484→    let onRoad = al <= s.hw;
   485→    if (al > s.hw) {
   486→      if (s.wallRideR && lat > 0) onRoad = true;
   487→      else if (s.dualT > 0 && lat < 0 && lat > 3) onRoad = true;
   488→      else {
   489→        const sh = lat > 0 ? s.shoulderR : s.shoulderL;
   490→        if (sh > 0) h -= 0.55 * Math.min(1, (al - s.hw) / Math.max(0.5, sh));
   491→      }
   492→    }
   493→    const sHere = wrapIdx(i, this.N) * this.step;
   494→    for (const k of this.kickers) {
   495→      let ds = Math.abs(sHere - k.s);
   496→      ds = Math.min(ds, this.length - ds);
   497→      const dl = lat - k.lat;
   498→      const d2 = ds * ds + dl * dl;
   499→      if (d2 < k.r * k.r) {
   500→        const q = Math.sqrt(d2) / k.r;
   501→        h += k.h * (0.5 + 0.5 * Math.cos(Math.PI * q));
   502→      }
   503→    }
   504→    out.h = h;
   505→    out.onRoad = onRoad;
   506→    return out;
   507→  }

   509→  heightAt(i: number, lat: number): number {
   510→    return this.groundAt(i, lat, TMP_GROUND).h;
   511→  }

   513→  // ---- continuous (interpolated) sampling ----
   514→  // The physics used to snap to the discrete 2 m sample grid, which made the
   515→  // car's height (and everything derived from it) pop at every sample boundary
   516→  // — a visible buzz/jiggle at speed. These queries blend adjacent samples so
   517→  // the driven surface matches the rendered road exactly and stays smooth.
   518→
   519→  /** Ground query at a continuous sample coordinate p (= idx + fraction ahead). */
   520→  groundAtP(p: number, lat: number, out: GroundInfo): GroundInfo {
   521→    const i0 = Math.floor(p);
   522→    const fr = clamp(p - i0, 0, 1);
   523→    const a = this.groundAt(i0, lat, TMP_GA);
   524→    const b = this.groundAt(i0 + 1, lat, TMP_GB);
   525→    if (a.inPitZone !== b.inPitZone) {
   526→      // cliff lip at the gap — keep the step crisp so launches still fire
   527→      const use = fr < 0.5 ? a : b;
   528→      out.h = use.h;
   529→      out.onRoad = use.onRoad;
   530→      out.inPitZone = use.inPitZone;
   531→      return out;
   532→    }
   533→    out.h = a.h + (b.h - a.h) * fr;
   534→    out.onRoad = fr < 0.5 ? a.onRoad : b.onRoad;
   535→    out.inPitZone = a.inPitZone;
   536→    return out;
   537→  }
   538→
   539→  heightAtP(p: number, lat: number): number {
   540→    return this.groundAtP(p, lat, TMP_GROUND).h;
   541→  }
   542→
   543→  /** Lateral slope at a continuous sample coordinate. */
   544→  slopeLatAtP(p: number, lat: number): number {
   545→    return (this.heightAtP(p, lat + 0.9) - this.heightAtP(p, lat - 0.9)) / 1.8;
   546→  }
   547→
   548→  /** Forward slope at a continuous sample coordinate. */
   549→  slopeFwdAtP(p: number, lat: number): number {
   550→    const d = 0.9 / this.step;
   551→    return (this.heightAtP(p + d, lat) - this.heightAtP(p - d, lat)) / 1.8;
   552→  }
   553→
   554→  slopeLatAt(i: number, lat: number): number {
   555→    const a = this.heightAt(wrapIdx(i, this.N), lat + 0.9);
   556→    const b = this.heightAt(wrapIdx(i, this.N), lat - 0.9);
   557→    return (a - b) / 1.8;
   558→  }
   559→
   560→  slopeFwdAt(i: number, lat: number): number {
   561→    const a = this.heightAt(wrapIdx(i + 1, this.N), lat);
   562→    const b = this.heightAt(wrapIdx(i - 1, this.N), lat);
   563→    return (a - b) / (2 * this.step);
   564→  }
   565→
   566→  wallDist(i: number, side: 1 | -1): number {
   567→    const s = this.sample(i);
   568→    return side > 0 ? s.hw + s.shoulderR + 0.3 : s.hw + s.shoulderL + 0.3;
   569→  }
   570→
   571→  nearestIdx(x: number, z: number, hint: number): number {
   572→    let best = hint;
   573→    let bd = Infinity;
   574→    for (let k = -35; k <= 95; k++) {
   575→      const i = wrapIdx(hint + k, this.N);
   576→      const s = this.samples[i];
   577→      const d = (s.x - x) * (s.x - x) + (s.z - z) * (s.z - z);
   578→      if (d < bd) {
   579→        bd = d;
   580→        best = i;
   581→      }
   582→    }
   583→    if (bd > 90 * 90) best = this.idxNear(x, z);
   584→    return best;
   585→  }
   586→
   587→  lateralOf(i: number, x: number, z: number): number {
   588→    const s = this.sample(i);
   589→    return (x - s.x) * s.rx + (z - s.z) * s.rz;
   590→  }
   591→
   592→  posAt(i: number, lat: number, y?: number): THREE.Vector3 {
   593→    const s = this.sample(i);
   594→    return new THREE.Vector3(s.x + s.rx * lat, y ?? s.y, s.z + s.rz * lat);
   595→  }

   614→  private buildWalls() {
   615→    const S = this.samples;
   616→    const n = this.N;
   617→    for (const side of [-1, 1] as const) {
   618→      const pos: number[] = [];
   619→      const col: number[] = [];
   620→      const idx: number[] = [];
   621→      const strata = PALETTE.rockStrata.map((c) => new THREE.Color(c));
   622→      const cBase = new THREE.Color(PALETTE.rockBase);
   623→      for (let i = 0; i < n; i++) {
   624→        const s = S[i];
   625→        const wd = this.wallDist(i, side) - 0.3;
   626→        const bx = s.x + s.rx * wd * side;
   627→        const bz = s.z + s.rz * wd * side;
   628→        const wallH = side > 0 ? s.wallHR : s.wallHL;
   629→        const lean = 7 + ((i * 37) % 5);
   630→        const tx = s.x + (s.rx * wd + lean) * side;
   631→        const tz = s.z + (s.rz * wd + lean) * side;
   632→        const by = s.gap ? s.y + pitDepthAt(s.gapT) - 0.4 : this.heightAt(i, wd * side) - 0.2;
   633→        const ty = s.y + wallH;
   634→        pos.push(bx, by, bz, tx, ty, tz);
   635→        for (const yv of [by, ty]) {
   636→          const band = Math.floor((yv + Math.sin(i * 0.7 + side) * 1.4) / 4.2);
   637→          const c = strata[((band % strata.length) + strata.length) % strata.length].clone();
   638→          c.lerp(cBase, 0.25 + 0.2 * Math.abs(Math.sin(i * 0.13 + band)));
   639→          col.push(c.r, c.g, c.b);
   640→        }
   641→      }
   642→      for (let i = 0; i < n; i++) {
   643→        const a = i * 2;
   644→        const b = ((i + 1) % n) * 2;
   645→        if (side > 0) {
   646→          // normal faces the track (-right)
   647→          idx.push(a, a + 1, b, a + 1, b + 1, b);
   648→        } else {
   649→          idx.push(a, b, a + 1, b, b + 1, a + 1);
   650→        }
   651→      }
   652→      const g = new THREE.BufferGeometry();
   653→      g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   654→      g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   655→      g.setIndex(idx);
   656→      g.computeVertexNormals();
   657→      const m = new THREE.MeshLambertMaterial({ vertexColors: true, flatShading: true });
   658→      const mesh = new THREE.Mesh(g, m);
   659→      mesh.castShadow = true;
   660→      mesh.receiveShadow = true;
   661→      this.group.add(mesh);
   662→      this.disposables.push(g, m);
   663→    }
   664→  }

   666→  private ringLats(i: number): number[] {
   667→    const s = this.sample(i);
   668→    const wdL = -this.wallDist(i, -1);
   669→    const wdR = this.wallDist(i, 1);
   670→    const lats: number[] = [wdL];
   671→    for (const l of ROAD_COLS) {
   672→      if (l > wdL + 0.05 && l < wdR - 0.05) lats.push(l);
   673→    }
   674→    lats.push(wdR);
   675→    return lats;
   676→  }

   678→  private buildRoad() {
   679→    const n = this.N;
   680→    const pos: number[] = [];
   681→    const col: number[] = [];
   682→    const uv: number[] = [];
   683→    const idx: number[] = [];
   684→    const cRoad = new THREE.Color(PALETTE.road);
   685→    const cRoadDust = new THREE.Color(PALETTE.roadDust);
   686→    const cSand = new THREE.Color(PALETTE.sand);
   687→    const cPit = new THREE.Color(PALETTE.pitSand);
   688→    const cTerr = new THREE.Color(0xcf8a52);
   689→    const gi: GroundInfo = { h: 0, onRoad: false, inPitZone: false };
   690→
   691→    const ringLats: number[][] = [];
   692→    const ringStart: number[] = [];
   693→    for (let i = 0; i < n; i++) {
   694→      ringStart[i] = pos.length / 3;
   695→      const s = this.samples[i];
   696→      const lats = this.ringLats(i);
   697→      ringLats.push(lats);
   698→      for (const l of lats) {
   699→        this.groundAt(i, l, gi);
   700→        let c: THREE.Color;
   701→        if (gi.inPitZone) c = cPit;
   702→        else if (Math.abs(l) <= s.hw + 0.01) c = cRoad.clone().lerp(cRoadDust, 0.3 + 0.2 * Math.sin(i * 0.37));
   703→        else if (s.dualT > 0 && l < -3) c = cTerr.clone();
   704→        else if (s.wallRideR && l > s.hw) {
   705→          const t = clamp((l - 4) / 24, 0, 1);
   706→          c = cRoad.clone().lerp(cTerr, t * 0.8);
   707→        } else c = cSand.clone().offsetHSL(0, 0, Math.sin(i * 0.53 + l) * 0.015);
   708→        pos.push(s.x + s.rx * l, gi.h, s.z + s.rz * l);
   709→        col.push(c.r, c.g, c.b);
   710→        uv.push(l / 12, (i * this.step) / 7);
   711→      }
   712→    }
   713→    for (let i = 0; i < n; i++) {
   714→      const j = (i + 1) % n;
   715→      if (this.samples[i].gap || this.samples[j].gap) continue;
   716→      const a = ringStart[i];
   717→      const b = ringStart[j];
   718→      const m = ringLats[i].length; // stitch with ring i's column count on both rings
   719→      for (let k = 0; k < m - 1; k++) {
   720→        idx.push(a + k, a + k + 1, b + k, a + k + 1, b + k + 1, b + k);
   721→      }
   722→    }
   723→    const g = new THREE.BufferGeometry();
   724→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   725→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   726→    g.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
   727→    g.setIndex(idx);
   728→    g.computeVertexNormals();
   729→    const tex = makeSandNoiseTexture();
   730→    const m = new THREE.MeshLambertMaterial({ vertexColors: true, map: tex, flatShading: true });
   731→    const mesh = new THREE.Mesh(g, m);
   732→    mesh.receiveShadow = true;
   733→    this.group.add(mesh);
   734→    this.disposables.push(g, m, tex);
   735→  }

   737→  /** Painted route markings — edge lines + forward chevrons so the line is always readable. */
   738→  private buildRouteMarks() {
   739→    const n = this.N;
   740→    const lineMat = new THREE.MeshBasicMaterial({
   741→      color: 0xf7e7c3,
   742→      fog: true,
   743→      side: THREE.DoubleSide,
   744→      polygonOffset: true,
   745→      polygonOffsetFactor: -3,
   746→      polygonOffsetUnits: -3,
   747→    });
   748→    const chevMat = new THREE.MeshBasicMaterial({
   749→      color: 0xfffbe8,
   750→      fog: true,
   751→      side: THREE.DoubleSide,
   752→      polygonOffset: true,
   753→      polygonOffsetFactor: -4,
   754→      polygonOffsetUnits: -4,
   755→    });
   756→
   757→    // ---- continuous edge lines hugging the road edge ----
   758→    const pos: number[] = [];
   759→    const idx: number[] = [];
   760→    const lineHalf = 0.19;
   761→    for (const side of [-1, 1] as const) {
   762→      const start = pos.length / 3;
   763→      for (let i = 0; i < n; i++) {
   764→        const s = this.samples[i];
   765→        const latE = (s.hw - 0.85) * side;
   766→        const y = this.heightAt(i, latE) + 0.05;
   767→        pos.push(
   768→          s.x + s.rx * (latE - lineHalf), y, s.z + s.rz * (latE - lineHalf),
   769→          s.x + s.rx * (latE + lineHalf), y, s.z + s.rz * (latE + lineHalf)
   770→        );
   771→      }
   772→      for (let i = 0; i < n; i++) {
   773→        const j = (i + 1) % n;
   774→        if (this.samples[i].gap || this.samples[j].gap) continue;
   775→        const a = start + i * 2;
   776→        const b = start + j * 2;
   777→        idx.push(a, a + 1, b, a + 1, b + 1, b);
   778→      }
   779→    }
   780→    const lineGeo = new THREE.BufferGeometry();
   781→    lineGeo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   782→    lineGeo.setIndex(idx);
   783→    const lineMesh = new THREE.Mesh(lineGeo, lineMat);
   784→    lineMesh.frustumCulled = false; // spans the whole loop
   785→    this.group.add(lineMesh);
   786→    this.disposables.push(lineGeo, lineMat);
   787→
   788→    // ---- forward-pointing chevrons along the centerline ----
   789→    const gateBlocked = new Set<number>();
   790→    for (const g of this.gates) {
   791→      for (let k = -4; k <= 4; k++) gateBlocked.add(wrapIdx(g.idx + k, n));
   792→    }
   793→    const cpos: number[] = [];
   794→    const cidx: number[] = [];
   795→    const arm = (f0: number, l0: number, f1: number, l1: number, w: number, i: number) => {
   796→      const df = f1 - f0;
   797→      const dl = l1 - l0;
   798→      const len = Math.hypot(df, dl);
   799→      if (len < 1e-4) return;
   800→      const px = (-dl / len) * (w * 0.5);
   801→      const pl = (df / len) * (w * 0.5);
   802→      const corners: [number, number][] = [
   803→        [f0 + px, l0 + pl],
   804→        [f1 + px, l1 + pl],
   805→        [f1 - px, l1 - pl],
   806→        [f0 - px, l0 - pl],
   807→      ];
   808→      const base = cpos.length / 3;
   809→      const s = this.samples[i];
   810→      for (const [f, l] of corners) {
   811→        const x = s.x + s.dx * f + s.rx * l;
   812→        const z = s.z + s.dz * f + s.rz * l;
   813→        const y = this.heightAtP(i + f / this.step, l) + 0.07;
   814→        cpos.push(x, y, z);
   815→      }
   816→      cidx.push(base, base + 1, base + 2, base, base + 2, base + 3);
   817→    };
   818→    for (let i = 6; i < n; i += 14) {
   819→      const s = this.samples[i];
   820→      if (s.gap || gateBlocked.has(i)) continue;
   821→      arm(1.15, 0, -0.45, 1.35, 0.46, i);
   822→      arm(1.15, 0, -0.45, -1.35, 0.46, i);
   823→    }
   824→    const chevGeo = new THREE.BufferGeometry();
   825→    chevGeo.setAttribute("position", new THREE.Float32BufferAttribute(cpos, 3));
   826→    chevGeo.setIndex(cidx);
   827→    const chevMesh = new THREE.Mesh(chevGeo, chevMat);
   828→    chevMesh.frustumCulled = false;
   829→    this.group.add(chevMesh);
   830→    this.disposables.push(chevGeo, chevMat);
   831→  }

   833→  private buildPit() {
   834→    const n = this.N;
   835→    const pos: number[] = [];
   836→    const idx: number[] = [];
   837→    const col: number[] = [];
   838→    const cPit = new THREE.Color(PALETTE.pitSand);
   839→    const cDark = new THREE.Color(0xb08a58);
   840→    let v = 0;
   841→    const pushTri = (a: number[], b: number[], c: number[], shade: number) => {
   842→      for (const p of [a, b, c]) pos.push(p[0], p[1], p[2]);
   843→      const cc = cPit.clone().lerp(cDark, shade);
   844→      col.push(cc.r, cc.g, cc.b, cc.r, cc.g, cc.b, cc.r, cc.g, cc.b);
   845→      idx.push(v, v + 1, v + 2);
   846→      v += 3;
   847→    };
   848→    const pushQuad = (a: number[], b: number[], c: number[], d: number[]) => {
   849→      pushTri(a, b, c, 0);
   850→      pushTri(a, c, d, 0.25);
   851→    };
   852→    const PL = 12.4;
   853→    let i = 0;
   854→    while (i < n) {
   855→      if (!this.samples[i].gap) {
   856→        i++;
   857→        continue;
   858→      }
   859→      let j = i;
   860→      while (j < n && this.samples[j].gap) j++;
   861→      const jEnd = Math.min(j, n - 1); // gap never crosses the wrap in this track
   862→      // smooth valley floor — follows the same profile the physics drives on,
   863→      // so the rendered sand matches the collision surface exactly
   864→      const floorY = (k: number) => {
   865→        const sk = this.samples[wrapIdx(k, n)];
   866→        return sk.y + pitDepthAt(sk.gapT);
   867→      };
   868→      for (let k = i; k < jEnd; k++) {
   869→        const s0 = this.samples[k];
   870→        const s1 = this.samples[k + 1];
   871→        for (const [l0, l1] of [
   872→          [-PL, -PL / 2],
   873→          [-PL / 2, PL / 2],
   874→          [PL / 2, PL],
   875→        ] as [number, number][]) {
   876→          pushQuad(
   877→            [s0.x + s0.rx * l0, floorY(k), s0.z + s0.rz * l0],
   878→            [s0.x + s0.rx * l1, floorY(k), s0.z + s0.rz * l1],
   879→            [s1.x + s1.rx * l1, floorY(k + 1), s1.z + s1.rz * l1],
   880→            [s1.x + s1.rx * l0, floorY(k + 1), s1.z + s1.rz * l0]
   881→          );
   882→        }
   883→      }
   884→      // near face: crisp cliff under the launch edge (keeps launches snappy)
   885→      const sp = this.samples[wrapIdx(i - 1, n)];
   886→      const sa = this.samples[i];
   887→      for (const [l0, l1] of [
   888→        [-PL, 0],
   889→        [0, PL],
   890→      ] as [number, number][]) {
   891→        pushQuad(
   892→          [sp.x + sp.rx * l0, this.heightAt(wrapIdx(i - 1, n), l0), sp.z + sp.rz * l0],
   893→          [sp.x + sp.rx * l1, this.heightAt(wrapIdx(i - 1, n), l1), sp.z + sp.rz * l1],
   894→          [sa.x + sa.rx * l1, floorY(i), sa.z + sa.rz * l1],
   895→          [sa.x + sa.rx * l0, floorY(i), sa.z + sa.rz * l0]
   896→        );
   897→      }
   898→      // far face: the valley floor has already climbed to road level here —
   899→      // just a short skirt so there is no visible seam at the join
   900→      const sf = this.samples[wrapIdx(jEnd, n)];
   901→      const sl = this.samples[wrapIdx(jEnd - 1, n)];
   902→      for (const [l0, l1] of [
   903→        [-PL, 0],
   904→        [0, PL],
   905→      ] as [number, number][]) {
   906→        pushQuad(
   907→          [sl.x + sl.rx * l0, floorY(jEnd - 1), sl.z + sl.rz * l0],
   908→          [sl.x + sl.rx * l1, floorY(jEnd - 1), sl.z + sl.rz * l1],
   909→          [sf.x + sf.rx * l1, this.heightAt(wrapIdx(jEnd, n), l1), sf.z + sf.rz * l1],
   910→          [sf.x + sf.rx * l0, this.heightAt(wrapIdx(jEnd, n), l0), sf.z + sf.rz * l0]
   911→        );
   912→      }
   913→      i = j;
   914→    }
   915→    const g = new THREE.BufferGeometry();
   916→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   917→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   918→    g.setIndex(idx);
   919→    g.computeVertexNormals();
   920→    const m = new THREE.MeshLambertMaterial({
   921→      vertexColors: true,
   922→      flatShading: true,
   923→      side: THREE.DoubleSide,
   924→    });
   925→    const mesh = new THREE.Mesh(g, m);
   926→    mesh.receiveShadow = true;
   927→    this.group.add(mesh);
   928→    this.disposables.push(g, m);
   929→  }

   931→  private buildTunnel() {
   932→    const n = this.N;
   933→    const tunnelIdx: number[] = [];
   934→    for (let i = 0; i < n; i++) if (this.samples[i].tunnel) tunnelIdx.push(i);
   935→    if (tunnelIdx.length < 8) return;
   936→    const pos: number[] = [];
   937→    const col: number[] = [];
   938→    const idx: number[] = [];
   939→    const strata = PALETTE.rockStrata.map((c) => new THREE.Color(c));
   940→    const cDark = new THREE.Color(0x6e3524);
   941→    const SEG = 7;
   942→    const R = 12.2;
   943→    for (let r = 0; r < tunnelIdx.length; r++) {
   944→      const i = tunnelIdx[r];
   945→      const s = this.samples[i];
   946→      for (let k = 0; k <= SEG; k++) {
   947→        const a = Math.PI * (k / SEG);
   948→        const lx = -Math.cos(a) * R;
   949→        const ly = Math.sin(a) * R * 0.92;
   950→        pos.push(s.x + s.rx * lx, s.y + ly - 1.8, s.z + s.rz * lx);
   951→        const band = Math.floor((k + i * 0.2) / 2);
   952→        const c = strata[((band % strata.length) + strata.length) % strata.length].clone().lerp(cDark, 0.5);
   953→        col.push(c.r, c.g, c.b);
   954→      }
   955→      if (r > 0) {
   956→        for (let k = 0; k < SEG; k++) {
   957→          const a = (r - 1) * (SEG + 1) + k;
   958→          const b = r * (SEG + 1) + k;
   959→          idx.push(a, a + 1, b, a + 1, b + 1, b);
   960→        }
   961→      }
   962→    }
   963→    const g = new THREE.BufferGeometry();
   964→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   965→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   966→    g.setIndex(idx);
   967→    g.computeVertexNormals();
   968→    const m = new THREE.MeshLambertMaterial({

  1318→  /**

  1320→   * firing lines on the racing line. Smash one crate and the whole stack
  1321→   * chain-breaks into flying pieces. The full candidate set is generated once
  1322→   * (max count); difficulty decides how many stacks go live.
  1323→   */
  1324→  private buildStacks() {
  1325→    const n = this.N;
  1326→    const MAX_STACKS = 14;
  1327→    const rnd = mulberry32(9155);
  1328→
  1329→    // zones to keep clear: gates, ramps, the gap, the tunnel, the start grid, the break wall
  1330→    const blocked = new Uint8Array(n);
  1331→    for (const g of this.gates) {
  1332→      for (let k = -8; k <= 8; k++) blocked[wrapIdx(g.idx + k, n)] = 1;
  1333→    }
  1334→    for (let i = 0; i < n; i++) {
  1335→      if (this.samples[i].gap || this.samples[i].tunnel || this.samples[i].rampP >= 0) {
  1336→        for (let k = -7; k <= 7; k++) blocked[wrapIdx(i + k, n)] = 1;
  1337→      }
  1338→    }
  1339→    for (let k = -60; k <= 60; k++) blocked[wrapIdx(k, n)] = 1; // start/finish + grid
  1340→    if (this.breakWallIdx >= 0) {
  1341→      for (let k = -14; k <= 14; k++) blocked[wrapIdx(this.breakWallIdx + k, n)] = 1;
  1342→    }
  1343→
  1344→    const crateTex = makeCrateTexture();
  1345→    const crateGeo = new THREE.BoxGeometry(1.24, 1.24, 1.24);
  1346→    const crateMat = new THREE.MeshLambertMaterial({ map: crateTex });
  1347→    this.disposables.push(crateGeo, crateMat, crateTex);
  1348→
  1349→    // stack layout patterns: [lateral cell offset, stack level]
  1350→    const PATTERNS: [number, number][][] = [
  1351→      [
  1352→        [-1.32, 0],
  1353→        [0, 0],
  1354→        [1.32, 0],
  1355→        [-0.66, 1],
  1356→        [0.66, 1],
  1357→        [0, 2],
  1358→      ], // pyramid
  1359→      [
  1360→        [-0.65, 0],
  1361→        [0.65, 0],
  1362→        [0, 1],
  1363→      ], // pair + top
  1364→      [
  1365→        [-1.35, 0],
  1366→        [0, 0],
  1367→        [1.35, 0],
  1368→      ], // firing line across the road
  1369→      [
  1370→        [-0.66, 0],
  1371→        [0.66, 0],
  1372→        [-0.66, 1],
  1373→        [0.66, 1],
  1374→      ], // 2×2 block
  1375→    ];
  1376→
  1377→    const used: number[] = [];
  1378→    let cluster = 0;
  1379→    let placed = 0;
  1380→    let guard = 0;
  1381→    while (placed < MAX_STACKS && guard++ < 9000) {
  1382→      const idx = 60 + Math.floor(rnd() * (n - 120));
  1383→      if (blocked[idx]) continue;
  1384→      // stack spacing — each stack reads as its own little event on approach
  1385→      let tooClose = false;
  1386→      for (const u of used) {
  1387→        let d = Math.abs(idx - u);
  1388→        d = Math.min(d, n - d);
  1389→        if (d < 26) {
  1390→          tooClose = true;
  1391→          break;
  1392→        }
  1393→      }
  1394→      if (tooClose) continue;
  1395→      const s = this.samples[idx];
  1396→      // bias stacks toward the middle of the road
  1397→      const u = rnd() * 2 - 1;
  1398→      const lat = Math.sign(u) * Math.pow(Math.abs(u), 1.35) * Math.max(1.2, s.hw - 4.5);
  1399→      // keep clear of registered boulders / pillars
  1400→      let clash = false;
  1401→      for (const ob of this.obstacles) {
  1402→        const dx = ob.x - (s.x + s.rx * lat);
  1403→        const dz = ob.z - (s.z + s.rz * lat);

  1416→        const z = s.z + s.rz * blat;

  1418→        const mesh = new THREE.Mesh(crateGeo, crateMat);

  1425→          z,

  1427→          idx,

  1437→      }
  1438→      used.push(idx);
  1439→      cluster++;
  1440→      placed++;
  1441→    }
  1442→  }

  1444→  dispose() {

  1455→function mulberry32(seed: number) {
  1456→  let a = seed >>> 0;
  1457→  return () => {
  1458→    a |= 0;
  1459→    a = (a + 0x6d2b79f5) | 0;
  1460→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
  1461→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  1462→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  1463→  };
  1464→}
