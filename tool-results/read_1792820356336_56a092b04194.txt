   172→    this.scene.add(this.debris.group);

   176→    this.scene.add(this.car.group);
   177→    this.ghostCar = new CarVisual(true);
   178→    this.ghostCar.group.visible = false;
   179→    this.scene.add(this.ghostCar.group);
   180→
   181→    // AI rivals (identical performance, different paint & skill)
   182→    for (let i = 0; i < OPPONENTS.length; i++) {
   183→      const op = new Opponent(OPPONENTS[i], this.track, OPPONENT_GRID_LAT[i]);
   184→      // rival crate smashes shatter too (debris + a soft distant crack)
   185→      op.boxHitHook = (box) => {
   186→        const v = op.physics.vel;
   187→        const sp = Math.hypot(v.x, v.z);
   188→        const dirX = sp > 2 ? v.x / sp : 0;
   189→        const dirZ = sp > 2 ? v.z / sp : 1;
   190→        const power = clamp(6 + sp * 0.5, 7, 20);
   191→        let n = 0;
   192→        for (const b of this.track.boxes) {
   193→          if (b.cluster !== box.cluster || (!b.active && b !== box)) continue;
   194→          this.smashBox(b, dirX, dirZ, power, n * 0.045);
   195→          n++;
   196→        }
   197→        const d = box.x - this.physics.pos.x;
   198→        const dz = box.z - this.physics.pos.z;
   199→        if (d * d + dz * dz < 70 * 70) {
   200→          if (box.wall) this.audio.wallSmash();
   201→          else this.audio.boxHit();
   202→        }
   203→      };
   204→      this.scene.add(op.visual.group);
   205→      this.opponents.push(op);
   206→    }
   207→    this.scatterOpponentsOnTrack();
   208→
   209→    this.physics = new CarPhysics(this.track, {
   210→      onJump: () => this.audio.jump(),
   211→      onLanding: (i, clean) => {
   212→        this.audio.landing(clean, i);
   213→        this.trauma = Math.min(1, this.trauma + 0.18 + i * 0.3);
   214→        this.input.rumble(0.2 + i * 0.5, 120);
   215→        this.dustBurst(6 + Math.round(i * 6));
   216→      },
   217→      onCleanLand: (at) => {
   218→        gameStore.getState().pushPopup("cleanland", `CLEAN LANDING +${Math.round(Math.min(30, at * 16))}`);
   219→      },

   221→        this.bigAirCount++;

   223→      },
   224→      onDriftBank: (mult) => {
   225→        gameStore.getState().pushPopup("driftbank", `DRIFT CHAIN x${mult.toFixed(1)}`);
   226→        this.audio.driftBank(mult);
   227→      },
   228→      onNearMiss: () => {
   229→        gameStore.getState().pushPopup("nearmiss", "NEAR MISS +9");
   230→        this.audio.nearMiss();
   231→        this.trauma = Math.min(1, this.trauma + 0.08);
   232→      },
   233→      onWallRideStart: () => {
   234→        gameStore.getState().pushPopup("wallride", "WALL RIDE +");
   235→      },
   236→      onCrash: (i) => {
   237→        this.audio.crash(i);
   238→        this.trauma = Math.min(1, this.trauma + 0.25 + i * 0.45);
   239→        this.input.rumble(0.4 + i * 0.6, 200);
   240→        const p = this.physics.pos;
   241→        this.particles.emitSparks(p.x, p.y, p.z, -Math.sin(this.physics.yaw), -Math.cos(this.physics.yaw), 8 + Math.round(i * 10));
   242→      },
   243→      onBoostStart: () => {
   244→        this.audio.boostStart();
   245→        this.input.rumble(0.25, 90);
   246→      },
   247→      onBoxHit: (box) => {
   248→        // the whole stack / barricade shatters — chain-break every piece of
   249→        // the cluster, with a tiny stagger so the break ripples outward
   250→        const v = this.physics.vel;
   251→        const sp = Math.hypot(v.x, v.z);
   252→        const dirX = sp > 2 ? v.x / sp : Math.sin(this.physics.yaw);
   253→        const dirZ = sp > 2 ? v.z / sp : Math.cos(this.physics.yaw);
   254→        const power = clamp(6 + sp * 0.55, 7, 24);
   255→        let n = 0;
   256→        for (const b of this.track.boxes) {
   257→          if (b.cluster !== box.cluster || (!b.active && b !== box)) continue;
   258→          this.smashBox(b, dirX, dirZ, power, n * 0.045);
   259→          n++;
   260→        }
   261→        if (box.wall) {
   262→          this.audio.wallSmash();
   263→          this.trauma = Math.min(1, this.trauma + 0.38);
   264→          this.input.rumble(0.55, 240);
   265→          this.physics.boostMeter = clamp(this.physics.boostMeter + 15, 0, 100);
   266→          gameStore.getState().pushPopup("smash", "BREAK THROUGH! +15");
   267→        } else {
   268→          this.audio.boxHit();
   269→          this.trauma = Math.min(1, this.trauma + 0.16);
   270→          this.input.rumble(0.3, 130);
   271→          if (n > 1) {
   272→            this.physics.boostMeter = clamp(this.physics.boostMeter + 6, 0, 100);
   273→            gameStore.getState().pushPopup("smash", "CRATES SMASHED +6");
   274→          }
   275→        }
   276→      },
   277→      requestRecovery: () => this.startRecovery(),
   278→    });
   279→
   280→    this.loadBest();
   281→    this.physics.place(this.track.N - 10, 0, 0);
   282→    this.syncCarVisual(0.016);
   283→    this.updateSun();
   284→
   285→    this.resize();
   286→    // prewarm every shader while the loading overlay is still up — the first
   287→    // real frame then paints instantly instead of hitching on compilation
   288→    this.renderer.compile(this.scene, this.camera);
   289→    this.lastT = performance.now();
   290→    this.loop();
   291→  }

   297→  private buildSky() {
   298→    const geo = new THREE.SphereGeometry(3900, 32, 16);
   299→    const mat = new THREE.ShaderMaterial({
   300→      side: THREE.BackSide,
   301→      depthWrite: false,
   302→      fog: false,
   303→      uniforms: {
   304→        uZen: { value: new THREE.Color(PALETTE.skyZenith) },
   305→        uMid: { value: new THREE.Color(PALETTE.skyMid) },
   306→        uHor: { value: new THREE.Color(PALETTE.skyHorizon) },
   307→        uSun: { value: SUN_DIR.clone() },
   308→      },
   309→      vertexShader: `
   310→        varying vec3 vW;
   311→        void main() {
   312→          vW = (modelMatrix * vec4(position, 1.0)).xyz;
   313→          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   314→        }
   315→      `,
   316→      fragmentShader: `
   317→        uniform vec3 uZen; uniform vec3 uMid; uniform vec3 uHor; uniform vec3 uSun;
   318→        varying vec3 vW;
   319→        void main() {
   320→          vec3 d = normalize(vW);
   321→          float h = d.y;
   322→          vec3 col = h > 0.16 ? mix(uMid, uZen, smoothstep(0.16, 0.6, h))
   323→                              : mix(uHor, uMid, smoothstep(-0.06, 0.16, h));
   324→          float sd = max(dot(d, uSun), 0.0);
   325→          col += vec3(1.0, 0.88, 0.62) * (pow(sd, 900.0) * 1.2 + pow(sd, 16.0) * 0.3 + pow(sd, 4.0) * 0.1);
   326→          gl_FragColor = vec4(col, 1.0);
   327→        }
   328→      `,
   329→    });
   330→    const sky = new THREE.Mesh(geo, mat);
   331→    sky.frustumCulled = false;
   332→    this.scene.add(sky);
   333→  }

   353→    }

   365→  applySettings(s: GameSettings) {
   366→    const prevShadows = this.settings.shadows;
   367→    this.settings = { ...s };
   368→    this.physics.assist = s.assist;
   369→    this.particles.setHighDensity(s.highParticles);
   370→    this.debris.setHighDensity(s.highParticles);
   371→    this.audio.setVolumes(s.musicVol, s.sfxVol);
   372→    this.input.setMouseSteer(s.mouseSteer);
   373→    // difficulty: crate traffic, rival pace, boost economy
   374→    const diff = DIFFICULTIES[s.difficulty] ?? DIFFICULTIES.medium;
   375→    this.track.setActiveStacks(diff.stacks);
   376→    this.physics.boostScale = diff.boostGain;
   377→    for (const op of this.opponents) op.skillScale = diff.aiSkill;
   378→    if (prevShadows !== s.shadows) {
   379→      this.renderer.shadowMap.enabled = s.shadows;
   380→      this.sun.castShadow = s.shadows;
   381→      this.scene.traverse((o) => {
   382→        const mesh = o as THREE.Mesh;
   383→        if (mesh.isMesh) {
   384→          const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   385→          for (const m of mats) (m as THREE.Material).needsUpdate = true;
   386→        }
   387→      });
   388→    }

   424→  /** 2×2 start grid — two rivals on the front row, one alongside the player. */
   425→  private placeRaceGrid() {
   426→    const t = this.track;
   427→    this.physics.place(t.N - 15, 2.6, 0);
   428→    this.playerStartS = this.physics.progressS;
   429→    for (let i = 0; i < this.opponents.length; i++) {
   430→      const idx = i < 2 ? t.N - 8 : t.N - 15;
   431→      this.opponents[i].place(idx, OPPONENT_GRID_LAT[i], 0);
   432→    }
   433→    this.syncCarVisual(0.016);
   434→  }

   436→  /** Drop the rivals around the loop so the title flyover feels alive. */
   437→  private scatterOpponentsOnTrack() {
   438→    const t = this.track;
   439→    for (let i = 0; i < this.opponents.length; i++) {
   440→      this.opponents[i].place(
   441→        wrapI(t.N - 40 - i * 110, t.N),
   442→        (i % 2 === 0 ? -1 : 1) * 2.2,
   443→        22
   444→      );
   445→    }
   446→  }

   509→  // ============================================================
   510→  // main loop
   511→  // ============================================================
   512→
   513→  private loop = () => {
   514→    if (this.disposed) return;
   515→    this.raf = requestAnimationFrame(this.loop);
   516→    if (this.debugSelftestActive) return; // selftest drives physics deterministically
   517→    const now = performance.now();
   518→    let dt = (now - this.lastT) / 1000;
   519→    this.lastT = now;
   520→    dt = clamp(dt, 0.0005, 1 / 30);
   521→    if (this.paused) dt = 0;
   522→
   523→    this.time += dt;
   524→
   525→    const input = this.input.sample(Math.max(dt, 0.0001));
   526→    this.curInput = input;
   527→    // longitudinal accel estimate for body pitch
   528→    const sp = this.physics.speed;
   529→    if (dt > 0.0001) {
   530→      this.lastAccel = damp(this.lastAccel, (sp - this.prevSpeed) / dt, 6, dt);
   531→    }
   532→    this.prevSpeed = sp;
   533→    this.noteInput(input.steer, input.brake, this.lastAccel);
   534→
   535→    if (this.phase === "countdown") this.updateCountdown(dt);
   536→    else if (this.phase === "racing") this.updateRacing(dt, input);
   537→    else if (this.phase === "finished") this.updateFinished(dt);
   538→    else this.updateTitle(dt);
   539→
   540→    if (this.recoverT > 0) this.recoverT -= dt;
   541→
   542→    this.updateCamera(dt);
   543→    this.updateSun();
   544→    this.particles.update(dt, this.time);

   656→  private gateAbs(k: number): number {
   657→    const t = this.track;
   658→    const gate = t.gates[k];
   659→    if (!gate) return Infinity;
   660→    return this.lineAbs + (gate.isFinish ? t.length : gate.idx * t.step);

   663→  private checkGates(prevS: number) {
   664→    const t = this.track;
   665→    while (this.physics.progressS >= this.nextGateAbs) {
   666→      const gate = t.gates[this.nextGateK];
   667→      if (!gate) break;
   668→      const latOk = Math.abs(this.physics.lateral) < t.sample(gate.idx).hw + 10;
   669→      const prevOk = prevS <= this.nextGateAbs + 1;
   670→      if (!latOk || !prevOk) break;
   671→      if (gate.isFinish) {
   672→        this.completeLap();
   673→      } else {
   674→        const lapMs = this.raceTime - this.lapStart;
   675→        this.splits.push(lapMs);
   676→        const best = this.bestSplits[this.splits.length - 1];
   677→        const delta = best != null ? lapMs - best : null;
   678→        gameStore.getState().pushPopup("checkpoint", `CHECKPOINT ${this.splits.length}`, { deltaMs: delta ?? undefined });
   679→        this.physics.boostMeter = clamp(this.physics.boostMeter + 12, 0, 100);
   680→        this.audio.checkpoint(delta);
   681→        this.nextGateK++;
   682→        this.nextGateAbs = this.gateAbs(this.nextGateK);
   683→      }
   684→    }
   685→  }

   687→  private completeLap() {
   688→    const lapMs = this.raceTime - this.lapStart;
   689→    this.lapTimes.push(lapMs);
   690→    const lapIsBest = this.bestLapMs == null || lapMs < this.bestLapMs;
   691→    if (lapIsBest) {
   692→      this.bestLapMs = lapMs;
   693→      if (this.ghostRec.length > 10) this.ghost = this.ghostRec;
   694→    }
   695→    this.lapsDone++;
   696→    const store = gameStore.getState();
   697→    if (this.lapsDone >= this.totalLaps) {
   698→      // ---- finish ----
   699→      const totalMs = this.raceTime;
   700→      const bestLap = Math.min(...this.lapTimes);
   701→      const isNewRecord = this.bestTotalMs == null || totalMs < this.bestTotalMs;
   702→      if (isNewRecord) {
   703→        this.bestTotalMs = totalMs;
   704→        this.bestSplits = [...this.splits];
   705→      }
   706→      try {
   707→        localStorage.setItem(
   708→          "canyonrush.best.v1",
   709→          JSON.stringify({
   710→            totalMs: this.bestTotalMs,
   711→            bestLapMs: this.bestLapMs,
   712→            splits: this.bestSplits,
   713→            laps: this.totalLaps,
   714→          })
   715→        );
   716→        if (this.ghost.length > 10) {
   717→          localStorage.setItem(GHOST_KEY, JSON.stringify(this.ghost));
   718→        }
   719→      } catch {
   720→        /* ignore */
   721→      }
   722→      if (isNewRecord) {
   723→        store.saveBest({ totalMs, bestLapMs: this.bestLapMs, splits: this.bestSplits, laps: this.totalLaps });
   724→        store.pushPopup("newrecord", "NEW RECORD!");
   725→      }
   726→      this.audio.finish(isNewRecord);
   727→      this.playerFinishMs = totalMs;
   728→      this.pushStandings(true);
   729→      const position = this.playerPos;
   730→      this.setPhase("finished");
   731→      this.finishOrbitT = 0;
   732→      this.ghostCar.group.visible = false;
   733→      // CRITICAL: advance past the finish gate so checkGates' while-loop
   734→      // terminates instead of re-triggering completeLap forever (tab freeze)
   735→      this.nextGateK = this.track.gates.length;
   736→      this.nextGateAbs = Infinity;
   737→      this.cbs.onFinish({
   738→        totalMs,
   739→        bestLapMs: bestLap,
   740→        topSpeedKmh: Math.round(this.physics.topSpeed),
   741→        driftScore: Math.round(this.physics.driftScore),
   742→        nearMisses: this.physics.nearMisses,
   743→        bigAirCount: this.bigAirCount,
   744→        laps: this.totalLaps,
   745→        isNewRecord,
   746→        splits: [...this.splits],
   747→        position,
   748→        totalCars: 1 + this.opponents.length,
   749→      });
   750→      return;
   751→    }
   752→    // lap complete, more to go
   753→    store.pushPopup("lap", `LAP ${this.lapsDone}/${this.totalLaps} — ${fmt(lapMs)}`);
   754→    this.audio.checkpoint(-1);
   755→    this.lapStart = this.raceTime;
   756→    this.lineAbs += this.track.length;
   757→    this.nextGateK = 0;
   758→    this.nextGateAbs = this.gateAbs(0);
   759→    this.ghostRec = [];
   760→    this.ghostRecT = 0;
   761→    this.ghostCar.group.visible = this.ghost.length > 0;
   762→  }

   801→  /** Soft sphere collisions between all cars (player + rivals). */
   802→  private resolveCarCollisions() {
   803→    const cars: CarPhysics[] = [this.physics];
   804→    for (const op of this.opponents) cars.push(op.physics);
   805→    const R = 2.55;
   806→    const R2 = R * R;
   807→    for (let i = 0; i < cars.length; i++) {
   808→      for (let j = i + 1; j < cars.length; j++) {
   809→        const a = cars[i];
   810→        const b = cars[j];
   811→        if (Math.abs(a.pos.y - b.pos.y) > 2.2) continue;
   812→        const dx = b.pos.x - a.pos.x;
   813→        const dz = b.pos.z - a.pos.z;
   814→        const d2 = dx * dx + dz * dz;
   815→        if (d2 >= R2 || d2 < 1e-6) continue;
   816→        const d = Math.sqrt(d2);
   817→        const nx = dx / d;
   818→        const nz = dz / d;
   819→        const pen = (R - d) * 0.5;
   820→        a.pos.x -= nx * pen;
   821→        a.pos.z -= nz * pen;
   822→        b.pos.x += nx * pen;
   823→        b.pos.z += nz * pen;
   824→        const rvx = b.vel.x - a.vel.x;
   825→        const rvz = b.vel.z - a.vel.z;
   826→        const vn = rvx * nx + rvz * nz;
   827→        if (vn < 0) {
   828→          // equal-mass bump with a bit of bounce
   829→          const imp = -vn * 0.7;
   830→          a.vel.x -= nx * imp;
   831→          a.vel.z -= nz * imp;
   832→          b.vel.x += nx * imp;
   833→          b.vel.z += nz * imp;
   834→          if ((a.isPlayer || b.isPlayer) && -vn > 4.5 && this.time - this.lastBumpT > 0.6) {
   835→            this.lastBumpT = this.time;
   836→            const impact = -vn;
   837→            this.audio.crash(clamp(impact / 26, 0.12, 0.45));
   838→            this.trauma = Math.min(1, this.trauma + 0.1 + impact * 0.012);
   839→            this.input.rumble(clamp(impact / 20, 0.15, 0.6), 110);
   840→          }
   841→        }
   842→      }
   843→    }
   844→  }

   914→  private syncCarVisual(dt: number) {

   957→  private emitEffects(dt: number) {
   958→    const p = this.physics;
   959→    const fwd = this.tmpV.set(Math.sin(p.yaw), 0, Math.cos(p.yaw));
   960→    const right = this.tmpV2.set(-Math.cos(p.yaw), 0, Math.sin(p.yaw));
   961→
   962→    const speedH = p.speed;
   963→
   964→    if (p.grounded) {
   965→      // drift smoke + skids
   966→      if (p.drifting && speedH > 8) {
   967→        const q = clamp(Math.abs(p.slip) / 0.5, 0.2, 1);
   968→        for (const side of [-1, 1]) {
   969→          const wx = p.pos.x - fwd.x * 1.25 + right.x * side * 0.85;
   970→          const wz = p.pos.z - fwd.z * 1.25 + right.z * side * 0.85;
   971→          const wy = p.pos.y;
   972→          this.particles.emitSmoke(wx, wy, wz, q);
   973→        }
   974→        this.skidTimer -= dt;
   975→        if (this.skidTimer <= 0) {
   976→          this.skidTimer = 0.03;
   977→          const vx = p.vel.x / Math.max(1, speedH);
   978→          const vz = p.vel.z / Math.max(1, speedH);
   979→          for (const side of [-1, 1]) {
   980→            const wx = p.pos.x - fwd.x * 1.25 + right.x * side * 0.85;
   981→            const wz = p.pos.z - fwd.z * 1.25 + right.z * side * 0.85;
   982→            this.particles.emitSkid(wx, p.pos.y, wz, vx, vz, clamp(Math.abs(p.slip) * 2, 0.3, 1), this.time);
   983→          }
   984→        }
   985→      }
   986→      // off-road dust
   987→      if (p.offRoad && speedH > 9) {
   988→        this.particles.emitDust(
   989→          p.pos.x - fwd.x * 1.4, p.pos.y, p.pos.z - fwd.z * 1.4,
   990→          clamp(speedH / 30, 0.3, 1.2), p.vel.x, p.vel.z
   991→        );
   992→      }
   993→      // wall-ride sparks
   994→      if (p.wallRiding) {
   995→        this.sparkTimer -= dt;
   996→        if (this.sparkTimer <= 0) {
   997→          this.sparkTimer = 0.06;
   998→          const side = p.lateral > 0 ? 1 : -1;
   999→          this.particles.emitSparks(
  1000→            p.pos.x + right.x * side * 1.1, p.pos.y + 0.2, p.pos.z + right.z * side * 1.1,
  1001→            -right.x * side, -right.z * side, 3
  1002→          );
  1003→        }
  1004→      }
  1005→    }
  1006→    // boost trail
  1007→    if (p.boosting) {
  1008→      this.boostTrailTimer -= dt;
  1009→      if (this.boostTrailTimer <= 0) {
  1010→        this.boostTrailTimer = 0.02;
  1011→        this.particles.emitBoost(p.pos.x - fwd.x * 2.0, p.pos.y + 0.45, p.pos.z - fwd.z * 2.0, -fwd.x, -fwd.z);
  1012→      }
  1013→    }
  1014→  }

  1032→      const passed = this.phase === "racing" && k < this.nextGateK;
  1033→      if (isNext) {
  1034→        g.glow.emissiveIntensity = 1.6 + Math.sin(this.time * 5) * 0.6;
  1035→      } else {
  1036→        g.glow.emissiveIntensity = passed ? 0.5 : 1.1;

  1038→    }

  1055→  private playGhost() {
  1056→    if (!this.ghost.length) {
  1057→      this.ghostCar.group.visible = false;
  1058→      return;
  1059→    }
  1060→    this.ghostCar.group.visible = true;
  1061→    const tMs = this.raceTime;
  1062→    const frames = this.ghost;
  1063→    // binary search
  1064→    let lo = 0;
  1065→    let hi = frames.length - 1;
  1066→    while (lo < hi) {
  1067→      const mid = (lo + hi) >> 1;
  1068→      if (frames[mid].t < tMs) lo = mid + 1;
  1069→      else hi = mid;
  1070→    }
  1071→    const i = Math.max(1, lo);
  1072→    const a = frames[i - 1];
  1073→    const b = frames[Math.min(frames.length - 1, i)];
  1074→    const span = Math.max(1e-3, b.t - a.t);
  1075→    const f = clamp((tMs - a.t) / span, 0, 1);
  1076→    const g = this.ghostCar.group;
  1077→    g.position.set(a.x + (b.x - a.x) * f, a.y + (b.y - a.y) * f, a.z + (b.z - a.z) * f);
  1078→    let dy = b.yaw - a.yaw;
  1079→    while (dy > Math.PI) dy -= Math.PI * 2;
  1080→    while (dy < -Math.PI) dy += Math.PI * 2;
  1081→    g.rotation.set(0, a.yaw + dy * f, 0);
  1082→  }

  1088→  private updateCamera(dt: number) {
  1089→    if (this.phase === "title") return; // handled in updateTitle
  1090→    const p = this.physics;
  1091→    const fwd = this.tmpV.set(Math.sin(p.yaw), 0, Math.cos(p.yaw));
  1092→
  1093→    if (this.phase === "countdown") {
  1094→      // cinematic sweep down to the chase position
  1095→      const t = clamp(1 - this.countdownT / 3, 0, 1);
  1096→      const e = t * t * (3 - 2 * t);
  1097→      const ang = 2.4 - e * 2.4;
  1098→      const dist = 30 - 24 * e;
  1099→      this.camPos.set(
  1100→        p.pos.x + Math.sin(ang) * dist + fwd.x * -4,
  1101→        p.pos.y + 14 - 11.5 * e,
  1102→        p.pos.z + Math.cos(ang) * dist + fwd.z * -4
  1103→      );
  1104→      this.camLook.set(p.pos.x + fwd.x * 6, p.pos.y + 1.2, p.pos.z + fwd.z * 6);
  1105→      this.camera.position.copy(this.camPos);
  1106→      this.camera.up.set(0, 1, 0);
  1107→      this.camera.lookAt(this.camLook);
  1108→      this.camera.fov = damp(this.camera.fov, 66, 3, dt);
  1109→      this.camera.updateProjectionMatrix();
  1110→      return;
  1111→    }
  1112→
  1113→    if (this.phase === "finished") {
  1114→      const a = this.finishOrbitT * 0.5;
  1115→      this.camPos.set(p.pos.x + Math.sin(a) * 9, p.pos.y + 3.2, p.pos.z + Math.cos(a) * 9);
  1116→      this.camLook.copy(p.pos).add(new THREE.Vector3(0, 1, 0));
  1117→      this.camera.position.copy(this.camPos);
  1118→      this.camera.up.set(0, 1, 0);
  1119→      this.camera.lookAt(this.camLook);
  1120→      return;
  1121→    }
  1122→
  1123→    // chase camera
  1124→    const mode = this.settings.cameraMode;
  1125→    const dist = mode === 1 ? 9.6 : 6.4;
  1126→    const height = mode === 1 ? 3.8 : 2.55;
  1127→    if (mode === 2) {
  1128→      // hood cam
  1129→      this.camPos.set(p.pos.x + fwd.x * 0.5, p.pos.y + 1.35, p.pos.z + fwd.z * 0.5);
  1130→      this.camLook.set(p.pos.x + fwd.x * 12, p.pos.y + 1.6, p.pos.z + fwd.z * 12);
  1131→      this.camera.position.copy(this.camPos);
  1132→      this.camera.up.set(0, 1, 0);
  1133→      this.camera.lookAt(this.camLook);
  1134→    } else {
  1135→      const ffX = clamp(p.vel.x * 0.16, -7, 7);
  1136→      const ffZ = clamp(p.vel.z * 0.16, -7, 7);
  1137→      const tx = p.pos.x - fwd.x * dist + ffX;
  1138→      const ty = p.pos.y + height;
  1139→      const tz = p.pos.z - fwd.z * dist + ffZ;
  1140→      const rate = 6.5;
  1141→      this.camPos.x = damp(this.camPos.x, tx, rate, dt);
  1142→      this.camPos.y = damp(this.camPos.y, ty, rate, dt);
  1143→      this.camPos.z = damp(this.camPos.z, tz, rate, dt);
  1144→      this.camLook.set(p.pos.x + fwd.x * 7, p.pos.y + 1.3, p.pos.z + fwd.z * 7);
  1145→      this.camera.position.copy(this.camPos);
  1146→      this.camera.up.set(0, 1, 0);
  1147→      this.camera.lookAt(this.camLook);
  1148→      // banking roll (positive rotateZ banks right, matching steer > 0)
  1149→      const targetRoll = clamp(-p.slip * 0.3 + this.lastSteer * 0.05, -0.24, 0.24);
  1150→      this.camRoll = damp(this.camRoll, targetRoll, 5, dt);
  1151→      this.camera.rotateZ(this.camRoll);
  1152→    }
  1153→
  1154→    // wall-ride lean
  1155→    this.wallRideLean = damp(this.wallRideLean, p.wallRiding ? clamp(p.lateral / 24, 0, 1) * 0.5 : 0, 4, dt);
  1156→    if (this.wallRideLean > 0.01) this.camera.rotateZ(this.wallRideLean);
  1157→
  1158→    // FOV kick
  1159→    const speedFrac = clamp(p.speed / 55, 0, 1.1);
  1160→    const targetFov = 64 + speedFrac * 10 + (p.boosting ? 9 : 0);
  1161→    this.fov = damp(this.fov, targetFov, 4, dt);
  1162→    this.camera.fov = this.fov;
  1163→    this.camera.updateProjectionMatrix();
  1164→
  1165→    // restrained screen shake
  1166→    this.trauma = Math.max(0, this.trauma - dt * 1.6);
  1167→    const sh = this.trauma * this.trauma;
  1168→    if (sh > 0.001) {
  1169→      this.camera.position.x += (Math.random() - 0.5) * sh * 0.55;
  1170→      this.camera.position.y += (Math.random() - 0.5) * sh * 0.4;
  1171→      this.camera.position.z += (Math.random() - 0.5) * sh * 0.55;
  1172→      this.camera.rotateZ((Math.random() - 0.5) * sh * 0.03);
  1173→    }
  1174→  }
