   130→  constructor(
   131→    canvas: HTMLCanvasElement,
   132→    audio: AudioSystem,
   133→    input: InputManager,
   134→    cbs: EngineCallbacks
   135→  ) {
   136→    this.audio = audio;
   137→    this.input = input;
   138→    this.cbs = cbs;
   139→
   140→    this.renderer = new THREE.WebGLRenderer({
   141→      canvas,
   142→      antialias: true,
   143→      powerPreference: "high-performance",
   144→    });
   145→    const isMobile =
   146→      typeof window !== "undefined" && window.matchMedia("(pointer: coarse)").matches;
   147→    this.renderer.setPixelRatio(Math.min(window.devicePixelRatio || 1, isMobile ? 1.5 : 2));
   148→    this.renderer.shadowMap.enabled = true;
   149→    this.renderer.shadowMap.type = THREE.PCFShadowMap;
   150→    this.renderer.toneMapping = THREE.ACESFilmicToneMapping;
   151→    this.renderer.toneMappingExposure = 1.05;
   152→    this.renderer.outputColorSpace = THREE.SRGBColorSpace;
   153→
   154→    this.camera = new THREE.PerspectiveCamera(66, 16 / 9, 0.3, 4200);
   155→    // generous fog band — the road ahead must stay readable at racing speed
   156→    this.scene.fog = new THREE.Fog(PALETTE.fog, 430, 1500);
   157→
   158→    // sky
   159→    this.buildSky();
   160→
   161→    // lights
   162→    const hemi = new THREE.HemisphereLight(0xffe9c9, 0xcaa06a, 0.8);
   163→    this.scene.add(hemi);
   164→    this.sun = new THREE.DirectionalLight(0xffd9a8, 1.8);
   165→    this.sun.castShadow = !isMobile;
   166→    this.sun.shadow.mapSize.set(isMobile ? 1024 : 2048, isMobile ? 1024 : 2048);
   167→    const sc = this.sun.shadow.camera;
   168→    sc.left = -95;
   169→    sc.right = 95;
   170→    sc.top = 95;
   171→    sc.bottom = -95;
   172→    sc.near = 20;
   173→    sc.far = 420;
   174→    this.sun.shadow.bias = -0.0006;
   175→    this.sun.shadow.normalBias = 0.5;
   176→    this.scene.add(this.sun);
   177→    this.scene.add(this.sun.target);
   178→
   179→    // world
   180→    this.track = new Track();
   181→    this.scene.add(this.track.group);
   182→    this.scene.add(this.particles.group);
   183→    this.debris = new DebrisSystem(isMobile);
   184→    this.scene.add(this.debris.group);
   185→
   186→    // cars
   187→    this.car = new CarVisual(false);
   188→    this.scene.add(this.car.group);
   189→    this.ghostCar = new CarVisual(true);
   190→    this.ghostCar.group.visible = false;
   191→    this.scene.add(this.ghostCar.group);
   192→
   193→    // AI rivals (identical performance, different paint & skill)
   194→    for (let i = 0; i < OPPONENTS.length; i++) {
   195→      const op = new Opponent(OPPONENTS[i], this.track, OPPONENT_GRID_LAT[i]);
   196→      // rival crate smashes shatter too (debris + a soft distant crack)
   197→      op.boxHitHook = (box) => {
   198→        const v = op.physics.vel;
   199→        const sp = Math.hypot(v.x, v.z);
   200→        const dirX = sp > 2 ? v.x / sp : 0;
   201→        const dirZ = sp > 2 ? v.z / sp : 1;
   202→        const power = clamp(6 + sp * 0.5, 7, 20);
   203→        let n = 0;
   204→        for (const b of this.track.boxes) {
   205→          if (b.cluster !== box.cluster || (!b.active && b !== box)) continue;
   206→          this.smashBox(b, dirX, dirZ, power, n * 0.045);
   207→          n++;
   208→        }
   209→        const d = box.x - this.physics.pos.x;
   210→        const dz = box.z - this.physics.pos.z;
   211→        if (d * d + dz * dz < 70 * 70) {
   212→          if (box.wall) this.audio.wallSmash();
   213→          else this.audio.boxHit();
   214→        }
   215→      };
   216→      this.scene.add(op.visual.group);
   217→      this.opponents.push(op);
   218→    }
   219→    this.scatterOpponentsOnTrack();
   220→
   221→    this.physics = new CarPhysics(this.track, {
   222→      onJump: () => this.audio.jump(),
   223→      onLanding: (i, clean) => {
   224→        this.audio.landing(clean, i);
   225→        this.trauma = Math.min(1, this.trauma + 0.18 + i * 0.3);
   226→        this.input.rumble(0.2 + i * 0.5, 120);
   227→        this.dustBurst(6 + Math.round(i * 6));
   228→      },
   229→      onCleanLand: (at) => {
   230→        gameStore.getState().pushPopup("cleanland", `CLEAN LANDING +${Math.round(Math.min(30, at * 16))}`);
   231→      },
   232→      onBigAir: (at) => {
   233→        this.bigAirCount++;
   234→        gameStore.getState().pushPopup("airtime", `BIG AIR ${at.toFixed(1)}s +${Math.round(Math.min(30, at * 16))}`);
   235→      },
   236→      onDriftBank: (mult) => {
   237→        gameStore.getState().pushPopup("driftbank", `DRIFT CHAIN x${mult.toFixed(1)}`);
   238→        this.audio.driftBank(mult);
   239→      },
   240→      onNearMiss: () => {
   241→        gameStore.getState().pushPopup("nearmiss", "NEAR MISS +9");
   242→        this.audio.nearMiss();
   243→        this.trauma = Math.min(1, this.trauma + 0.08);
   244→      },
   245→      onWallRideStart: () => {
   246→        gameStore.getState().pushPopup("wallride", "WALL RIDE +");
   247→      },
   248→      onCrash: (i) => {
   249→        this.audio.crash(i);
   250→        this.trauma = Math.min(1, this.trauma + 0.25 + i * 0.45);
   251→        this.input.rumble(0.4 + i * 0.6, 200);
   252→        const p = this.physics.pos;
   253→        this.particles.emitSparks(p.x, p.y, p.z, -Math.sin(this.physics.yaw), -Math.cos(this.physics.yaw), 8 + Math.round(i * 10));
   254→      },
   255→      onBoostStart: () => {
   256→        this.audio.boostStart();
   257→        this.input.rumble(0.25, 90);
   258→      },
   259→      onBoxHit: (box) => {
   260→        // the whole stack / barricade shatters — chain-break every piece of
   261→        // the cluster, with a tiny stagger so the break ripples outward
   262→        const v = this.physics.vel;
   263→        const sp = Math.hypot(v.x, v.z);
   264→        const dirX = sp > 2 ? v.x / sp : Math.sin(this.physics.yaw);
   265→        const dirZ = sp > 2 ? v.z / sp : Math.cos(this.physics.yaw);
   266→        const power = clamp(6 + sp * 0.55, 7, 24);
   267→        let n = 0;
   268→        for (const b of this.track.boxes) {
   269→          if (b.cluster !== box.cluster || (!b.active && b !== box)) continue;
   270→          this.smashBox(b, dirX, dirZ, power, n * 0.045);
   271→          n++;
   272→        }
   273→        if (box.wall) {
   274→          this.audio.wallSmash();
   275→          this.trauma = Math.min(1, this.trauma + 0.38);
   276→          this.input.rumble(0.55, 240);
   277→          const add = Math.round(15 * this.physics.boostScale);
   278→          this.physics.boostMeter = clamp(this.physics.boostMeter + add, 0, 100);
   279→          gameStore.getState().pushPopup("smash", `BREAK THROUGH! +${add}`);
   280→        } else {
   281→          this.audio.boxHit();
   282→          this.trauma = Math.min(1, this.trauma + 0.16);
   283→          this.input.rumble(0.3, 130);
   284→          if (n > 1) {
   285→            const add = Math.round(6 * this.physics.boostScale);
   286→            this.physics.boostMeter = clamp(this.physics.boostMeter + add, 0, 100);
   287→            gameStore.getState().pushPopup("smash", `CRATES SMASHED +${add}`);
   288→          }
   289→        }
   290→      },
   291→      requestRecovery: () => this.startRecovery(),
   292→    });
   293→
   294→    this.loadBest();
   295→    this.physics.place(this.track.N - 10, 0, 0);
   296→    this.syncCarVisual(0.016);
   297→    this.updateSun();
   298→
   299→    this.resize();
   300→    // prewarm every shader while the loading overlay is still up — the first
   301→    // real frame then paints instantly instead of hitching on compilation
   302→    this.renderer.compile(this.scene, this.camera);
   303→    this.lastT = performance.now();
   304→    this.loop();
   305→  }

   311→  private buildSky() {
   312→    const geo = new THREE.SphereGeometry(3900, 32, 16);
   313→    const mat = new THREE.ShaderMaterial({
   314→      side: THREE.BackSide,
   315→      depthWrite: false,
   316→      fog: false,
   317→      uniforms: {
   318→        uZen: { value: new THREE.Color(PALETTE.skyZenith) },
   319→        uMid: { value: new THREE.Color(PALETTE.skyMid) },
   320→        uHor: { value: new THREE.Color(PALETTE.skyHorizon) },
   321→        uSun: { value: SUN_DIR.clone() },
   322→      },
   323→      vertexShader: `
   324→        varying vec3 vW;
   325→        void main() {
   326→          vW = (modelMatrix * vec4(position, 1.0)).xyz;
   327→          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   328→        }
   329→      `,
   330→      fragmentShader: `
   331→        uniform vec3 uZen; uniform vec3 uMid; uniform vec3 uHor; uniform vec3 uSun;
   332→        varying vec3 vW;
   333→        void main() {
   334→          vec3 d = normalize(vW);
   335→          float h = d.y;
   336→          vec3 col = h > 0.16 ? mix(uMid, uZen, smoothstep(0.16, 0.6, h))
   337→                              : mix(uHor, uMid, smoothstep(-0.06, 0.16, h));
   338→          float sd = max(dot(d, uSun), 0.0);
   339→          col += vec3(1.0, 0.88, 0.62) * (pow(sd, 900.0) * 1.2 + pow(sd, 16.0) * 0.3 + pow(sd, 4.0) * 0.1);
   340→          gl_FragColor = vec4(col, 1.0);
   341→        }
   342→      `,
   343→    });
   344→    const sky = new THREE.Mesh(geo, mat);
   345→    sky.frustumCulled = false;
   346→    this.scene.add(sky);
   347→  }

   349→  private loadBest() {
   350→    try {
   351→      const raw = localStorage.getItem("canyonrush.best.v1");
   352→      if (raw) {
   353→        const d = JSON.parse(raw) as {
   354→          totalMs: number;
   355→          bestLapMs: number;
   356→          splits: number[];
   357→          laps: number;
   358→        };
   359→        this.bestTotalMs = d.totalMs;
   360→        this.bestLapMs = d.bestLapMs;
   361→        this.bestSplits = d.splits ?? [];
   362→      }
   363→      const graw = localStorage.getItem(GHOST_KEY);
   364→      if (graw) this.ghost = JSON.parse(graw) as GhostFrame[];
   365→    } catch {
   366→      /* ignore */
   367→    }
   368→  }

   379→  applySettings(s: GameSettings) {
   380→    const prevShadows = this.settings.shadows;
   381→    this.settings = { ...s };
   382→    // HARD strips the safety net — no auto-correct, no anti-spin
   383→    // (multiplayer always uses the player's own assist preference)
   384→    this.physics.assist = this.mpActive
   385→      ? s.assist
   386→      : s.difficulty === "hard"
   387→        ? false
   388→        : s.assist;
   389→    this.particles.setHighDensity(s.highParticles);
   390→    this.debris.setHighDensity(s.highParticles);
   391→    this.audio.setVolumes(s.musicVol, s.sfxVol);
   392→    this.input.setMouseSteer(s.mouseSteer);
   393→    // difficulty: crate traffic, rival pace/brain, boost economy
   394→    // (multiplayer races use fixed MEDIUM handling so every car is equal)
   395→    const diff = this.mpActive
   396→      ? DIFFICULTIES.medium
   397→      : DIFFICULTIES[s.difficulty] ?? DIFFICULTIES.medium;
   398→    this.track.setActiveStacks(diff.stacks);
   399→    this.physics.boostScale = diff.boostGain;
   400→    for (const op of this.opponents) {
   401→      op.brain = {
   402→        skillScale: diff.aiSkill,
   403→        mistakeEvery: diff.aiMistakeEvery,
   404→        rubberAhead: diff.rubberAhead,
   405→        rubberBehind: diff.rubberBehind,
   406→        boostAggro: diff.aiBoostAggro,
   407→        blocking: diff.blocking,
   408→      };
   409→      op.physics.speedCapScale = diff.aiSpeedCap;
   410→    }
   411→    if (prevShadows !== s.shadows) {
   412→      this.renderer.shadowMap.enabled = s.shadows;
   413→      this.sun.castShadow = s.shadows;
   414→      this.scene.traverse((o) => {
   415→        const mesh = o as THREE.Mesh;
   416→        if (mesh.isMesh) {
   417→          const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   418→          for (const m of mats) (m as THREE.Material).needsUpdate = true;
   419→        }
   420→      });
   421→    }
   422→  }

   424→  startRace(laps: number) {
   425→    this.totalLaps = laps;
   426→    this.loadBest();
   427→    this.lapsDone = 0;
   428→    this.raceTime = 0;
   429→    this.lapStart = 0;
   430→    this.splits = [];
   431→    this.lapTimes = [];
   432→    this.bigAirCount = 0;
   433→    this.physics.boostMeter = (DIFFICULTIES[this.settings.difficulty] ?? DIFFICULTIES.medium)
   434→      .startBoost;
   435→    this.physics.driftScore = 0;
   436→    this.physics.nearMisses = 0;
   437→    this.physics.topSpeed = 0;
   438→    this.placeRaceGrid();
   439→    // fresh crates for the new race (respecting the difficulty limit)
   440→    this.track.setActiveStacks((DIFFICULTIES[this.settings.difficulty] ?? DIFFICULTIES.medium).stacks);
   441→    this.particles.clearSkids(this.time);
   442→    this.ghostRec = [];
   443→    this.ghostRecT = 0;
   444→    this.ghostCar.group.visible = this.ghost.length > 0;
   445→    this.countdownT = 3;
   446→    this.countdownShown = -99;
   447→    this.recoverT = 0;
   448→    this.wallWarned.clear();
   449→    this.lineAbs = this.track.length; // first line crossing = one length from spawn
   450→    this.nextGateK = 0;
   451→    this.nextGateAbs = this.gateAbs(0);
   452→    this.playerFinishMs = null;
   453→    this.lastStandingsKey = "";
   454→    this.pushStandings(true);
   455→    this.setPhase("countdown");
   456→    this.audio.startMusic();
   457→  }

   459→  /** 2×2 start grid — two rivals on the front row, one alongside the player. */
   460→  private placeRaceGrid() {
   461→    const t = this.track;
   462→    this.physics.place(t.N - 15, 2.6, 0);
   463→    this.playerStartS = this.physics.progressS;
   464→    for (let i = 0; i < this.opponents.length; i++) {
   465→      const idx = i < 2 ? t.N - 8 : t.N - 15;
   466→      this.opponents[i].place(idx, OPPONENT_GRID_LAT[i], 0);
   467→    }
   468→    this.syncCarVisual(0.016);
   469→  }
   470→
   471→  /** Drop the rivals around the loop so the title flyover feels alive. */
   472→  private scatterOpponentsOnTrack() {
   473→    const t = this.track;
   474→    for (let i = 0; i < this.opponents.length; i++) {
   475→      this.opponents[i].place(
   476→        wrapI(t.N - 40 - i * 110, t.N),
   477→        (i % 2 === 0 ? -1 : 1) * 2.2,
   478→        22
   479→      );
   480→    }
   481→  }

   507→  /** Enter/leave a multiplayer session: AI rivals are hidden and network
   508→   *  cars take their place. Single-player flow is untouched when inactive. */
   509→  setMultiplayerSession(active: boolean) {
   510→    if (this.mpActive === active) return;
   511→    this.mpActive = active;
   512→    if (!active) {
   513→      for (const rp of this.remotes) {
   514→        this.scene.remove(rp.visual.group);
   515→        rp.dispose();
   516→      }
   517→      this.remotes = [];
   518→      this.mpSpectating = false;
   519→      this.car.group.visible = true;
   520→    }
   521→    for (const op of this.opponents) op.visual.group.visible = !active;
   522→    // re-apply difficulty handling for the new mode (fair medium in MP)
   523→    this.applySettings(this.settings);
   524→  }
   525→
   526→  /** Late joiners / quitters watch the canyon flythrough instead of racing. */
   527→  setSpectating(v: boolean) {
   528→    if (this.mpSpectating === v) return;
   529→    this.mpSpectating = v;
   530→    this.car.group.visible = !v;
   531→    if (v) this.ghostCar.group.visible = false;
   532→  }

   688→  private loop = () => {
   689→    if (this.disposed) return;
   690→    this.raf = requestAnimationFrame(this.loop);
   691→    if (this.debugSelftestActive) return; // selftest drives physics deterministically
   692→    const now = performance.now();
   693→    let dt = (now - this.lastT) / 1000;
   694→    this.lastT = now;
   695→    dt = clamp(dt, 0.0005, 1 / 30);
   696→    if (this.paused) dt = 0;
   697→
   698→    this.time += dt;
   699→
   700→    const input = this.input.sample(Math.max(dt, 0.0001));
   701→    this.curInput = input;
   702→    // longitudinal accel estimate for body pitch
   703→    const sp = this.physics.speed;
   704→    if (dt > 0.0001) {
   705→      this.lastAccel = damp(this.lastAccel, (sp - this.prevSpeed) / dt, 6, dt);
   706→    }
   707→    this.prevSpeed = sp;
   708→    this.noteInput(input.steer, input.brake, this.lastAccel);
   709→
   710→    if (this.phase === "countdown") this.updateCountdown(dt);
   711→    else if (this.phase === "racing") this.updateRacing(dt, input);
   712→    else if (this.phase === "finished") this.updateFinished(dt);
   713→    else this.updateTitle(dt);
   714→
   715→    if (this.recoverT > 0) this.recoverT -= dt;
   716→
   717→    this.updateCamera(dt);
   718→    this.updateSun();
   719→    this.updateRemotes(dt);
   720→    this.particles.update(dt, this.time);
   721→    this.debris.update(dt);
   722→    this.updateGateGlow();
   723→    this.updateWallFx();
   724→    this.track.updateBoxes(this.time * 1000); // smashed crates come back
   725→    this.emitTelemetry();
   726→
   727→    // audio
   728→    const { rpm } = this.physics.gearAndRpm();
   729→    const intensity =
   730→      this.phase === "title"
   731→        ? 0
   732→        : this.phase === "countdown"
   733→          ? 1
   734→          : this.phase === "finished"
   735→            ? 1
   736→            : clamp(1 + (this.physics.speed > 26 ? 1 : 0) + (this.physics.boosting || this.physics.boostMeter > 82 ? 1 : 0), 0, 3);
   737→    this.audio.update(dt, {
   738→      rpm:
   739→        this.phase === "countdown"
   740→          ? 0.25 + input.throttle * 0.55
   741→          : this.phase === "racing" || this.phase === "finished"
   742→            ? rpm
   743→            : 0.22,
   744→      throttle: this.phase === "racing" ? input.throttle : this.phase === "countdown" ? input.throttle * 0.7 : 0,
   745→      speed: this.physics.speed,
   746→      driftIntensity: this.physics.drifting ? clamp(Math.abs(this.physics.slip) / 0.5, 0, 1) : 0,
   747→      boosting: this.physics.boosting && this.phase === "racing",
   748→      wallRide: this.physics.wallRiding,
   749→      musicIntensity: intensity,
   750→    });
   751→
   752→    // render directly to the canvas — keeps MSAA + tone mapping crisp
   753→    this.renderer.render(this.scene, this.camera);
   754→  };

   756→  private updateCountdown(dt: number) {
   757→    this.countdownT -= dt;
   758→    const v = Math.ceil(this.countdownT);
   759→    if (v !== this.countdownShown && v >= 0) {
   760→      this.countdownShown = v;
   761→      // multiplayer races pre-roll up to ~4s before GO — keep the classic
   762→      // 3·2·1 display and only beep inside the final three
   763→      this.cbs.onCountdown(Math.min(v, 3));
   764→      if (v <= 3) this.audio.countdownBeep(v === 0);
   765→    }
   766→    if (this.countdownT <= 0) {
   767→      this.setPhase("racing");
   768→      this.raceTime = 0;
   769→      this.lapStart = 0;
   770→    }

   835→  private gateAbs(k: number): number {
   836→    const t = this.track;
   837→    const gate = t.gates[k];
   838→    if (!gate) return Infinity;
   839→    return this.lineAbs + (gate.isFinish ? t.length : gate.idx * t.step);
   840→  }

   842→  private checkGates(prevS: number) {
   843→    const t = this.track;
   844→    while (this.physics.progressS >= this.nextGateAbs) {
   845→      const gate = t.gates[this.nextGateK];
   846→      if (!gate) break;
   847→      const latOk = Math.abs(this.physics.lateral) < t.sample(gate.idx).hw + 10;
   848→      const prevOk = prevS <= this.nextGateAbs + 1;
   849→      if (!latOk || !prevOk) break;
   850→      if (gate.isFinish) {
   851→        this.completeLap();
   852→      } else {
   853→        const lapMs = this.raceTime - this.lapStart;
   854→        this.splits.push(lapMs);
   855→        const best = this.bestSplits[this.splits.length - 1];
   856→        const delta = best != null ? lapMs - best : null;
   857→        gameStore.getState().pushPopup("checkpoint", `CHECKPOINT ${this.splits.length}`, { deltaMs: delta ?? undefined });
   858→        // HARD trims the checkpoint top-up along with every other boost gain
   859→        this.physics.boostMeter = clamp(
   860→          this.physics.boostMeter + 12 * this.physics.boostScale,
   861→          0,
   862→          100
   863→        );
   864→        this.audio.checkpoint(delta);
   865→        this.nextGateK++;
   866→        this.nextGateAbs = this.gateAbs(this.nextGateK);
   867→      }
   868→    }
   869→  }
   870→
   871→  private completeLap() {
   872→    const lapMs = this.raceTime - this.lapStart;
   873→    this.lapTimes.push(lapMs);
   874→    const lapIsBest = this.bestLapMs == null || lapMs < this.bestLapMs;
   875→    if (lapIsBest) {
   876→      this.bestLapMs = lapMs;
   877→      if (this.ghostRec.length > 10) this.ghost = this.ghostRec;
   878→    }
   879→    this.lapsDone++;
   880→    // multiplayer: report the lap for server-side validation
   881→    this.onMpLap?.(this.lapsDone, lapMs);
   882→    const store = gameStore.getState();
   883→    if (this.lapsDone >= this.totalLaps) {
   884→      // ---- finish ----
   885→      const totalMs = this.raceTime;
   886→      const bestLap = Math.min(...this.lapTimes);
   887→      const isNewRecord = this.bestTotalMs == null || totalMs < this.bestTotalMs;
   888→      if (isNewRecord) {
   889→        this.bestTotalMs = totalMs;
   890→        this.bestSplits = [...this.splits];
   891→      }
   892→      try {
   893→        localStorage.setItem(
   894→          "canyonrush.best.v1",
   895→          JSON.stringify({
   896→            totalMs: this.bestTotalMs,
   897→            bestLapMs: this.bestLapMs,
   898→            splits: this.bestSplits,
   899→            laps: this.totalLaps,
   900→          })
   901→        );
   902→        if (this.ghost.length > 10) {
   903→          localStorage.setItem(GHOST_KEY, JSON.stringify(this.ghost));
   904→        }
   905→      } catch {
   906→        /* ignore */
   907→      }
   908→      if (isNewRecord) {
   909→        store.saveBest({ totalMs, bestLapMs: this.bestLapMs, splits: this.bestSplits, laps: this.totalLaps });
   910→        store.pushPopup("newrecord", "NEW RECORD!");
   911→      }
   912→      this.audio.finish(isNewRecord);
   913→      this.playerFinishMs = totalMs;
   914→      this.pushStandings(true);
   915→      const position = this.playerPos;
   916→      this.setPhase("finished");
   917→      this.finishOrbitT = 0;
   918→      this.ghostCar.group.visible = false;
   919→      // CRITICAL: advance past the finish gate so checkGates' while-loop
   920→      // terminates instead of re-triggering completeLap forever (tab freeze)
   921→      this.nextGateK = this.track.gates.length;
   922→      this.nextGateAbs = Infinity;
   923→      this.cbs.onFinish({
   924→        totalMs,
   925→        bestLapMs: bestLap,
   926→        topSpeedKmh: Math.round(this.physics.topSpeed),
   927→        driftScore: Math.round(this.physics.driftScore),
   928→        nearMisses: this.physics.nearMisses,
   929→        bigAirCount: this.bigAirCount,
   930→        laps: this.totalLaps,
   931→        isNewRecord,
   932→        splits: [...this.splits],
   933→        position,
   934→        totalCars: this.mpActive ? 1 + this.remotes.length : 1 + this.opponents.length,
   935→      });
   936→      return;
   937→    }
   938→    // lap complete, more to go
   939→    store.pushPopup("lap", `LAP ${this.lapsDone}/${this.totalLaps} — ${fmt(lapMs)}`);
   940→    this.audio.checkpoint(-1);
   941→    this.lapStart = this.raceTime;
   942→    this.lineAbs += this.track.length;
   943→    this.nextGateK = 0;
   944→    this.nextGateAbs = this.gateAbs(0);
   945→    this.ghostRec = [];
   946→    this.ghostRecT = 0;
   947→    this.ghostCar.group.visible = this.ghost.length > 0;
   948→  }

   964→      this.physics.place(t.N - 10, 0, 13, false);

   998→  /** Soft sphere collisions between all cars (player + rivals). */
   999→  private resolveCarCollisions() {
  1000→    const cars: CarPhysics[] = [this.physics];
  1001→    for (const op of this.opponents) cars.push(op.physics);
  1002→    const R = 2.55;
  1003→    const R2 = R * R;
  1004→    for (let i = 0; i < cars.length; i++) {
  1005→      for (let j = i + 1; j < cars.length; j++) {
  1006→        const a = cars[i];
  1007→        const b = cars[j];
  1008→        if (Math.abs(a.pos.y - b.pos.y) > 2.2) continue;
  1009→        const dx = b.pos.x - a.pos.x;
  1010→        const dz = b.pos.z - a.pos.z;
  1011→        const d2 = dx * dx + dz * dz;
  1012→        if (d2 >= R2 || d2 < 1e-6) continue;
  1013→        const d = Math.sqrt(d2);
  1014→        const nx = dx / d;
  1015→        const nz = dz / d;
  1016→        const pen = (R - d) * 0.5;
  1017→        a.pos.x -= nx * pen;
  1018→        a.pos.z -= nz * pen;
  1019→        b.pos.x += nx * pen;
  1020→        b.pos.z += nz * pen;
  1021→        const rvx = b.vel.x - a.vel.x;
  1022→        const rvz = b.vel.z - a.vel.z;
  1023→        const vn = rvx * nx + rvz * nz;
  1024→        if (vn < 0) {
  1025→          // equal-mass bump with a bit of bounce
  1026→          const imp = -vn * 0.7;
  1027→          a.vel.x -= nx * imp;
  1028→          a.vel.z -= nz * imp;
  1029→          b.vel.x += nx * imp;
  1030→          b.vel.z += nz * imp;
  1031→          if ((a.isPlayer || b.isPlayer) && -vn > 4.5 && this.time - this.lastBumpT > 0.6) {
  1032→            this.lastBumpT = this.time;
  1033→            const impact = -vn;
  1034→            this.audio.crash(clamp(impact / 26, 0.12, 0.45));
  1035→            this.trauma = Math.min(1, this.trauma + 0.1 + impact * 0.012);
  1036→            this.input.rumble(clamp(impact / 20, 0.15, 0.6), 110);
  1037→          }
  1038→        }
  1039→      }
  1040→    }
  1041→    if (this.mpActive && !this.mpSpectating) this.resolveRemoteCollisions();
  1042→  }
  1043→
  1044→  /** Player vs network cars: the local car reacts (owner stays authoritative
  1045→   *  on their own client — both sides feel the bump locally). */
  1046→  private resolveRemoteCollisions() {
  1047→    const a = this.physics;
  1048→    const R = 2.55;
  1049→    const R2 = R * R;
  1050→    for (const rp of this.remotes) {
  1051→      if (!rp.active) continue;
  1052→      if (Math.abs(a.pos.y - rp.ry) > 2.2) continue;

  1054→      const dz = rp.rz - a.pos.z;

  1059→      const nz = dz / d;

  1061→      a.pos.x -= nx * pen;
  1062→      a.pos.z -= nz * pen;
  1063→      const rvx = rp.vel.x - a.vel.x;
  1064→      const rvz = rp.vel.z - a.vel.z;

  1067→        const imp = -vn * 0.7;
  1068→        a.vel.x -= nx * imp;
  1069→        a.vel.z -= nz * imp;

  1073→          this.audio.crash(clamp(impact / 26, 0.12, 0.45));

  1356→  private playGhost() {
  1357→    if (this.mpActive) {
  1358→      this.ghostCar.group.visible = false;
  1359→      return;
  1360→    }
  1361→    if (!this.ghost.length) {
  1362→      this.ghostCar.group.visible = false;
  1363→      return;
  1364→    }
  1365→    this.ghostCar.group.visible = true;
  1366→    const tMs = this.raceTime;
  1367→    const frames = this.ghost;
  1368→    // binary search
  1369→    let lo = 0;
  1370→    let hi = frames.length - 1;
  1371→    while (lo < hi) {
  1372→      const mid = (lo + hi) >> 1;
  1373→      if (frames[mid].t < tMs) lo = mid + 1;
  1374→      else hi = mid;
  1375→    }
  1376→    const i = Math.max(1, lo);
  1377→    const a = frames[i - 1];
  1378→    const b = frames[Math.min(frames.length - 1, i)];
  1379→    const span = Math.max(1e-3, b.t - a.t);
  1380→    const f = clamp((tMs - a.t) / span, 0, 1);
  1381→    const g = this.ghostCar.group;
  1382→    g.position.set(a.x + (b.x - a.x) * f, a.y + (b.y - a.y) * f, a.z + (b.z - a.z) * f);
  1383→    let dy = b.yaw - a.yaw;
  1384→    while (dy > Math.PI) dy -= Math.PI * 2;
  1385→    while (dy < -Math.PI) dy += Math.PI * 2;
  1386→    g.rotation.set(0, a.yaw + dy * f, 0);
  1387→  }
