     7→     7→function sstep(a: number, b: number, x: number): number {
     8→     8→  const t = clamp((x - a) / (b - a), 0, 1);

   127→   127→  private buildSpline() {
   128→   128→    const pts = CONTROL.map((p) => new THREE.Vector3(p[0], p[1], p[2]));
   129→   129→    const curve = new THREE.CatmullRomCurve3(pts, true, "catmullrom", 0.5);
   130→   130→    const approxLen = curve.getLength();
   131→   131→    const n = Math.max(600, Math.round(approxLen / this.step));
   132→   132→    const spaced = curve.getSpacedPoints(n);
   133→   133→    this.N = n;
   134→   134→    this.length = n * this.step;
   135→   135→    const raw = spaced.slice(0, n);
   136→   136→
   137→   137→    const dirs: THREE.Vector2[] = [];
   138→   138→    for (let i = 0; i < n; i++) {
   139→   139→      const a = raw[wrapIdx(i - 1, n)];
   140→   140→      const b = raw[wrapIdx(i + 1, n)];
   141→   141→      const d = new THREE.Vector2(b.x - a.x, b.z - a.z);
   142→   142→      if (d.lengthSq() < 1e-8) d.set(0, -1);
   143→   143→      dirs.push(d.normalize());
   144→   144→    }
   145→   145→    // curvature → banking (outer edge raised)
   146→   146→    const banks = new Array<number>(n).fill(0);
   147→   147→    for (let i = 0; i < n; i++) {
   148→   148→      const d0 = dirs[wrapIdx(i - 2, n)];
   149→   149→      const d1 = dirs[wrapIdx(i + 2, n)];
   150→   150→      const crossY = d0.x * d1.y - d0.y * d1.x; // >0 = left turn
   151→   151→      const dot = clamp(d0.dot(d1), -1, 1);
   152→   152→      const dTheta = Math.atan2(crossY, dot);
   153→   153→      banks[i] = clamp((dTheta / (4 * this.step)) * 46, -1, 1) * 3.0;
   154→   154→    }
   155→   155→    const sm = new Array<number>(n).fill(0);
   156→   156→    for (let i = 0; i < n; i++) {
   157→   157→      let acc = 0;
   158→   158→      for (let k = -9; k <= 9; k++) acc += banks[wrapIdx(i + k, n)];
   159→   159→      sm[i] = acc / 19;
   160→   160→    }
   161→   161→
   162→   162→    for (let i = 0; i < n; i++) {
   163→   163→      const p = raw[i];
   164→   164→      const d = dirs[i];
   165→   165→      const r = new THREE.Vector2(-d.y, d.x); // right = (-dz, dx)
   166→   166→      this.samples.push({
   167→   167→        x: p.x,
   168→   168→        y: p.y,
   169→   169→        z: p.z,
   170→   170→        dx: d.x,
   171→   171→        dz: d.y,
   172→   172→        rx: r.x,
   173→   173→        rz: r.y,
   174→   174→        hw: 9,
   175→   175→        shoulderL: 8,
   176→   176→        shoulderR: 8,
   177→   177→        bank: sm[i],
   178→   178→        gap: false,
   179→   179→        rampP: -1,
   180→   180→        dualT: 0,
   181→   181→        wallRideR: false,
   182→   182→        wallRideL: false,
   183→   183→        riverbed: false,
   184→   184→        tunnel: false,
   185→   185→        wallHL: 24,
   186→   186→        wallHR: 24,
   187→   187→      });
   188→   188→    }
   189→   189→  }

   191→   191→  private idxNear(x: number, z: number): number {
   192→   192→    let best = 0;
   193→   193→    let bd = Infinity;
   194→   194→    for (let i = 0; i < this.N; i++) {
   195→   195→      const s = this.samples[i];
   196→   196→      const d = (s.x - x) * (s.x - x) + (s.z - z) * (s.z - z);
   197→   197→      if (d < bd) {
   198→   198→        bd = d;
   199→   199→        best = i;
   200→   200→      }
   201→   201→    }
   202→   202→    return best;
   203→   203→  }

   205→   205→  private range(a: number, b: number): [number, number] {
   206→   206→    return [
   207→   207→      this.idxNear(CONTROL[a][0], CONTROL[a][2]),
   208→   208→      this.idxNear(CONTROL[b][0], CONTROL[b][2]),

   212→   212→  private forEachInRange(r: [number, number], fn: (i: number, t: number) => void) {
   213→   213→    const [a, b] = r;
   214→   214→    const span = (b - a + this.N) % this.N;
   215→   215→    for (let k = 0; k <= span; k++) fn(wrapIdx(a + k, this.N), span > 0 ? k / span : 1);
   216→   216→  }

   222→   222→  private buildZones() {
   223→   223→    const S = this.samples;
   224→   224→
   225→   225→    // --- launch ramp before the gap (last 58% of approach → crest) ---
   226→   226→    const ramp = this.range(2, 3);
   227→   227→    this.forEachInRange(ramp, (i, t) => {
   228→   228→      S[i].rampP = clamp((t - 0.42) / 0.58, 0, 1);
   229→   229→    });
   230→   230→
   231→   231→    // --- the GAP over the dry riverbed canyon ---
   232→   232→    const crestI = this.idxNear(CONTROL[3][0], CONTROL[3][2]);
   233→   233→    const landI = this.idxNear(CONTROL[4][0], CONTROL[4][2]);
   234→   234→    const gapSpan = (landI - crestI + this.N) % this.N;
   235→   235→    const gapStart = crestI + 2;
   236→   236→    const gapEnd = crestI + Math.round(gapSpan * 0.62);
   237→   237→    for (let k = gapStart; k <= gapEnd; k++) {
   238→   238→      const s = S[wrapIdx(k, this.N)];
   239→   239→      s.gap = true;
   240→   240→      s.shoulderL = 3.2;
   241→   241→      s.shoulderR = 3.2;
   242→   242→      s.wallHL = 30;
   243→   243→      s.wallHR = 30;
   244→   244→    }
   245→   245→
   246→   246→    // --- natural arch zone: taller walls ---
   247→   247→    const archI = this.idxNear(CONTROL[6][0], CONTROL[6][2]);
   248→   248→    for (let k = archI - 14; k <= archI + 14; k++) {
   249→   249→      const s = S[wrapIdx(k, this.N)];
   250→   250→      s.wallHL = 42;
   251→   251→      s.wallHR = 42;
   252→   252→    }
   253→   253→
   254→   254→    // --- wall-ride carousel: banked outer wall on the right ---
   255→   255→    const carousel = this.range(8, 11);
   256→   256→    this.forEachInRange(carousel, (i, t) => {
   257→   257→      const edge = sstep(0, 0.14, t) * (1 - sstep(0.86, 1, t));
   258→   258→      const s = S[i];
   259→   259→      s.wallRideR = edge > 0.01;
   260→   260→      s.shoulderR = 26 * edge + 8 * (1 - edge);
   261→   261→      s.wallHR = 30;
   262→   262→      s.bank = s.bank * (1 - edge) + (s.bank * 0.4 + 2.4) * edge;
   263→   263→    });
   264→   264→
   265→   265→    // --- tunnel through the mesa ---
   266→   266→    const tunnel = this.range(13, 16);
   267→   267→    this.forEachInRange(tunnel, (i, t) => {
   268→   268→      const edge = sstep(0, 0.05, t) * (1 - sstep(0.95, 1, t));
   269→   269→      const s = S[i];
   270→   270→      s.tunnel = edge > 0.01;
   271→   271→      s.hw = 8.5 + 0.5 * (1 - edge);
   272→   272→      s.shoulderL = 1.5 + 6.5 * (1 - edge);
   273→   273→      s.shoulderR = 1.5 + 6.5 * (1 - edge);
   274→   274→    });
   275→   275→
   276→   276→    // --- dual-line downhill: high terrace on the left ---
   277→   277→    const dual = this.range(16, 18);
   278→   278→    this.forEachInRange(dual, (i, t) => {
   279→   279→      const s = S[i];
   280→   280→      s.dualT = sstep(0, 0.12, t) * (1 - sstep(0.88, 1, t));
   281→   281→      s.shoulderL = 9.5;
   282→   282→    });
   283→   283→
   284→   284→    // --- riverbed: wide sandy floor ---
   285→   285→    const river = this.range(19, 24);
   286→   286→    this.forEachInRange(river, (i, t) => {
   287→   287→      const edge = sstep(0, 0.06, t) * (1 - sstep(0.94, 1, t));
   288→   288→      const s = S[i];
   289→   289→      s.riverbed = true;
   290→   290→      s.hw = 9 + 11 * edge;
   291→   291→      s.shoulderL = 8 + 6 * edge;
   292→   292→      s.shoulderR = 8 + 6 * edge;
   293→   293→      s.wallHL = 20;
   294→   294→      s.wallHR = 20;
   295→   295→    });
   296→   296→
   297→   297→    // --- low walls near the start/finish for the vista ---
   298→   298→    for (let k = -30; k <= 40; k++) {
   299→   299→      const s = S[wrapIdx(k, this.N)];
   300→   300→      s.wallHL = Math.min(s.wallHL, 14);
   301→   301→      s.wallHR = Math.min(s.wallHR, 14);
   302→   302→    }
   303→   303→
   304→   304→    // wall height noise
   305→   305→    for (let i = 0; i < this.N; i++) {
   306→   306→      const s = S[i];
   307→   307→      const n =
   308→   308→        Math.sin(i * 0.093) * 0.5 + Math.sin(i * 0.031 + 2.1) * 0.3 + Math.sin(i * 0.017 + 4) * 0.2;
   309→   309→      s.wallHL = Math.max(10, s.wallHL + n * 7);
   310→   310→      s.wallHR = Math.max(10, s.wallHR + n * 7 + Math.sin(i * 0.05) * 2);
   311→   311→    }
   312→   312→
   313→   313→    // --- kicker bumps in the riverbed ---
   314→   314→    const rB = this.idxNear(CONTROL[21][0], CONTROL[21][2]);
   315→   315→    const rC = this.idxNear(CONTROL[22][0], CONTROL[22][2]);
   316→   316→    const rD = this.idxNear(CONTROL[23][0], CONTROL[23][2]);
   317→   317→    this.kickers = [
   318→   318→      { s: (rB + 9) * this.step, lat: -8, r: 7, h: 2.1 },
   319→   319→      { s: (rC + 10) * this.step, lat: 7, r: 7, h: 2.2 },
   320→   320→      { s: (rD + 12) * this.step, lat: -5, r: 6.5, h: 1.9 },
   321→   321→    ];
   322→   322→
   323→   323→    // --- checkpoint gates in racing order ---
   324→   324→    const gateAnchors: { c: number; finish: boolean }[] = [
   325→   325→      { c: 4, finish: false }, // gap landing
   326→   326→      { c: 6, finish: false }, // arch
   327→   327→      { c: 7, finish: false }, // climb
   328→   328→      { c: 10, finish: false }, // carousel apex
   329→   329→      { c: 13, finish: false }, // tunnel entry
   330→   330→      { c: 16, finish: false }, // tunnel exit
   331→   331→      { c: 18, finish: false }, // dual-line merge
   332→   332→      { c: 22, finish: false }, // riverbed mid
   333→   333→      { c: 27, finish: false }, // final arc
   334→   334→      { c: 0, finish: true }, // start/finish
   335→   335→    ];
   336→   336→    for (const g of gateAnchors) {
   337→   337→      const idx = this.idxNear(CONTROL[g.c][0], CONTROL[g.c][2]);
   338→   338→      const s = S[idx];
   339→   339→      this.gates.push({ idx, pos: new THREE.Vector3(s.x, s.y, s.z), isFinish: g.finish });
   340→   340→      this.gatePts.push({ x: s.x, z: s.z, finish: g.finish });
   341→   341→    }
   342→   342→
   343→   343→    for (let i = 0; i < this.N; i += 6) this.minimapPts.push({ x: S[i].x, z: S[i].z });
   344→   344→
   345→   345→    this.addObstacles();
   346→   346→  }

   417→   417→  groundAt(i: number, lat: number, out: GroundInfo): GroundInfo {
   418→   418→    const s = this.sample(i);
   419→   419→    const al = Math.abs(lat);
   420→   420→    if (s.gap && al < s.hw + 3) {
   421→   421→      out.h = s.y - 13;
   422→   422→      out.onRoad = false;
   423→   423→      out.inPitZone = true;
   424→   424→      return out;
   425→   425→    }
   426→   426→    out.inPitZone = false;
   427→   427→    let h = s.y;
   428→   428→    const u = clamp(lat / s.hw, -1, 1);
   429→   429→    h += s.bank * u;
   430→   430→    if (s.riverbed) h += 0.45 * u * u;
   431→   431→    if (s.dualT > 0 && lat > 3) {
   432→   432→      const t = sstep(3, 5, lat) * (1 - sstep(13, 15.5, lat));
   433→   433→      h += 2.6 * s.dualT * t;
   434→   434→    }
   435→   435→    if (s.wallRideR && lat > 4) h += 13 * Math.pow(Math.min(1, (lat - 4) / 24), 2.4);
   436→   436→    if (s.wallRideL && lat < -4) h += 13 * Math.pow(Math.min(1, (-lat - 4) / 24), 2.4);
   437→   437→    if (s.rampP >= 0 && al < s.hw + 1) {
   438→   438→      const latFade = 1 - sstep(4.5, s.hw, al);
   439→   439→      h += 3.4 * sstep(0.1, 0.92, s.rampP) * latFade;
   440→   440→    }
   441→   441→    let onRoad = al <= s.hw;
   442→   442→    if (al > s.hw) {
   443→   443→      if (s.wallRideR && lat > 0) onRoad = true;
   444→   444→      else if (s.dualT > 0 && lat < 0 && lat > 3) onRoad = true;
   445→   445→      else {
   446→   446→        const sh = lat > 0 ? s.shoulderR : s.shoulderL;
   447→   447→        if (sh > 0) h -= 0.55 * Math.min(1, (al - s.hw) / Math.max(0.5, sh));
   448→   448→      }
   449→   449→    }
   450→   450→    const sHere = wrapIdx(i, this.N) * this.step;
   451→   451→    for (const k of this.kickers) {
   452→   452→      let ds = Math.abs(sHere - k.s);
   453→   453→      ds = Math.min(ds, this.length - ds);
   454→   454→      const dl = lat - k.lat;
   455→   455→      const d2 = ds * ds + dl * dl;
   456→   456→      if (d2 < k.r * k.r) {
   457→   457→        const q = Math.sqrt(d2) / k.r;
   458→   458→        h += k.h * (0.5 + 0.5 * Math.cos(Math.PI * q));
   459→   459→      }
   460→   460→    }
   461→   461→    out.h = h;
   462→   462→    out.onRoad = onRoad;
   463→   463→    return out;
   464→   464→  }
   465→   465→
   466→   466→  heightAt(i: number, lat: number): number {
   467→   467→    return this.groundAt(i, lat, TMP_GROUND).h;
   468→   468→  }
   469→   469→
   470→   470→  slopeLatAt(i: number, lat: number): number {
   471→   471→    const a = this.heightAt(wrapIdx(i, this.N), lat + 0.9);
   472→   472→    const b = this.heightAt(wrapIdx(i, this.N), lat - 0.9);
   473→   473→    return (a - b) / 1.8;
   474→   474→  }
   475→   475→
   476→   476→  slopeFwdAt(i: number, lat: number): number {
   477→   477→    const a = this.heightAt(wrapIdx(i + 1, this.N), lat);
   478→   478→    const b = this.heightAt(wrapIdx(i - 1, this.N), lat);
   479→   479→    return (a - b) / (2 * this.step);
   480→   480→  }
   481→   481→
   482→   482→  wallDist(i: number, side: 1 | -1): number {
   483→   483→    const s = this.sample(i);
   484→   484→    return side > 0 ? s.hw + s.shoulderR + 0.3 : s.hw + s.shoulderL + 0.3;
   485→   485→  }
   486→   486→
   487→   487→  nearestIdx(x: number, z: number, hint: number): number {
   488→   488→    let best = hint;
   489→   489→    let bd = Infinity;
   490→   490→    for (let k = -35; k <= 95; k++) {
   491→   491→      const i = wrapIdx(hint + k, this.N);
   492→   492→      const s = this.samples[i];
   493→   493→      const d = (s.x - x) * (s.x - x) + (s.z - z) * (s.z - z);
   494→   494→      if (d < bd) {
   495→   495→        bd = d;
   496→   496→        best = i;
   497→   497→      }
   498→   498→    }
   499→   499→    if (bd > 90 * 90) best = this.idxNear(x, z);
   500→   500→    return best;
   501→   501→  }
   502→   502→
   503→   503→  lateralOf(i: number, x: number, z: number): number {
   504→   504→    const s = this.sample(i);
   505→   505→    return (x - s.x) * s.rx + (z - s.z) * s.rz;
   506→   506→  }
   507→   507→
   508→   508→  posAt(i: number, lat: number, y?: number): THREE.Vector3 {
   509→   509→    const s = this.sample(i);
   510→   510→    return new THREE.Vector3(s.x + s.rx * lat, y ?? s.y, s.z + s.rz * lat);
   511→   511→  }

   527→   527→  private buildWalls() {
   528→   528→    const S = this.samples;
   529→   529→    const n = this.N;
   530→   530→    for (const side of [-1, 1] as const) {
   531→   531→      const pos: number[] = [];
   532→   532→      const col: number[] = [];
   533→   533→      const idx: number[] = [];
   534→   534→      const strata = PALETTE.rockStrata.map((c) => new THREE.Color(c));
   535→   535→      const cBase = new THREE.Color(PALETTE.rockBase);
   536→   536→      for (let i = 0; i < n; i++) {
   537→   537→        const s = S[i];
   538→   538→        const wd = this.wallDist(i, side) - 0.3;
   539→   539→        const bx = s.x + s.rx * wd * side;
   540→   540→        const bz = s.z + s.rz * wd * side;
   541→   541→        const wallH = side > 0 ? s.wallHR : s.wallHL;
   542→   542→        const lean = 7 + ((i * 37) % 5);
   543→   543→        const tx = s.x + (s.rx * wd + lean) * side;
   544→   544→        const tz = s.z + (s.rz * wd + lean) * side;
   545→   545→        const by = s.gap ? s.y - 13.4 : this.heightAt(i, wd * side) - 0.2;
   546→   546→        const ty = s.y + wallH;
   547→   547→        pos.push(bx, by, bz, tx, ty, tz);
   548→   548→        for (const yv of [by, ty]) {
   549→   549→          const band = Math.floor((yv + Math.sin(i * 0.7 + side) * 1.4) / 4.2);
   550→   550→          const c = strata[((band % strata.length) + strata.length) % strata.length].clone();
   551→   551→          c.lerp(cBase, 0.25 + 0.2 * Math.abs(Math.sin(i * 0.13 + band)));
   552→   552→          col.push(c.r, c.g, c.b);
   553→   553→        }
   554→   554→      }
   555→   555→      for (let i = 0; i < n; i++) {
   556→   556→        const a = i * 2;
   557→   557→        const b = ((i + 1) % n) * 2;
   558→   558→        if (side > 0) {
   559→   559→          // normal faces the track (-right)
   560→   560→          idx.push(a, a + 1, b, a + 1, b + 1, b);
   561→   561→        } else {
   562→   562→          idx.push(a, b, a + 1, b, b + 1, a + 1);
   563→   563→        }
   564→   564→      }
   565→   565→      const g = new THREE.BufferGeometry();
   566→   566→      g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   567→   567→      g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   568→   568→      g.setIndex(idx);
   569→   569→      g.computeVertexNormals();
   570→   570→      const m = new THREE.MeshLambertMaterial({ vertexColors: true, flatShading: true });
   571→   571→      const mesh = new THREE.Mesh(g, m);
   572→   572→      mesh.castShadow = true;
   573→   573→      mesh.receiveShadow = true;
   574→   574→      this.group.add(mesh);
   575→   575→      this.disposables.push(g, m);
   576→   576→    }
   577→   577→  }

   579→   579→  private ringLats(i: number): number[] {
   580→   580→    const s = this.sample(i);
   581→   581→    const wdL = -this.wallDist(i, -1);
   582→   582→    const wdR = this.wallDist(i, 1);
   583→   583→    const lats: number[] = [wdL];
   584→   584→    for (const l of ROAD_COLS) {
   585→   585→      if (l > wdL + 0.05 && l < wdR - 0.05) lats.push(l);
   586→   586→    }
   587→   587→    lats.push(wdR);
   588→   588→    return lats;
   589→   589→  }

   591→   591→  private buildRoad() {
   592→   592→    const n = this.N;
   593→   593→    const pos: number[] = [];
   594→   594→    const col: number[] = [];
   595→   595→    const uv: number[] = [];
   596→   596→    const idx: number[] = [];
   597→   597→    const cRoad = new THREE.Color(PALETTE.road);
   598→   598→    const cRoadDust = new THREE.Color(PALETTE.roadDust);
   599→   599→    const cSand = new THREE.Color(PALETTE.sand);
   600→   600→    const cPit = new THREE.Color(PALETTE.pitSand);
   601→   601→    const cTerr = new THREE.Color(0xcf8a52);
   602→   602→    const gi: GroundInfo = { h: 0, onRoad: false, inPitZone: false };
   603→   603→
   604→   604→    const ringLats: number[][] = [];
   605→   605→    const ringStart: number[] = [];
   606→   606→    for (let i = 0; i < n; i++) {
   607→   607→      ringStart[i] = pos.length / 3;
   608→   608→      const s = this.samples[i];
   609→   609→      const lats = this.ringLats(i);
   610→   610→      ringLats.push(lats);
   611→   611→      for (const l of lats) {
   612→   612→        this.groundAt(i, l, gi);
   613→   613→        let c: THREE.Color;
   614→   614→        if (gi.inPitZone) c = cPit;
   615→   615→        else if (Math.abs(l) <= s.hw + 0.01) c = cRoad.clone().lerp(cRoadDust, 0.3 + 0.2 * Math.sin(i * 0.37));
   616→   616→        else if (s.dualT > 0 && l < -3) c = cTerr.clone();
   617→   617→        else if (s.wallRideR && l > s.hw) {
   618→   618→          const t = clamp((l - 4) / 24, 0, 1);
   619→   619→          c = cRoad.clone().lerp(cTerr, t * 0.8);
   620→   620→        } else c = cSand.clone().offsetHSL(0, 0, Math.sin(i * 0.53 + l) * 0.015);
   621→   621→        pos.push(s.x + s.rx * l, gi.h, s.z + s.rz * l);
   622→   622→        col.push(c.r, c.g, c.b);
   623→   623→        uv.push(l / 12, (i * this.step) / 7);
   624→   624→      }
   625→   625→    }
   626→   626→    for (let i = 0; i < n; i++) {
   627→   627→      const j = (i + 1) % n;
   628→   628→      if (this.samples[i].gap || this.samples[j].gap) continue;
   629→   629→      const a = ringStart[i];
   630→   630→      const b = ringStart[j];
   631→   631→      const m = ringLats[i].length; // stitch with ring i's column count on both rings
   632→   632→      for (let k = 0; k < m - 1; k++) {
   633→   633→        idx.push(a + k, a + k + 1, b + k, a + k + 1, b + k + 1, b + k);
   634→   634→      }
   635→   635→    }
   636→   636→    const g = new THREE.BufferGeometry();
   637→   637→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   638→   638→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   639→   639→    g.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
   640→   640→    g.setIndex(idx);
   641→   641→    g.computeVertexNormals();
   642→   642→    const tex = makeSandNoiseTexture();
   643→   643→    const m = new THREE.MeshLambertMaterial({ vertexColors: true, map: tex, flatShading: true });
   644→   644→    const mesh = new THREE.Mesh(g, m);
   645→   645→    mesh.receiveShadow = true;
   646→   646→    this.group.add(mesh);
   647→   647→    this.disposables.push(g, m, tex);
   648→   648→  }

   650→   650→  private buildPit() {
   651→   651→    const n = this.N;
   652→   652→    const pos: number[] = [];
   653→   653→    const idx: number[] = [];
   654→   654→    const col: number[] = [];
   655→   655→    const cPit = new THREE.Color(PALETTE.pitSand);
   656→   656→    const cDark = new THREE.Color(0xb08a58);
   657→   657→    let v = 0;
   658→   658→    const pushTri = (a: number[], b: number[], c: number[], shade: number) => {
   659→   659→      for (const p of [a, b, c]) pos.push(p[0], p[1], p[2]);
   660→   660→      const cc = cPit.clone().lerp(cDark, shade);
   661→   661→      col.push(cc.r, cc.g, cc.b, cc.r, cc.g, cc.b, cc.r, cc.g, cc.b);
   662→   662→      idx.push(v, v + 1, v + 2);
   663→   663→      v += 3;
   664→   664→    };
   665→   665→    const pushQuad = (a: number[], b: number[], c: number[], d: number[]) => {
   666→   666→      pushTri(a, b, c, 0);
   667→   667→      pushTri(a, c, d, 0.25);
   668→   668→    };
   669→   669→    const PL = 12.4;
   670→   670→    let i = 0;
   671→   671→    while (i < n) {
   672→   672→      if (!this.samples[i].gap) {
   673→   673→        i++;
   674→   674→        continue;
   675→   675→      }
   676→   676→      let j = i;
   677→   677→      while (j < n && this.samples[j].gap) j++;
   678→   678→      const jEnd = Math.min(j, n - 1); // gap never crosses the wrap in this track
   679→   679→      // pit floor
   680→   680→      for (let k = i; k < jEnd; k++) {
   681→   681→        const s0 = this.samples[k];
   682→   682→        const s1 = this.samples[k + 1];
   683→   683→        for (const [l0, l1] of [
   684→   684→          [-PL, -PL / 2],
   685→   685→          [-PL / 2, PL / 2],
   686→   686→          [PL / 2, PL],
   687→   687→        ] as [number, number][]) {
   688→   688→          pushQuad(
   689→   689→            [s0.x + s0.rx * l0, s0.y - 13, s0.z + s0.rz * l0],
   690→   690→            [s0.x + s0.rx * l1, s0.y - 13, s0.z + s0.rz * l1],
   691→   691→            [s1.x + s1.rx * l1, s1.y - 13, s1.z + s1.rz * l1],
   692→   692→            [s1.x + s1.rx * l0, s1.y - 13, s1.z + s1.rz * l0]
   693→   693→          );
   694→   694→        }
   695→   695→      }
   696→   696→      // near face: road edge down into the pit
   697→   697→      const sp = this.samples[wrapIdx(i - 1, n)];
   698→   698→      const sa = this.samples[i];
   699→   699→      for (const [l0, l1] of [
   700→   700→        [-PL, 0],
   701→   701→        [0, PL],
   702→   702→      ] as [number, number][]) {
   703→   703→        pushQuad(
   704→   704→          [sp.x + sp.rx * l0, this.heightAt(wrapIdx(i - 1, n), l0), sp.z + sp.rz * l0],
   705→   705→          [sp.x + sp.rx * l1, this.heightAt(wrapIdx(i - 1, n), l1), sp.z + sp.rz * l1],
   706→   706→          [sa.x + sa.rx * l1, sa.y - 13, sa.z + sa.rz * l1],
   707→   707→          [sa.x + sa.rx * l0, sa.y - 13, sa.z + sa.rz * l0]
   708→   708→        );
   709→   709→      }
   710→   710→      // far face: pit up to the landing road
   711→   711→      const sf = this.samples[wrapIdx(jEnd, n)];
   712→   712→      const sl = this.samples[wrapIdx(jEnd - 1, n)];
   713→   713→      for (const [l0, l1] of [
   714→   714→        [-PL, 0],
   715→   715→        [0, PL],
   716→   716→      ] as [number, number][]) {
   717→   717→        pushQuad(
   718→   718→          [sl.x + sl.rx * l0, sl.y - 13, sl.z + sl.rz * l0],
   719→   719→          [sl.x + sl.rx * l1, sl.y - 13, sl.z + sl.rz * l1],
   720→   720→          [sf.x + sf.rx * l1, this.heightAt(wrapIdx(jEnd, n), l1), sf.z + sf.rz * l1],
   721→   721→          [sf.x + sf.rx * l0, this.heightAt(wrapIdx(jEnd, n), l0), sf.z + sf.rz * l0]
   722→   722→        );
   723→   723→      }
   724→   724→      i = j;
   725→   725→    }
   726→   726→    const g = new THREE.BufferGeometry();
   727→   727→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   728→   728→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   729→   729→    g.setIndex(idx);
   730→   730→    g.computeVertexNormals();
   731→   731→    const m = new THREE.MeshLambertMaterial({
   732→   732→      vertexColors: true,
   733→   733→      flatShading: true,
   734→   734→      side: THREE.DoubleSide,
   735→   735→    });
   736→   736→    const mesh = new THREE.Mesh(g, m);
   737→   737→    mesh.receiveShadow = true;
   738→   738→    this.group.add(mesh);
   739→   739→    this.disposables.push(g, m);
   740→   740→  }

   742→   742→  private buildTunnel() {
   743→   743→    const n = this.N;
   744→   744→    const tunnelIdx: number[] = [];
   745→   745→    for (let i = 0; i < n; i++) if (this.samples[i].tunnel) tunnelIdx.push(i);
   746→   746→    if (tunnelIdx.length < 8) return;
   747→   747→    const pos: number[] = [];
   748→   748→    const col: number[] = [];
   749→   749→    const idx: number[] = [];
   750→   750→    const strata = PALETTE.rockStrata.map((c) => new THREE.Color(c));
   751→   751→    const cDark = new THREE.Color(0x6e3524);
   752→   752→    const SEG = 7;
   753→   753→    const R = 12.2;
   754→   754→    for (let r = 0; r < tunnelIdx.length; r++) {
   755→   755→      const i = tunnelIdx[r];
   756→   756→      const s = this.samples[i];
   757→   757→      for (let k = 0; k <= SEG; k++) {
   758→   758→        const a = Math.PI * (k / SEG);
   759→   759→        const lx = -Math.cos(a) * R;
   760→   760→        const ly = Math.sin(a) * R * 0.92;
   761→   761→        pos.push(s.x + s.rx * lx, s.y + ly - 1.8, s.z + s.rz * lx);
   762→   762→        const band = Math.floor((k + i * 0.2) / 2);
   763→   763→        const c = strata[((band % strata.length) + strata.length) % strata.length].clone().lerp(cDark, 0.5);
   764→   764→        col.push(c.r, c.g, c.b);
   765→   765→      }
   766→   766→      if (r > 0) {
   767→   767→        for (let k = 0; k < SEG; k++) {
   768→   768→          const a = (r - 1) * (SEG + 1) + k;
   769→   769→          const b = r * (SEG + 1) + k;
   770→   770→          idx.push(a, a + 1, b, a + 1, b + 1, b);
   771→   771→        }
   772→   772→      }
   773→   773→    }
   774→   774→    const g = new THREE.BufferGeometry();
   775→   775→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   776→   776→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   777→   777→    g.setIndex(idx);
   778→   778→    g.computeVertexNormals();
   779→   779→    const m = new THREE.MeshLambertMaterial({
   780→   780→      vertexColors: true,
   781→   781→      flatShading: true,
   782→   782→      side: THREE.DoubleSide,
   783→   783→    });
   784→   784→    const mesh = new THREE.Mesh(g, m);
   785→   785→    mesh.castShadow = true;
   786→   786→    mesh.receiveShadow = true;
   787→   787→    this.group.add(mesh);
   788→   788→    this.disposables.push(g, m);
   789→   789→
   790→   790→    // amber light strips along the ceiling
   791→   791→    const stripGeo = new THREE.BoxGeometry(0.5, 0.15, 2.4);
   792→   792→    const stripMat = new THREE.MeshBasicMaterial({ color: 0xffc766 });
   793→   793→    for (let r = 5; r < tunnelIdx.length - 5; r += 10) {
   794→   794→      const i = tunnelIdx[r];
   795→   795→      const s = this.samples[i];
   796→   796→      const strip = new THREE.Mesh(stripGeo, stripMat);
   797→   797→      strip.position.set(s.x, s.y + R * 0.92 - 1.2, s.z);
   798→   798→      strip.lookAt(s.x + s.dx, s.y + R * 0.92 - 1.2, s.z + s.dz);
   799→   799→      this.group.add(strip);
   800→   800→    }
   801→   801→    this.disposables.push(stripGeo, stripMat);
   802→   802→
   803→   803→    // the mesa blob covering the tunnel
   804→   804→    const mid = tunnelIdx[Math.floor(tunnelIdx.length / 2)];
   805→   805→    const ms = this.samples[mid];
   806→   806→    const mesaGeo = new THREE.SphereGeometry(1, 12, 8);
   807→   807→    const mesaMat = new THREE.MeshLambertMaterial({ color: 0xb0532f, flatShading: true });
   808→   808→    const mesa = new THREE.Mesh(mesaGeo, mesaMat);
   809→   809→    mesa.scale.set(150, 40, 58);
   810→   810→    mesa.position.set(ms.x, ms.y + 4, ms.z);
   811→   811→    mesa.rotation.y = Math.atan2(ms.dx, ms.dz) + Math.PI / 2;
   812→   812→    mesa.castShadow = true;
   813→   813→    this.group.add(mesa);
   814→   814→    this.disposables.push(mesaGeo, mesaMat);
   815→   815→  }

   817→   817→  private buildArch() {
   818→   818→    const i = this.idxNear(CONTROL[6][0], CONTROL[6][2]);
   819→   819→    const s = this.samples[i];
   820→   820→    const geo = new THREE.TorusGeometry(30, 7.5, 8, 14, Math.PI);

  1025→  1025→function mulberry32(seed: number) {
  1026→  1026→  let a = seed >>> 0;
  1027→  1027→  return () => {
  1028→  1028→    a |= 0;
  1029→  1029→    a = (a + 0x6d2b79f5) | 0;
  1030→  1030→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
  1031→  1031→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  1032→  1032→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  1033→  1033→  };
  1034→  1034→}
