   226→  private buildSky() {
   227→    const geo = new THREE.SphereGeometry(3900, 32, 16);
   228→    const mat = new THREE.ShaderMaterial({
   229→      side: THREE.BackSide,
   230→      depthWrite: false,
   231→      fog: false,
   232→      uniforms: {
   233→        uZen: { value: new THREE.Color(PALETTE.skyZenith) },
   234→        uMid: { value: new THREE.Color(PALETTE.skyMid) },
   235→        uHor: { value: new THREE.Color(PALETTE.skyHorizon) },
   236→        uSun: { value: SUN_DIR.clone() },
   237→      },
   238→      vertexShader: `
   239→        varying vec3 vW;
   240→        void main() {
   241→          vW = (modelMatrix * vec4(position, 1.0)).xyz;
   242→          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   243→        }
   244→      `,
   245→      fragmentShader: `
   246→        uniform vec3 uZen; uniform vec3 uMid; uniform vec3 uHor; uniform vec3 uSun;
   247→        varying vec3 vW;
   248→        void main() {
   249→          vec3 d = normalize(vW);
   250→          float h = d.y;
   251→          vec3 col = h > 0.16 ? mix(uMid, uZen, smoothstep(0.16, 0.6, h))
   252→                              : mix(uHor, uMid, smoothstep(-0.06, 0.16, h));
   253→          float sd = max(dot(d, uSun), 0.0);
   254→          col += vec3(1.0, 0.88, 0.62) * (pow(sd, 900.0) * 1.2 + pow(sd, 16.0) * 0.3 + pow(sd, 4.0) * 0.1);
   255→          gl_FragColor = vec4(col, 1.0);
   256→        }
   257→      `,
   258→    });
   259→    const sky = new THREE.Mesh(geo, mat);
   260→    sky.frustumCulled = false;
   261→    this.scene.add(sky);
   262→  }
   263→
   264→  private buildHaze() {
   265→    this.hazeRT = new THREE.WebGLRenderTarget(2, 2, {
   266→      type: THREE.HalfFloatType,
   267→    });
   268→    this.hazeMat = new THREE.ShaderMaterial({
   269→      uniforms: {
   270→        tDiffuse: { value: this.hazeRT.texture },
   271→        uTime: { value: 0 },
   272→        uAmount: { value: 1 },
   273→      },
   274→      vertexShader: `
   275→        varying vec2 vUv;
   276→        void main() { vUv = uv; gl_Position = vec4(position.xy, 0.0, 1.0); }
   277→      `,
   278→      fragmentShader: `
   279→        uniform sampler2D tDiffuse; uniform float uTime; uniform float uAmount;
   280→        varying vec2 vUv;
   281→        void main() {
   282→          vec2 uv = vUv;
   283→          float mask = smoothstep(0.85, 0.45, uv.y) * 0.75 + 0.1;
   284→          uv.x += sin(uv.y * 110.0 + uTime * 3.3) * 0.0026 * mask * uAmount;
   285→          uv.y += sin(uv.x * 70.0 + uTime * 2.1) * 0.0018 * mask * uAmount;
   286→          vec3 c = texture2D(tDiffuse, uv).rgb;
   287→          // subtle warm vignette
   288→          vec2 q = vUv - 0.5;
   289→          c *= 1.0 - dot(q, q) * 0.35;
   290→          gl_FragColor = vec4(c, 1.0);
   291→        }
   292→      `,
   293→    });
   294→    this.hazeScene = new THREE.Scene();
   295→    const quadGeo = new THREE.PlaneGeometry(2, 2);
   296→    this.hazeScene.add(new THREE.Mesh(quadGeo, this.hazeMat));
   297→  }

   299→  private loadBest() {
   300→    try {
   301→      const raw = localStorage.getItem("canyonrush.best.v1");
   302→      if (raw) {
   303→        const d = JSON.parse(raw) as {
   304→          totalMs: number;
   305→          bestLapMs: number;
   306→          splits: number[];
   307→          laps: number;
   308→        };
   309→        this.bestTotalMs = d.totalMs;
   310→        this.bestLapMs = d.bestLapMs;
   311→        this.bestSplits = d.splits ?? [];
   312→      }
   313→      const graw = localStorage.getItem(GHOST_KEY);
   314→      if (graw) this.ghost = JSON.parse(graw) as GhostFrame[];
   315→    } catch {
   316→      /* ignore */
   317→    }
   318→  }

   329→  applySettings(s: GameSettings) {
   330→    const prevShadows = this.settings.shadows;
   331→    this.settings = { ...s };
   332→    this.physics.assist = s.assist;
   333→    this.particles.setHighDensity(s.highParticles);
   334→    this.audio.setVolumes(s.musicVol, s.sfxVol);
   335→    this.input.setMouseSteer(s.mouseSteer);
   336→    if (prevShadows !== s.shadows) {
   337→      this.renderer.shadowMap.enabled = s.shadows;
   338→      this.sun.castShadow = s.shadows;
   339→      this.scene.traverse((o) => {
   340→        const mesh = o as THREE.Mesh;
   341→        if (mesh.isMesh) {
   342→          const mats = Array.isArray(mesh.material) ? mesh.material : [mesh.material];
   343→          for (const m of mats) (m as THREE.Material).needsUpdate = true;
   344→        }
   345→      });
   346→    }

   431→    this.renderer.setSize(w, h, false);

   444→  private loop = () => {
   445→    if (this.disposed) return;
   446→    this.raf = requestAnimationFrame(this.loop);
   447→    if (this.debugSelftestActive) return; // selftest drives physics deterministically
   448→    const now = performance.now();
   449→    let dt = (now - this.lastT) / 1000;
   450→    this.lastT = now;
   451→    dt = clamp(dt, 0.0005, 1 / 30);
   452→    if (this.paused) dt = 0;
   453→
   454→    this.time += dt;
   455→
   456→    const input = this.input.sample(Math.max(dt, 0.0001));
   457→    this.curInput = input;
   458→    // longitudinal accel estimate for body pitch
   459→    const sp = this.physics.speed;
   460→    if (dt > 0.0001) {
   461→      this.lastAccel = damp(this.lastAccel, (sp - this.prevSpeed) / dt, 6, dt);
   462→    }
   463→    this.prevSpeed = sp;
   464→    this.noteInput(input.steer, input.brake, this.lastAccel);
   465→
   466→    if (this.phase === "countdown") this.updateCountdown(dt);
   467→    else if (this.phase === "racing") this.updateRacing(dt, input);
   468→    else if (this.phase === "finished") this.updateFinished(dt);
   469→    else this.updateTitle(dt);
   470→
   471→    if (this.recoverT > 0) this.recoverT -= dt;
   472→
   473→    this.updateCamera(dt);
   474→    this.updateSun();
   475→    this.particles.update(dt, this.time);
   476→    this.updateGateGlow();
   477→    this.emitTelemetry();
   478→
   479→    // audio
   480→    const { rpm } = this.physics.gearAndRpm();
   481→    const intensity =
   482→      this.phase === "title"
   483→        ? 0
   484→        : this.phase === "countdown"
   485→          ? 1
   486→          : this.phase === "finished"
   487→            ? 1
   488→            : clamp(1 + (this.physics.speed > 26 ? 1 : 0) + (this.physics.boosting || this.physics.boostMeter > 82 ? 1 : 0), 0, 3);
   489→    this.audio.update(dt, {
   490→      rpm:
   491→        this.phase === "countdown"
   492→          ? 0.25 + input.throttle * 0.55
   493→          : this.phase === "racing" || this.phase === "finished"
   494→            ? rpm
   495→            : 0.22,
   496→      throttle: this.phase === "racing" ? input.throttle : this.phase === "countdown" ? input.throttle * 0.7 : 0,
   497→      speed: this.physics.speed,
   498→      driftIntensity: this.physics.drifting ? clamp(Math.abs(this.physics.slip) / 0.5, 0, 1) : 0,
   499→      boosting: this.physics.boosting && this.phase === "racing",
   500→      wallRide: this.physics.wallRiding,
   501→      musicIntensity: intensity,
   502→    });
   503→
   504→    // render (with optional heat haze pass)
   505→    if (this.settings.heatHaze && this.hazeRT && this.hazeScene && this.hazeMat) {
   506→      this.hazeMat.uniforms.uTime.value = this.time;
   507→      this.renderer.setRenderTarget(this.hazeRT);
   508→      this.renderer.clear();
   509→      this.renderer.render(this.scene, this.camera);
   510→      this.renderer.setRenderTarget(null);
   511→      this.renderer.render(this.hazeScene, this.hazeCam);
   512→    } else {
   513→      this.renderer.render(this.scene, this.camera);
   514→    }
   515→  };
   516→
   517→  private updateCountdown(dt: number) {
   518→    this.countdownT -= dt;
   519→    const v = Math.ceil(this.countdownT);
   520→    if (v !== this.countdownShown && v >= 0) {
   521→      this.countdownShown = v;
   522→      this.cbs.onCountdown(v);
   523→      this.audio.countdownBeep(v === 0);
   524→    }
   525→    if (this.countdownT <= 0) {
   526→      this.setPhase("racing");
   527→      this.raceTime = 0;
   528→      this.lapStart = 0;
   529→    }
   530→  }
   531→
   532→  private updateRacing(dt: number, input: DriveInput) {
   533→    if (this.recoverT > 0) return;
   534→    const prevS = this.physics.progressS;
   535→    this.physics.update(input, dt);
   536→    this.raceTime += dt * 1000;
   537→    this.syncCarVisual(dt);
   538→    this.emitEffects(dt);
   539→    this.recordGhost(dt);
   540→    this.playGhost();
   541→    this.checkGates(prevS);
   542→    // wrong-way popup (throttled)
   543→    if (this.physics.wrongWay && this.time - this.lastWrongWayPopup > 2.5) {
   544→      this.lastWrongWayPopup = this.time;
   545→      gameStore.getState().pushPopup("wrongway", "WRONG WAY!");
   546→    }
   547→  }
   548→
   549→  private lastWrongWayPopup = -10;
   550→
   551→  private updateFinished(dt: number) {
   552→    this.physics.update({ steer: 0, throttle: 0, brake: 0.25, handbrake: false, boost: false }, dt);
   553→    this.syncCarVisual(dt);
   554→    this.finishOrbitT += dt;
   555→  }

   582→    const t = this.track;

   588→  private checkGates(prevS: number) {
   589→    const t = this.track;
   590→    while (this.physics.progressS >= this.nextGateAbs) {
   591→      const gate = t.gates[this.nextGateK];
   592→      if (!gate) break;
   593→      const latOk = Math.abs(this.physics.lateral) < t.sample(gate.idx).hw + 10;
   594→      const prevOk = prevS <= this.nextGateAbs + 1;
   595→      if (!latOk || !prevOk) break;
   596→      if (gate.isFinish) {
   597→        this.completeLap();
   598→      } else {
   599→        const lapMs = this.raceTime - this.lapStart;
   600→        this.splits.push(lapMs);
   601→        const best = this.bestSplits[this.splits.length - 1];
   602→        const delta = best != null ? lapMs - best : null;
   603→        gameStore.getState().pushPopup("checkpoint", `CHECKPOINT ${this.splits.length}`, { deltaMs: delta ?? undefined });
   604→        this.physics.boostMeter = clamp(this.physics.boostMeter + 12, 0, 100);
   605→        this.audio.checkpoint(delta);
   606→        this.nextGateK++;
   607→        this.nextGateAbs = this.gateAbs(this.nextGateK);
   608→      }
   609→    }
   610→  }

   612→  private completeLap() {
   613→    const lapMs = this.raceTime - this.lapStart;
   614→    this.lapTimes.push(lapMs);
   615→    const lapIsBest = this.bestLapMs == null || lapMs < this.bestLapMs;
   616→    if (lapIsBest) {
   617→      this.bestLapMs = lapMs;
   618→      if (this.ghostRec.length > 10) this.ghost = this.ghostRec;
   619→    }
   620→    this.lapsDone++;
   621→    const store = gameStore.getState();
   622→    if (this.lapsDone >= this.totalLaps) {
   623→      // ---- finish ----
   624→      const totalMs = this.raceTime;
   625→      const bestLap = Math.min(...this.lapTimes);
   626→      const isNewRecord = this.bestTotalMs == null || totalMs < this.bestTotalMs;
   627→      if (isNewRecord) {
   628→        this.bestTotalMs = totalMs;
   629→        this.bestSplits = [...this.splits];
   630→      }
   631→      try {
   632→        localStorage.setItem(
   633→          "canyonrush.best.v1",
   634→          JSON.stringify({
   635→            totalMs: this.bestTotalMs,
   636→            bestLapMs: this.bestLapMs,
   637→            splits: this.bestSplits,
   638→            laps: this.totalLaps,
   639→          })
   640→        );
   641→        if (this.ghost.length > 10) {
   642→          localStorage.setItem(GHOST_KEY, JSON.stringify(this.ghost));
   643→        }
   644→      } catch {
   645→        /* ignore */
   646→      }
   647→      if (isNewRecord) {
   648→        store.saveBest({ totalMs, bestLapMs: this.bestLapMs, splits: this.bestSplits, laps: this.totalLaps });
   649→        store.pushPopup("newrecord", "NEW RECORD!");
   650→      }
   651→      this.audio.finish(isNewRecord);
   652→      this.setPhase("finished");
   653→      this.finishOrbitT = 0;
   654→      this.ghostCar.group.visible = false;
   655→      this.cbs.onFinish({
   656→        totalMs,
   657→        bestLapMs: bestLap,
   658→        topSpeedKmh: Math.round(this.physics.topSpeed),
   659→        driftScore: Math.round(this.physics.driftScore),
   660→        nearMisses: this.physics.nearMisses,
   661→        bigAirCount: this.bigAirCount,
   662→        laps: this.totalLaps,
   663→        isNewRecord,
   664→        splits: [...this.splits],
   665→      });
   666→      return;
   667→    }
   668→    // lap complete, more to go
   669→    store.pushPopup("lap", `LAP ${this.lapsDone}/${this.totalLaps} — ${fmt(lapMs)}`);
   670→    this.audio.checkpoint(-1);
   671→    this.lapStart = this.raceTime;
   672→    this.lineAbs += this.track.length;
   673→    this.nextGateK = 0;
   674→    this.nextGateAbs = this.gateAbs(0);
   675→    this.ghostRec = [];
   676→    this.ghostRecT = 0;
   677→    this.ghostCar.group.visible = this.ghost.length > 0;
   678→  }

   680→  private startRecovery() {
   681→    if (this.recoverT > 0 || this.phase !== "racing") return;
   682→    this.recoverT = 0.85;
   683→    this.audio.recovery();
   684→    gameStore.getState().pushPopup("recovery", "RECOVERED — HOLD YOUR LINE!");
   685→    const t = this.track;
   686→    if (this.nextGateK > 0) {

   688→      const idx = wrapI(gate.idx + 4, t.N);
   689→      const baseS = this.lineAbs + gate.idx * t.step;
   690→      this.physics.place(idx, 0, 13, false);
   691→      this.physics.syncProgress(baseS + 4 * t.step);
   692→    } else {
   693→      // no gate passed yet this lap → back to the lap start line
   694→      this.physics.place(t.N - 10, 0, 13, false);
   695→      this.physics.syncProgress(this.lineAbs - 20);
   696→    }
   697→    this.syncCarVisual(0.016);
   698→    this.trauma = Math.min(1, this.trauma + 0.3);
   699→  }

   722→      g.rotation.order = "YXZ";

   724→      g.rotation.z = damp(g.rotation.z, 0, 4, dt);

   728→    void rpm;

   748→  private emitEffects(dt: number) {
   749→    const p = this.physics;
   750→    const fwd = this.tmpV.set(Math.sin(p.yaw), 0, Math.cos(p.yaw));
   751→    const right = this.tmpV2.set(-Math.cos(p.yaw), 0, Math.sin(p.yaw));
   752→
   753→    const speedH = p.speed;
   754→
   755→    if (p.grounded) {
   756→      // drift smoke + skids
   757→      if (p.drifting && speedH > 8) {
   758→        const q = clamp(Math.abs(p.slip) / 0.5, 0.2, 1);
   759→        for (const side of [-1, 1]) {
   760→          const wx = p.pos.x - fwd.x * 1.25 + right.x * side * 0.85;
   761→          const wz = p.pos.z - fwd.z * 1.25 + right.z * side * 0.85;
   762→          const wy = p.pos.y;
   763→          this.particles.emitSmoke(wx, wy, wz, q);
   764→        }
   765→        this.skidTimer -= dt;
   766→        if (this.skidTimer <= 0) {
   767→          this.skidTimer = 0.03;
   768→          const vx = p.vel.x / Math.max(1, speedH);
   769→          const vz = p.vel.z / Math.max(1, speedH);
   770→          for (const side of [-1, 1]) {
   771→            const wx = p.pos.x - fwd.x * 1.25 + right.x * side * 0.85;
   772→            const wz = p.pos.z - fwd.z * 1.25 + right.z * side * 0.85;
   773→            this.particles.emitSkid(wx, p.pos.y, wz, vx, vz, clamp(Math.abs(p.slip) * 2, 0.3, 1), this.time);
   774→          }
   775→        }
   776→      }
   777→      // off-road dust
   778→      if (p.offRoad && speedH > 9) {
   779→        this.particles.emitDust(
   780→          p.pos.x - fwd.x * 1.4, p.pos.y, p.pos.z - fwd.z * 1.4,
   781→          clamp(speedH / 30, 0.3, 1.2), p.vel.x, p.vel.z
   782→        );
   783→      }
   784→      // wall-ride sparks
   785→      if (p.wallRiding) {
   786→        this.sparkTimer -= dt;
   787→        if (this.sparkTimer <= 0) {
   788→          this.sparkTimer = 0.06;
   789→          const side = p.lateral > 0 ? 1 : -1;
   790→          this.particles.emitSparks(
   791→            p.pos.x + right.x * side * 1.1, p.pos.y + 0.2, p.pos.z + right.z * side * 1.1,
   792→            -right.x * side, -right.z * side, 3
   793→          );
   794→        }
   795→      }
   796→    }
   797→    // boost trail
   798→    if (p.boosting) {
   799→      this.boostTrailTimer -= dt;
   800→      if (this.boostTrailTimer <= 0) {
   801→        this.boostTrailTimer = 0.02;
   802→        this.particles.emitBoost(p.pos.x - fwd.x * 2.0, p.pos.y + 0.45, p.pos.z - fwd.z * 2.0, -fwd.x, -fwd.z);
   803→      }
   804→    }
   805→  }

   846→  private playGhost() {
   847→    if (!this.ghost.length) {
   848→      this.ghostCar.group.visible = false;
   849→      return;
   850→    }
   851→    this.ghostCar.group.visible = true;
   852→    const tMs = this.raceTime;
   853→    const frames = this.ghost;
   854→    // binary search
   855→    let lo = 0;
   856→    let hi = frames.length - 1;
   857→    while (lo < hi) {
   858→      const mid = (lo + hi) >> 1;
   859→      if (frames[mid].t < tMs) lo = mid + 1;
   860→      else hi = mid;
   861→    }
   862→    const i = Math.max(1, lo);
   863→    const a = frames[i - 1];
   864→    const b = frames[Math.min(frames.length - 1, i)];
   865→    const span = Math.max(1e-3, b.t - a.t);
   866→    const f = clamp((tMs - a.t) / span, 0, 1);
   867→    const g = this.ghostCar.group;
   868→    g.position.set(a.x + (b.x - a.x) * f, a.y + (b.y - a.y) * f, a.z + (b.z - a.z) * f);
   869→    let dy = b.yaw - a.yaw;
   870→    while (dy > Math.PI) dy -= Math.PI * 2;
   871→    while (dy < -Math.PI) dy += Math.PI * 2;
   872→    g.rotation.set(0, a.yaw + dy * f, 0);
   873→  }

   879→  private updateCamera(dt: number) {
   880→    if (this.phase === "title") return; // handled in updateTitle
   881→    const p = this.physics;
   882→    const fwd = this.tmpV.set(Math.sin(p.yaw), 0, Math.cos(p.yaw));
   883→
   884→    if (this.phase === "countdown") {
   885→      // cinematic sweep down to the chase position
   886→      const t = clamp(1 - this.countdownT / 3, 0, 1);
   887→      const e = t * t * (3 - 2 * t);
   888→      const ang = 2.4 - e * 2.4;
   889→      const dist = 30 - 24 * e;
   890→      this.camPos.set(
   891→        p.pos.x + Math.sin(ang) * dist + fwd.x * -4,
   892→        p.pos.y + 14 - 11.5 * e,
   893→        p.pos.z + Math.cos(ang) * dist + fwd.z * -4
   894→      );
   895→      this.camLook.set(p.pos.x + fwd.x * 6, p.pos.y + 1.2, p.pos.z + fwd.z * 6);
   896→      this.camera.position.copy(this.camPos);
   897→      this.camera.up.set(0, 1, 0);
   898→      this.camera.lookAt(this.camLook);
   899→      this.camera.fov = damp(this.camera.fov, 66, 3, dt);
   900→      this.camera.updateProjectionMatrix();
   901→      return;
   902→    }
   903→
   904→    if (this.phase === "finished") {
   905→      const a = this.finishOrbitT * 0.5;
   906→      this.camPos.set(p.pos.x + Math.sin(a) * 9, p.pos.y + 3.2, p.pos.z + Math.cos(a) * 9);
   907→      this.camLook.copy(p.pos).add(new THREE.Vector3(0, 1, 0));
   908→      this.camera.position.copy(this.camPos);
   909→      this.camera.up.set(0, 1, 0);
   910→      this.camera.lookAt(this.camLook);
   911→      return;
   912→    }
   913→
   914→    // chase camera
   915→    const mode = this.settings.cameraMode;
   916→    const dist = mode === 1 ? 9.6 : 6.4;
   917→    const height = mode === 1 ? 3.8 : 2.55;
   918→    if (mode === 2) {
   919→      // hood cam
   920→      this.camPos.set(p.pos.x + fwd.x * 0.5, p.pos.y + 1.35, p.pos.z + fwd.z * 0.5);
   921→      this.camLook.set(p.pos.x + fwd.x * 12, p.pos.y + 1.6, p.pos.z + fwd.z * 12);
   922→      this.camera.position.copy(this.camPos);
   923→      this.camera.up.set(0, 1, 0);
   924→      this.camera.lookAt(this.camLook);
   925→    } else {
   926→      const ffX = clamp(p.vel.x * 0.16, -7, 7);
   927→      const ffZ = clamp(p.vel.z * 0.16, -7, 7);
   928→      const tx = p.pos.x - fwd.x * dist + ffX;
   929→      const ty = p.pos.y + height;
   930→      const tz = p.pos.z - fwd.z * dist + ffZ;
   931→      const rate = 6.5;
   932→      this.camPos.x = damp(this.camPos.x, tx, rate, dt);
   933→      this.camPos.y = damp(this.camPos.y, ty, rate, dt);
   934→      this.camPos.z = damp(this.camPos.z, tz, rate, dt);
   935→      this.camLook.set(p.pos.x + fwd.x * 7, p.pos.y + 1.3, p.pos.z + fwd.z * 7);
   936→      this.camera.position.copy(this.camPos);
   937→      this.camera.up.set(0, 1, 0);
   938→      this.camera.lookAt(this.camLook);
   939→      // banking roll
   940→      const targetRoll = clamp(-p.slip * 0.3 - this.lastSteer * 0.05, -0.24, 0.24);
   941→      this.camRoll = damp(this.camRoll, targetRoll, 5, dt);
   942→      this.camera.rotateZ(this.camRoll);
   943→    }
   944→
   945→    // wall-ride lean
   946→    this.wallRideLean = damp(this.wallRideLean, p.wallRiding ? clamp(p.lateral / 24, 0, 1) * 0.5 : 0, 4, dt);
   947→    if (this.wallRideLean > 0.01) this.camera.rotateZ(this.wallRideLean);
   948→
   949→    // FOV kick
   950→    const speedFrac = clamp(p.speed / 55, 0, 1.1);
   951→    const targetFov = 64 + speedFrac * 10 + (p.boosting ? 9 : 0);
   952→    this.fov = damp(this.fov, targetFov, 4, dt);
   953→    this.camera.fov = this.fov;
   954→    this.camera.updateProjectionMatrix();
   955→
   956→    // restrained screen shake
   957→    this.trauma = Math.max(0, this.trauma - dt * 1.6);
   958→    const sh = this.trauma * this.trauma;
   959→    if (sh > 0.001) {
   960→      this.camera.position.x += (Math.random() - 0.5) * sh * 0.55;
   961→      this.camera.position.y += (Math.random() - 0.5) * sh * 0.4;
   962→      this.camera.position.z += (Math.random() - 0.5) * sh * 0.55;
   963→      this.camera.rotateZ((Math.random() - 0.5) * sh * 0.03);
   964→    }
   965→  }

  1017→function fmt(ms: number): string {
