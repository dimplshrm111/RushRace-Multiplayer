     6→/** Smooth 0..1 step. */
     7→function sstep(a: number, b: number, x: number): number {
     8→  const t = clamp((x - a) / (b - a), 0, 1);
     9→  return t * t * (3 - 2 * t);
    10→}

    12→function wrapIdx(i: number, n: number): number {
    13→  return ((i % n) + n) % n;
    14→}

   135→/** How deep the dry-riverbed pit floor sits below road level at gap progress t (0..1). */
   136→const PIT_DEPTH = -9;
   137→export function pitDepthAt(gapT: number): number {
   138→  if (gapT < 0) return 0;
   139→  // flat landing floor under the cliff, then a smooth drivable climb back to road level
   140→  if (gapT < 0.12) return PIT_DEPTH;
   141→  if (gapT > 0.99) return 0;
   142→  const u = (gapT - 0.12) / 0.87; // 0..1 across the exit ramp
   143→  // smoothstep-blended ramp: gentle at both ends, steady grade in the middle
   144→  const s = u * u * (3 - 2 * u);
   145→  return PIT_DEPTH * (1 - s);
   146→}

   181→  private buildSpline() {
   182→    const pts = CONTROL.map((p) => new THREE.Vector3(p[0], p[1], p[2]));
   183→    const curve = new THREE.CatmullRomCurve3(pts, true, "catmullrom", 0.5);
   184→    const approxLen = curve.getLength();
   185→    const n = Math.max(600, Math.round(approxLen / this.step));
   186→    const spaced = curve.getSpacedPoints(n);
   187→    this.N = n;
   188→    this.length = n * this.step;
   189→    const raw = spaced.slice(0, n);
   190→
   191→    const dirs: THREE.Vector2[] = [];
   192→    for (let i = 0; i < n; i++) {
   193→      const a = raw[wrapIdx(i - 1, n)];
   194→      const b = raw[wrapIdx(i + 1, n)];
   195→      const d = new THREE.Vector2(b.x - a.x, b.z - a.z);
   196→      if (d.lengthSq() < 1e-8) d.set(0, -1);
   197→      dirs.push(d.normalize());
   198→    }
   199→    // curvature → banking (outer edge raised)
   200→    const banks = new Array<number>(n).fill(0);
   201→    for (let i = 0; i < n; i++) {
   202→      const d0 = dirs[wrapIdx(i - 2, n)];
   203→      const d1 = dirs[wrapIdx(i + 2, n)];
   204→      const crossY = d0.x * d1.y - d0.y * d1.x; // >0 = left turn
   205→      const dot = clamp(d0.dot(d1), -1, 1);
   206→      const dTheta = Math.atan2(crossY, dot);
   207→      banks[i] = clamp((dTheta / (4 * this.step)) * 46, -1, 1) * 3.0;
   208→    }
   209→    const sm = new Array<number>(n).fill(0);
   210→    for (let i = 0; i < n; i++) {
   211→      let acc = 0;
   212→      for (let k = -9; k <= 9; k++) acc += banks[wrapIdx(i + k, n)];
   213→      sm[i] = acc / 19;
   214→    }
   215→
   216→    for (let i = 0; i < n; i++) {
   217→      const p = raw[i];
   218→      const d = dirs[i];
   219→      const r = new THREE.Vector2(-d.y, d.x); // right = (-dz, dx)
   220→      this.samples.push({
   221→        x: p.x,
   222→        y: p.y,
   223→        z: p.z,
   224→        dx: d.x,
   225→        dz: d.y,
   226→        rx: r.x,
   227→        rz: r.y,
   228→        hw: 9,
   229→        shoulderL: 8,
   230→        shoulderR: 8,
   231→        bank: sm[i],
   232→        gap: false,
   233→        gapT: -1,
   234→        rampP: -1,
   235→        dualT: 0,
   236→        wallRideR: false,
   237→        wallRideL: false,
   238→        riverbed: false,
   239→        tunnel: false,
   240→        wallHL: 24,
   241→        wallHR: 24,
   242→      });
   243→    }
   244→  }
   245→
   246→  private idxNear(x: number, z: number): number {
   247→    let best = 0;
   248→    let bd = Infinity;
   249→    for (let i = 0; i < this.N; i++) {
   250→      const s = this.samples[i];
   251→      const d = (s.x - x) * (s.x - x) + (s.z - z) * (s.z - z);
   252→      if (d < bd) {
   253→        bd = d;
   254→        best = i;
   255→      }
   256→    }
   257→    return best;
   258→  }

   260→  private range(a: number, b: number): [number, number] {
   261→    return [
   262→      this.idxNear(CONTROL[a][0], CONTROL[a][2]),
   263→      this.idxNear(CONTROL[b][0], CONTROL[b][2]),
   264→    ];
   265→  }

   267→  private forEachInRange(r: [number, number], fn: (i: number, t: number) => void) {
   268→    const [a, b] = r;
   269→    const span = (b - a + this.N) % this.N;
   270→    for (let k = 0; k <= span; k++) fn(wrapIdx(a + k, this.N), span > 0 ? k / span : 1);
   271→  }

   277→  private buildZones() {
   278→    const S = this.samples;
   279→
   280→    // --- launch ramp before the gap (last 58% of approach → crest) ---
   281→    const ramp = this.range(2, 3);
   282→    this.forEachInRange(ramp, (i, t) => {
   283→      S[i].rampP = clamp((t - 0.42) / 0.58, 0, 1);
   284→    });
   285→
   286→    // --- the GAP over the dry riverbed canyon ---
   287→    // The pit is a smooth valley, not a box canyon: a short flat landing
   288→    // floor under the cliff, then a long drivable ramp that climbs back up
   289→    // to the landing road — undershoot the jump and you can still drive out.
   290→    const crestI = this.idxNear(CONTROL[3][0], CONTROL[3][2]);
   291→    const landI = this.idxNear(CONTROL[4][0], CONTROL[4][2]);
   292→    const gapSpan = (landI - crestI + this.N) % this.N;
   293→    const gapStart = crestI + 2;
   294→    const gapEnd = crestI + Math.round(gapSpan * 0.62);
   295→    for (let k = gapStart; k <= gapEnd; k++) {
   296→      const s = S[wrapIdx(k, this.N)];
   297→      s.gap = true;
   298→      s.gapT = (k - gapStart) / Math.max(1, gapEnd - gapStart);
   299→      s.shoulderL = 3.2;
   300→      s.shoulderR = 3.2;
   301→      s.wallHL = 30;
   302→      s.wallHR = 30;
   303→    }
   304→
   305→    // --- natural arch zone: taller walls ---
   306→    const archI = this.idxNear(CONTROL[6][0], CONTROL[6][2]);
   307→    for (let k = archI - 14; k <= archI + 14; k++) {
   308→      const s = S[wrapIdx(k, this.N)];
   309→      s.wallHL = 42;
   310→      s.wallHR = 42;
   311→    }
   312→
   313→    // --- wall-ride carousel: banked outer wall on the right ---
   314→    const carousel = this.range(8, 11);
   315→    this.forEachInRange(carousel, (i, t) => {
   316→      const edge = sstep(0, 0.14, t) * (1 - sstep(0.86, 1, t));
   317→      const s = S[i];
   318→      s.wallRideR = edge > 0.01;
   319→      s.shoulderR = 26 * edge + 8 * (1 - edge);
   320→      s.wallHR = 30;
   321→      s.bank = s.bank * (1 - edge) + (s.bank * 0.4 + 2.4) * edge;
   322→    });
   323→
   324→    // --- tunnel through the mesa ---
   325→    const tunnel = this.range(13, 16);
   326→    this.forEachInRange(tunnel, (i, t) => {
   327→      const edge = sstep(0, 0.05, t) * (1 - sstep(0.95, 1, t));
   328→      const s = S[i];
   329→      s.tunnel = edge > 0.01;
   330→      s.hw = 8.5 + 0.5 * (1 - edge);
   331→      s.shoulderL = 1.5 + 6.5 * (1 - edge);
   332→      s.shoulderR = 1.5 + 6.5 * (1 - edge);
   333→    });
   334→
   335→    // --- dual-line downhill: high terrace on the left ---
   336→    const dual = this.range(16, 18);
   337→    this.forEachInRange(dual, (i, t) => {
   338→      const s = S[i];
   339→      s.dualT = sstep(0, 0.12, t) * (1 - sstep(0.88, 1, t));
   340→      s.shoulderL = 9.5;
   341→    });
   342→
   343→    // --- riverbed: wide sandy floor ---
   344→    const river = this.range(19, 24);
   345→    this.forEachInRange(river, (i, t) => {
   346→      const edge = sstep(0, 0.06, t) * (1 - sstep(0.94, 1, t));
   347→      const s = S[i];
   348→      s.riverbed = true;
   349→      s.hw = 9 + 11 * edge;
   350→      s.shoulderL = 8 + 6 * edge;
   351→      s.shoulderR = 8 + 6 * edge;
   352→      s.wallHL = 20;
   353→      s.wallHR = 20;
   354→    });
   355→
   356→    // --- low walls near the start/finish for the vista ---
   357→    for (let k = -30; k <= 40; k++) {
   358→      const s = S[wrapIdx(k, this.N)];
   359→      s.wallHL = Math.min(s.wallHL, 14);
   360→      s.wallHR = Math.min(s.wallHR, 14);
   361→    }
   362→
   363→    // wall height noise
   364→    for (let i = 0; i < this.N; i++) {
   365→      const s = S[i];
   366→      const n =
   367→        Math.sin(i * 0.093) * 0.5 + Math.sin(i * 0.031 + 2.1) * 0.3 + Math.sin(i * 0.017 + 4) * 0.2;
   368→      s.wallHL = Math.max(10, s.wallHL + n * 7);
   369→      s.wallHR = Math.max(10, s.wallHR + n * 7 + Math.sin(i * 0.05) * 2);
   370→    }
   371→
   372→    // --- kicker bumps in the riverbed ---
   373→    const rB = this.idxNear(CONTROL[21][0], CONTROL[21][2]);
   374→    const rC = this.idxNear(CONTROL[22][0], CONTROL[22][2]);
   375→    const rD = this.idxNear(CONTROL[23][0], CONTROL[23][2]);
   376→    this.kickers = [
   377→      { s: (rB + 9) * this.step, lat: -8, r: 7, h: 2.1 },
   378→      { s: (rC + 10) * this.step, lat: 7, r: 7, h: 2.2 },
   379→      { s: (rD + 12) * this.step, lat: -5, r: 6.5, h: 1.9 },
   380→    ];
   381→
   382→    // --- checkpoint gates in racing order ---
   383→    const gateAnchors: { c: number; finish: boolean }[] = [
   384→      { c: 4, finish: false }, // gap landing
   385→      { c: 6, finish: false }, // arch
   386→      { c: 7, finish: false }, // climb
   387→      { c: 10, finish: false }, // carousel apex
   388→      { c: 13, finish: false }, // tunnel entry
   389→      { c: 16, finish: false }, // tunnel exit
   390→      { c: 18, finish: false }, // dual-line merge
   391→      { c: 22, finish: false }, // riverbed mid
   392→      { c: 27, finish: false }, // final arc
   393→      { c: 0, finish: true }, // start/finish
   394→    ];
   395→    for (const g of gateAnchors) {
   396→      const idx = this.idxNear(CONTROL[g.c][0], CONTROL[g.c][2]);
   397→      const s = S[idx];
   398→      this.gates.push({ idx, pos: new THREE.Vector3(s.x, s.y, s.z), isFinish: g.finish });
   399→      this.gatePts.push({ x: s.x, z: s.z, finish: g.finish });
   400→    }
   401→
   402→    for (let i = 0; i < this.N; i += 6) this.minimapPts.push({ x: S[i].x, z: S[i].z });
   403→
   404→    this.addObstacles();
   405→  }

   407→  private addObstacles() {
   408→    const rnd = mulberry32(20240607);
   409→    // riverbed boulders
   410→    const rA = this.idxNear(CONTROL[20][0], CONTROL[20][2]);
   411→    const rB = this.idxNear(CONTROL[23][0], CONTROL[23][2]);
   412→    const span = (rB - rA + this.N) % this.N;
   413→    for (let k = 0; k < 8; k++) {
   414→      const idx = wrapIdx(rA + Math.round(span * (0.12 + 0.78 * rnd())), this.N);
   415→      const s = this.samples[idx];
   416→      const lat = (rnd() * 9 + 4.5) * (rnd() > 0.5 ? 1 : -1);
   417→      const r = 1.7 + rnd() * 0.9;
   418→      this.obstacles.push({
   419→        x: s.x + s.rx * lat,
   420→        z: s.z + s.rz * lat,
   421→        r,
   422→        top: this.heightAt(idx, lat) + r * 1.5,
   423→        lastNear: -1e9,
   424→      });
   425→    }
   426→    // rock fins on the dual-line low (right) side
   427→    const dA = this.idxNear(CONTROL[17][0], CONTROL[17][2]);
   428→    for (let k = 0; k < 3; k++) {
   429→      const idx = wrapIdx(dA + 8 + k * 14, this.N);
   430→      const s = this.samples[idx];
   431→      const lat = 5.5 + rnd() * 3;
   432→      this.obstacles.push({
   433→        x: s.x + s.rx * lat,
   434→        z: s.z + s.rz * lat,
   435→        r: 1.6,
   436→        top: this.heightAt(idx, lat) + 3.4,
   437→        lastNear: -1e9,
   438→      });
   439→    }
   440→    // arch pillars
   441→    const archI = this.idxNear(CONTROL[6][0], CONTROL[6][2]);
   442→    const as = this.samples[archI];
   443→    for (const side of [-1, 1]) {
   444→      const lat = side * (as.hw + 2.2);
   445→      this.obstacles.push({
   446→        x: as.x + as.rx * lat,
   447→        z: as.z + as.rz * lat,
   448→        r: 2.3,
   449→        top: this.heightAt(archI, lat) + 9,
   450→        lastNear: -1e9,
   451→      });
   452→    }
   453→    // two rocks flanking the landing zone (near-miss bait)
   454→    const landI = this.idxNear(CONTROL[4][0], CONTROL[4][2]);
   455→    const ls = this.samples[landI];
   456→    for (const side of [-1, 1]) {
   457→      const lat = side * (ls.hw + 1.4);
   458→      this.obstacles.push({
   459→        x: ls.x + ls.rx * lat,
   460→        z: ls.z + ls.rz * lat,
   461→        r: 2.0,
   462→        top: this.heightAt(landI, lat) + 3,
   463→        lastNear: -1e9,
   464→      });
   465→    }
   466→  }

   476→  groundAt(i: number, lat: number, out: GroundInfo): GroundInfo {
   477→    const s = this.sample(i);
   478→    const al = Math.abs(lat);
   479→    if (s.gap && al < s.hw + 3) {
   480→      out.h = s.y + pitDepthAt(s.gapT);
   481→      out.onRoad = false;
   482→      out.inPitZone = true;
   483→      return out;
   484→    }
   485→    out.inPitZone = false;
   486→    let h = s.y;
   487→    const u = clamp(lat / s.hw, -1, 1);
   488→    h += s.bank * u;
   489→    if (s.riverbed) h += 0.45 * u * u;
   490→    if (s.dualT > 0 && lat > 3) {
   491→      const t = sstep(3, 5, lat) * (1 - sstep(13, 15.5, lat));
   492→      h += 2.6 * s.dualT * t;
   493→    }
   494→    if (s.wallRideR && lat > 4) h += 13 * Math.pow(Math.min(1, (lat - 4) / 24), 2.4);
   495→    if (s.wallRideL && lat < -4) h += 13 * Math.pow(Math.min(1, (-lat - 4) / 24), 2.4);
   496→    if (s.rampP >= 0 && al < s.hw + 1) {
   497→      const latFade = 1 - sstep(4.5, s.hw, al);
   498→      h += 3.4 * sstep(0.1, 0.92, s.rampP) * latFade;
   499→    }
   500→    let onRoad = al <= s.hw;
   501→    if (al > s.hw) {
   502→      if (s.wallRideR && lat > 0) onRoad = true;
   503→      else if (s.dualT > 0 && lat < 0 && lat > 3) onRoad = true;
   504→      else {
   505→        const sh = lat > 0 ? s.shoulderR : s.shoulderL;
   506→        if (sh > 0) h -= 0.55 * Math.min(1, (al - s.hw) / Math.max(0.5, sh));
   507→      }
   508→    }
   509→    const sHere = wrapIdx(i, this.N) * this.step;
   510→    for (const k of this.kickers) {
   511→      let ds = Math.abs(sHere - k.s);
   512→      ds = Math.min(ds, this.length - ds);
   513→      const dl = lat - k.lat;
   514→      const d2 = ds * ds + dl * dl;
   515→      if (d2 < k.r * k.r) {
   516→        const q = Math.sqrt(d2) / k.r;
   517→        h += k.h * (0.5 + 0.5 * Math.cos(Math.PI * q));
   518→      }
   519→    }
   520→    out.h = h;
   521→    out.onRoad = onRoad;
   522→    return out;
   523→  }
   524→
   525→  heightAt(i: number, lat: number): number {
   526→    return this.groundAt(i, lat, TMP_GROUND).h;
   527→  }
   528→
   529→  // ---- continuous (interpolated) sampling ----
   530→  // The physics used to snap to the discrete 2 m sample grid, which made the
   531→  // car's height (and everything derived from it) pop at every sample boundary
   532→  // — a visible buzz/jiggle at speed. These queries blend adjacent samples so
   533→  // the driven surface matches the rendered road exactly and stays smooth.
   534→
   535→  /** Ground query at a continuous sample coordinate p (= idx + fraction ahead). */
   536→  groundAtP(p: number, lat: number, out: GroundInfo): GroundInfo {
   537→    const i0 = Math.floor(p);
   538→    const fr = clamp(p - i0, 0, 1);
   539→    const a = this.groundAt(i0, lat, TMP_GA);
   540→    const b = this.groundAt(i0 + 1, lat, TMP_GB);
   541→    if (a.inPitZone !== b.inPitZone) {
   542→      // cliff lip at the gap — keep the step crisp so launches still fire
   543→      const use = fr < 0.5 ? a : b;
   544→      out.h = use.h;
   545→      out.onRoad = use.onRoad;
   546→      out.inPitZone = use.inPitZone;
   547→      return out;
   548→    }
   549→    out.h = a.h + (b.h - a.h) * fr;
   550→    out.onRoad = fr < 0.5 ? a.onRoad : b.onRoad;
   551→    out.inPitZone = a.inPitZone;
   552→    return out;
   553→  }
   554→
   555→  heightAtP(p: number, lat: number): number {
   556→    return this.groundAtP(p, lat, TMP_GROUND).h;
   557→  }
   558→
   559→  /** Lateral slope at a continuous sample coordinate. */
   560→  slopeLatAtP(p: number, lat: number): number {
   561→    return (this.heightAtP(p, lat + 0.9) - this.heightAtP(p, lat - 0.9)) / 1.8;
   562→  }
   563→
   564→  /** Forward slope at a continuous sample coordinate. */
   565→  slopeFwdAtP(p: number, lat: number): number {
   566→    const d = 0.9 / this.step;
   567→    return (this.heightAtP(p + d, lat) - this.heightAtP(p - d, lat)) / 1.8;
   568→  }
   569→
   570→  slopeLatAt(i: number, lat: number): number {
   571→    const a = this.heightAt(wrapIdx(i, this.N), lat + 0.9);
   572→    const b = this.heightAt(wrapIdx(i, this.N), lat - 0.9);
   573→    return (a - b) / 1.8;
   574→  }
   575→
   576→  slopeFwdAt(i: number, lat: number): number {
   577→    const a = this.heightAt(wrapIdx(i + 1, this.N), lat);
   578→    const b = this.heightAt(wrapIdx(i - 1, this.N), lat);
   579→    return (a - b) / (2 * this.step);
   580→  }
   581→
   582→  wallDist(i: number, side: 1 | -1): number {
   583→    const s = this.sample(i);
   584→    return side > 0 ? s.hw + s.shoulderR + 0.3 : s.hw + s.shoulderL + 0.3;
   585→  }
   586→
   587→  nearestIdx(x: number, z: number, hint: number): number {
   588→    let best = hint;
   589→    let bd = Infinity;
   590→    for (let k = -35; k <= 95; k++) {
   591→      const i = wrapIdx(hint + k, this.N);
   592→      const s = this.samples[i];
   593→      const d = (s.x - x) * (s.x - x) + (s.z - z) * (s.z - z);
   594→      if (d < bd) {
   595→        bd = d;
   596→        best = i;
   597→      }
   598→    }
   599→    if (bd > 90 * 90) best = this.idxNear(x, z);
   600→    return best;
   601→  }
   602→
   603→  lateralOf(i: number, x: number, z: number): number {
   604→    const s = this.sample(i);
   605→    return (x - s.x) * s.rx + (z - s.z) * s.rz;
   606→  }
   607→
   608→  posAt(i: number, lat: number, y?: number): THREE.Vector3 {
   609→    const s = this.sample(i);
   610→    return new THREE.Vector3(s.x + s.rx * lat, y ?? s.y, s.z + s.rz * lat);
   611→  }

   633→  private buildWalls() {
   634→    const S = this.samples;
   635→    const n = this.N;
   636→    for (const side of [-1, 1] as const) {
   637→      const pos: number[] = [];
   638→      const col: number[] = [];
   639→      const idx: number[] = [];
   640→      const strata = PALETTE.rockStrata.map((c) => new THREE.Color(c));
   641→      const cBase = new THREE.Color(PALETTE.rockBase);
   642→      for (let i = 0; i < n; i++) {
   643→        const s = S[i];
   644→        const wd = this.wallDist(i, side) - 0.3;
   645→        const bx = s.x + s.rx * wd * side;
   646→        const bz = s.z + s.rz * wd * side;
   647→        const wallH = side > 0 ? s.wallHR : s.wallHL;
   648→        const lean = 7 + ((i * 37) % 5);
   649→        const tx = s.x + (s.rx * wd + lean) * side;
   650→        const tz = s.z + (s.rz * wd + lean) * side;
   651→        const by = s.gap ? s.y + pitDepthAt(s.gapT) - 0.4 : this.heightAt(i, wd * side) - 0.2;
   652→        const ty = s.y + wallH;
   653→        pos.push(bx, by, bz, tx, ty, tz);
   654→        for (const yv of [by, ty]) {
   655→          const band = Math.floor((yv + Math.sin(i * 0.7 + side) * 1.4) / 4.2);
   656→          const c = strata[((band % strata.length) + strata.length) % strata.length].clone();
   657→          c.lerp(cBase, 0.25 + 0.2 * Math.abs(Math.sin(i * 0.13 + band)));
   658→          col.push(c.r, c.g, c.b);
   659→        }
   660→      }
   661→      for (let i = 0; i < n; i++) {
   662→        const a = i * 2;
   663→        const b = ((i + 1) % n) * 2;
   664→        if (side > 0) {
   665→          // normal faces the track (-right)
   666→          idx.push(a, a + 1, b, a + 1, b + 1, b);
   667→        } else {
   668→          idx.push(a, b, a + 1, b, b + 1, a + 1);
   669→        }
   670→      }
   671→      const g = new THREE.BufferGeometry();
   672→      g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   673→      g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   674→      g.setIndex(idx);
   675→      g.computeVertexNormals();
   676→      const m = new THREE.MeshLambertMaterial({ vertexColors: true, flatShading: true });
   677→      const mesh = new THREE.Mesh(g, m);
   678→      mesh.castShadow = true;
   679→      mesh.receiveShadow = true;
   680→      this.group.add(mesh);
   681→      this.disposables.push(g, m);
   682→    }
   683→  }

   685→  private ringLats(i: number): number[] {
   686→    const s = this.sample(i);
   687→    const wdL = -this.wallDist(i, -1);
   688→    const wdR = this.wallDist(i, 1);
   689→    const lats: number[] = [wdL];
   690→    for (const l of ROAD_COLS) {
   691→      if (l > wdL + 0.05 && l < wdR - 0.05) lats.push(l);
   692→    }
   693→    lats.push(wdR);
   694→    return lats;
   695→  }

   697→  private buildRoad() {
   698→    const n = this.N;
   699→    const pos: number[] = [];
   700→    const col: number[] = [];
   701→    const uv: number[] = [];
   702→    const idx: number[] = [];
   703→    const cRoad = new THREE.Color(PALETTE.road);
   704→    const cRoadDust = new THREE.Color(PALETTE.roadDust);
   705→    const cSand = new THREE.Color(PALETTE.sand);
   706→    const cPit = new THREE.Color(PALETTE.pitSand);
   707→    const cTerr = new THREE.Color(0xcf8a52);
   708→    const gi: GroundInfo = { h: 0, onRoad: false, inPitZone: false };
   709→
   710→    const ringLats: number[][] = [];
   711→    const ringStart: number[] = [];
   712→    for (let i = 0; i < n; i++) {
   713→      ringStart[i] = pos.length / 3;
   714→      const s = this.samples[i];
   715→      const lats = this.ringLats(i);
   716→      ringLats.push(lats);
   717→      for (const l of lats) {
   718→        this.groundAt(i, l, gi);
   719→        let c: THREE.Color;
   720→        if (gi.inPitZone) c = cPit;
   721→        else if (Math.abs(l) <= s.hw + 0.01) c = cRoad.clone().lerp(cRoadDust, 0.3 + 0.2 * Math.sin(i * 0.37));
   722→        else if (s.dualT > 0 && l < -3) c = cTerr.clone();
   723→        else if (s.wallRideR && l > s.hw) {
   724→          const t = clamp((l - 4) / 24, 0, 1);
   725→          c = cRoad.clone().lerp(cTerr, t * 0.8);
   726→        } else c = cSand.clone().offsetHSL(0, 0, Math.sin(i * 0.53 + l) * 0.015);
   727→        pos.push(s.x + s.rx * l, gi.h, s.z + s.rz * l);
   728→        col.push(c.r, c.g, c.b);
   729→        uv.push(l / 12, (i * this.step) / 7);
   730→      }
   731→    }
   732→    for (let i = 0; i < n; i++) {
   733→      const j = (i + 1) % n;
   734→      if (this.samples[i].gap || this.samples[j].gap) continue;
   735→      const a = ringStart[i];
   736→      const b = ringStart[j];
   737→      const m = ringLats[i].length; // stitch with ring i's column count on both rings
   738→      for (let k = 0; k < m - 1; k++) {
   739→        idx.push(a + k, a + k + 1, b + k, a + k + 1, b + k + 1, b + k);
   740→      }
   741→    }
   742→    const g = new THREE.BufferGeometry();
   743→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   744→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   745→    g.setAttribute("uv", new THREE.Float32BufferAttribute(uv, 2));
   746→    g.setIndex(idx);
   747→    g.computeVertexNormals();
   748→    const tex = makeSandNoiseTexture();
   749→    const m = new THREE.MeshLambertMaterial({ vertexColors: true, map: tex, flatShading: true });
   750→    const mesh = new THREE.Mesh(g, m);
   751→    mesh.receiveShadow = true;
   752→    this.group.add(mesh);
   753→    this.disposables.push(g, m, tex);
   754→  }

   756→  /** Painted route markings — edge lines + forward chevrons so the line is always readable. */
   757→  private buildRouteMarks() {
   758→    const n = this.N;
   759→    const lineMat = new THREE.MeshBasicMaterial({
   760→      color: 0xf7e7c3,
   761→      fog: true,
   762→      side: THREE.DoubleSide,
   763→      polygonOffset: true,
   764→      polygonOffsetFactor: -3,
   765→      polygonOffsetUnits: -3,
   766→    });
   767→    const chevMat = new THREE.MeshBasicMaterial({
   768→      color: 0xfffbe8,
   769→      fog: true,
   770→      side: THREE.DoubleSide,
   771→      polygonOffset: true,
   772→      polygonOffsetFactor: -4,
   773→      polygonOffsetUnits: -4,
   774→    });
   775→
   776→    // ---- continuous edge lines hugging the road edge ----
   777→    const pos: number[] = [];
   778→    const idx: number[] = [];
   779→    const lineHalf = 0.19;
   780→    for (const side of [-1, 1] as const) {
   781→      const start = pos.length / 3;
   782→      for (let i = 0; i < n; i++) {
   783→        const s = this.samples[i];
   784→        const latE = (s.hw - 0.85) * side;
   785→        const y = this.heightAt(i, latE) + 0.05;
   786→        pos.push(
   787→          s.x + s.rx * (latE - lineHalf), y, s.z + s.rz * (latE - lineHalf),
   788→          s.x + s.rx * (latE + lineHalf), y, s.z + s.rz * (latE + lineHalf)
   789→        );
   790→      }
   791→      for (let i = 0; i < n; i++) {
   792→        const j = (i + 1) % n;
   793→        if (this.samples[i].gap || this.samples[j].gap) continue;
   794→        const a = start + i * 2;
   795→        const b = start + j * 2;
   796→        idx.push(a, a + 1, b, a + 1, b + 1, b);
   797→      }
   798→    }
   799→    const lineGeo = new THREE.BufferGeometry();
   800→    lineGeo.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   801→    lineGeo.setIndex(idx);
   802→    const lineMesh = new THREE.Mesh(lineGeo, lineMat);
   803→    lineMesh.frustumCulled = false; // spans the whole loop
   804→    this.group.add(lineMesh);
   805→    this.disposables.push(lineGeo, lineMat);
   806→
   807→    // ---- forward-pointing chevrons along the centerline ----
   808→    const gateBlocked = new Set<number>();
   809→    for (const g of this.gates) {
   810→      for (let k = -4; k <= 4; k++) gateBlocked.add(wrapIdx(g.idx + k, n));
   811→    }
   812→    const cpos: number[] = [];
   813→    const cidx: number[] = [];
   814→    const arm = (f0: number, l0: number, f1: number, l1: number, w: number, i: number) => {
   815→      const df = f1 - f0;
   816→      const dl = l1 - l0;
   817→      const len = Math.hypot(df, dl);
   818→      if (len < 1e-4) return;
   819→      const px = (-dl / len) * (w * 0.5);
   820→      const pl = (df / len) * (w * 0.5);
   821→      const corners: [number, number][] = [
   822→        [f0 + px, l0 + pl],
   823→        [f1 + px, l1 + pl],
   824→        [f1 - px, l1 - pl],
   825→        [f0 - px, l0 - pl],
   826→      ];
   827→      const base = cpos.length / 3;
   828→      const s = this.samples[i];
   829→      for (const [f, l] of corners) {
   830→        const x = s.x + s.dx * f + s.rx * l;
   831→        const z = s.z + s.dz * f + s.rz * l;
   832→        const y = this.heightAtP(i + f / this.step, l) + 0.07;
   833→        cpos.push(x, y, z);
   834→      }
   835→      cidx.push(base, base + 1, base + 2, base, base + 2, base + 3);
   836→    };
   837→    for (let i = 6; i < n; i += 14) {
   838→      const s = this.samples[i];
   839→      if (s.gap || gateBlocked.has(i)) continue;
   840→      arm(1.15, 0, -0.45, 1.35, 0.46, i);
   841→      arm(1.15, 0, -0.45, -1.35, 0.46, i);
   842→    }
   843→    const chevGeo = new THREE.BufferGeometry();
   844→    chevGeo.setAttribute("position", new THREE.Float32BufferAttribute(cpos, 3));
   845→    chevGeo.setIndex(cidx);
   846→    const chevMesh = new THREE.Mesh(chevGeo, chevMat);
   847→    chevMesh.frustumCulled = false;
   848→    this.group.add(chevMesh);
   849→    this.disposables.push(chevGeo, chevMat);
   850→  }

   852→  private buildPit() {
   853→    const n = this.N;
   854→    const pos: number[] = [];
   855→    const idx: number[] = [];
   856→    const col: number[] = [];
   857→    const cPit = new THREE.Color(PALETTE.pitSand);
   858→    const cDark = new THREE.Color(0xb08a58);
   859→    let v = 0;
   860→    const pushTri = (a: number[], b: number[], c: number[], shade: number) => {
   861→      for (const p of [a, b, c]) pos.push(p[0], p[1], p[2]);
   862→      const cc = cPit.clone().lerp(cDark, shade);
   863→      col.push(cc.r, cc.g, cc.b, cc.r, cc.g, cc.b, cc.r, cc.g, cc.b);
   864→      idx.push(v, v + 1, v + 2);
   865→      v += 3;
   866→    };
   867→    const pushQuad = (a: number[], b: number[], c: number[], d: number[]) => {
   868→      pushTri(a, b, c, 0);
   869→      pushTri(a, c, d, 0.25);
   870→    };
   871→    const PL = 12.4;
   872→    let i = 0;
   873→    while (i < n) {
   874→      if (!this.samples[i].gap) {
   875→        i++;
   876→        continue;
   877→      }
   878→      let j = i;
   879→      while (j < n && this.samples[j].gap) j++;
   880→      const jEnd = Math.min(j, n - 1); // gap never crosses the wrap in this track
   881→      // smooth valley floor — follows the same profile the physics drives on,
   882→      // so the rendered sand matches the collision surface exactly
   883→      const floorY = (k: number) => {
   884→        const sk = this.samples[wrapIdx(k, n)];
   885→        return sk.y + pitDepthAt(sk.gapT);
   886→      };
   887→      for (let k = i; k < jEnd; k++) {
   888→        const s0 = this.samples[k];
   889→        const s1 = this.samples[k + 1];
   890→        for (const [l0, l1] of [
   891→          [-PL, -PL / 2],
   892→          [-PL / 2, PL / 2],
   893→          [PL / 2, PL],
   894→        ] as [number, number][]) {
   895→          pushQuad(
   896→            [s0.x + s0.rx * l0, floorY(k), s0.z + s0.rz * l0],
   897→            [s0.x + s0.rx * l1, floorY(k), s0.z + s0.rz * l1],
   898→            [s1.x + s1.rx * l1, floorY(k + 1), s1.z + s1.rz * l1],
   899→            [s1.x + s1.rx * l0, floorY(k + 1), s1.z + s1.rz * l0]
   900→          );
   901→        }
   902→      }
   903→      // near face: crisp cliff under the launch edge (keeps launches snappy)
   904→      const sp = this.samples[wrapIdx(i - 1, n)];
   905→      const sa = this.samples[i];
   906→      for (const [l0, l1] of [
   907→        [-PL, 0],
   908→        [0, PL],
   909→      ] as [number, number][]) {
   910→        pushQuad(
   911→          [sp.x + sp.rx * l0, this.heightAt(wrapIdx(i - 1, n), l0), sp.z + sp.rz * l0],
   912→          [sp.x + sp.rx * l1, this.heightAt(wrapIdx(i - 1, n), l1), sp.z + sp.rz * l1],
   913→          [sa.x + sa.rx * l1, floorY(i), sa.z + sa.rz * l1],
   914→          [sa.x + sa.rx * l0, floorY(i), sa.z + sa.rz * l0]
   915→        );
   916→      }
   917→      // far face: the valley floor has already climbed to road level here —
   918→      // just a short skirt so there is no visible seam at the join
   919→      const sf = this.samples[wrapIdx(jEnd, n)];
   920→      const sl = this.samples[wrapIdx(jEnd - 1, n)];
   921→      for (const [l0, l1] of [
   922→        [-PL, 0],
   923→        [0, PL],
   924→      ] as [number, number][]) {
   925→        pushQuad(
   926→          [sl.x + sl.rx * l0, floorY(jEnd - 1), sl.z + sl.rz * l0],
   927→          [sl.x + sl.rx * l1, floorY(jEnd - 1), sl.z + sl.rz * l1],
   928→          [sf.x + sf.rx * l1, this.heightAt(wrapIdx(jEnd, n), l1), sf.z + sf.rz * l1],
   929→          [sf.x + sf.rx * l0, this.heightAt(wrapIdx(jEnd, n), l0), sf.z + sf.rz * l0]
   930→        );
   931→      }
   932→      i = j;
   933→    }
   934→    const g = new THREE.BufferGeometry();
   935→    g.setAttribute("position", new THREE.Float32BufferAttribute(pos, 3));
   936→    g.setAttribute("color", new THREE.Float32BufferAttribute(col, 3));
   937→    g.setIndex(idx);
   938→    g.computeVertexNormals();
   939→    const m = new THREE.MeshLambertMaterial({
   940→      vertexColors: true,
   941→      flatShading: true,
   942→      side: THREE.DoubleSide,
   943→    });
   944→    const mesh = new THREE.Mesh(g, m);
   945→    mesh.receiveShadow = true;
   946→    this.group.add(mesh);
   947→    this.disposables.push(g, m);
   948→  }

   950→  private buildTunnel() {
   951→    const n = this.N;
   952→    const tunnelIdx: number[] = [];
   953→    for (let i = 0; i < n; i++) if (this.samples[i].tunnel) tunnelIdx.push(i);
   954→    if (tunnelIdx.length < 8) return;
   955→    const pos: number[] = [];
   956→    const col: number[] = [];
   957→    const idx: number[] = [];
   958→    const strata = PALETTE.rockStrata.map((c) => new THREE.Color(c));
   959→    const cDark = new THREE.Color(0x6e3524);
   960→    const SEG = 7;
   961→    const R = 12.2;
   962→    for (let r = 0; r < tunnelIdx.length; r++) {
   963→      const i = tunnelIdx[r];
   964→      const s = this.samples[i];
   965→      for (let k = 0; k <= SEG; k++) {
   966→        const a = Math.PI * (k / SEG);
   967→        const lx = -Math.cos(a) * R;
   968→        const ly = Math.sin(a) * R * 0.92;

  1389→      sign.rotation.y = Math.atan2(-ss.dx, -ss.dz); // face oncoming riders

  1391→      signPost.position.set(ss.x + ss.rx * slat, sy - 1.55, ss.z + ss.rz * slat);
  1392→      signPost.castShadow = true;
  1393→      this.group.add(sign);
  1394→      this.group.add(signPost);

  1396→      beacon.position.set(ss.x + ss.rx * slat, sy + 1.8, ss.z + ss.rz * slat);
  1397→      glow.add(beacon);

  1401→    const chevTex = makeChevronTexture();
  1402→    const chevGeo = new THREE.PlaneGeometry(4.8, 3.4);
  1403→    chevGeo.rotateX(-Math.PI / 2);
  1404→    const chevMat = new THREE.MeshBasicMaterial({ map: chevTex, transparent: true, depthWrite: false });
  1405→    this.disposables.push(chevTex, chevGeo, chevMat);
  1406→    for (const back of [26, 17, 8]) {
  1407→      const ci = wrapIdx(idx - back, this.N);
  1408→      const cs = this.samples[ci];
  1409→      const m = new THREE.Mesh(chevGeo, chevMat);
  1410→      m.position.set(cs.x, this.heightAt(ci, 0) + 0.09, cs.z);
  1411→      m.rotation.y = Math.atan2(-cs.dx, -cs.dz); // chevrons point toward the wall
  1412→      m.renderOrder = 1;
  1413→      this.group.add(m);

  1417→  /**
  1418→   * Breakable wooden crates arranged in STACKS — pyramids, pairs, blocks and
  1419→   * firing lines on the racing line. Smash one crate and the whole stack
  1420→   * chain-breaks into flying pieces. The full candidate set is generated once
  1421→   * (max count); difficulty decides how many stacks go live.
  1422→   */
  1423→  private buildStacks() {
  1424→    const n = this.N;
  1425→    const MAX_STACKS = 14;
  1426→    const rnd = mulberry32(9155);
  1427→
  1428→    // zones to keep clear: gates, ramps, the gap, the tunnel, the start grid, the break wall
  1429→    const blocked = new Uint8Array(n);
  1430→    for (const g of this.gates) {
  1431→      for (let k = -8; k <= 8; k++) blocked[wrapIdx(g.idx + k, n)] = 1;
  1432→    }
  1433→    for (let i = 0; i < n; i++) {
  1434→      if (this.samples[i].gap || this.samples[i].tunnel || this.samples[i].rampP >= 0) {
  1435→        for (let k = -7; k <= 7; k++) blocked[wrapIdx(i + k, n)] = 1;
  1436→      }
  1437→    }
  1438→    for (let k = -60; k <= 60; k++) blocked[wrapIdx(k, n)] = 1; // start/finish + grid
  1439→    for (const wi of this.breakWallIdxs) {
  1440→      for (let k = -16; k <= 16; k++) blocked[wrapIdx(wi + k, n)] = 1;
  1441→    }
  1442→
  1443→    const crateTex = makeCrateTexture();
  1444→    const crateGeo = new THREE.BoxGeometry(1.24, 1.24, 1.24);
  1445→    const crateMat = new THREE.MeshLambertMaterial({ map: crateTex });
  1446→    this.disposables.push(crateGeo, crateMat, crateTex);
  1447→
  1448→    // stack layout patterns: [lateral cell offset, stack level]
  1449→    const PATTERNS: [number, number][][] = [
  1450→      [
  1451→        [-1.32, 0],
  1452→        [0, 0],
  1453→        [1.32, 0],
  1454→        [-0.66, 1],
  1455→        [0.66, 1],
  1456→        [0, 2],
  1457→      ], // pyramid
  1458→      [
  1459→        [-0.65, 0],
  1460→        [0.65, 0],
  1461→        [0, 1],
  1462→      ], // pair + top
  1463→      [
  1464→        [-1.35, 0],
  1465→        [0, 0],
  1466→        [1.35, 0],
  1467→      ], // firing line across the road
  1468→      [
  1469→        [-0.66, 0],
  1470→        [0.66, 0],
  1471→        [-0.66, 1],
  1472→        [0.66, 1],
  1473→      ], // 2×2 block
  1474→    ];
  1475→
  1476→    const used: number[] = [];
  1477→    let cluster = 0;
  1478→    let placed = 0;
  1479→    let guard = 0;
  1480→    while (placed < MAX_STACKS && guard++ < 9000) {
  1481→      const idx = 60 + Math.floor(rnd() * (n - 120));
  1482→      if (blocked[idx]) continue;
  1483→      // stack spacing — each stack reads as its own little event on approach
  1484→      let tooClose = false;
  1485→      for (const u of used) {
  1486→        let d = Math.abs(idx - u);
  1487→        d = Math.min(d, n - d);
  1488→        if (d < 26) {
  1489→          tooClose = true;
  1490→          break;
  1491→        }
  1492→      }
  1493→      if (tooClose) continue;
  1494→      const s = this.samples[idx];
  1495→      // bias stacks toward the middle of the road
  1496→      const u = rnd() * 2 - 1;
  1497→      const lat = Math.sign(u) * Math.pow(Math.abs(u), 1.35) * Math.max(1.2, s.hw - 4.5);
  1498→      // keep clear of registered boulders / pillars
  1499→      let clash = false;
  1500→      for (const ob of this.obstacles) {
  1501→        const dx = ob.x - (s.x + s.rx * lat);
  1502→        const dz = ob.z - (s.z + s.rz * lat);
  1503→        if (dx * dx + dz * dz < 34) {
  1504→          clash = true;
  1505→          break;
  1506→        }
  1507→      }
  1508→      if (clash) continue;
  1509→      const pattern = PATTERNS[Math.floor(rnd() * PATTERNS.length)];
  1510→      const count = Math.min(pattern.length, 3 + Math.floor(rnd() * (pattern.length - 2)));
  1511→      for (let c = 0; c < count; c++) {
  1512→        const [cl, lvl] = pattern[c];
  1513→        const blat = lat + cl;
  1514→        const x = s.x + s.rx * blat;
  1515→        const z = s.z + s.rz * blat;
  1516→        const y = this.heightAt(idx, blat) + 0.62 + lvl * 1.24;
  1517→        const mesh = new THREE.Mesh(crateGeo, crateMat);
  1518→        mesh.position.set(x, y, z);
  1519→        mesh.rotation.y = (rnd() - 0.5) * 0.55;
  1520→        mesh.castShadow = true;
  1521→        this.group.add(mesh);
  1522→        this.boxes.push({
  1523→          x,
  1524→          z,
  1525→          y,
  1526→          idx,
  1527→          lat: blat,
  1528→          cluster,
  1529→          halfH: 0.62,
  1530→          wall: false,
  1531→          enabled: false,
  1532→          active: false,
  1533→          respawnAt: 0,
  1534→          mesh,
  1535→        });
  1536→      }
  1537→      used.push(idx);
  1538→      cluster++;
  1539→      placed++;
  1540→    }
  1541→  }
  1542→
  1543→  dispose() {
  1544→    for (const d of this.disposables) d.dispose();
  1545→  }
  1546→}

  1554→function mulberry32(seed: number) {
  1555→  let a = seed >>> 0;
  1556→  return () => {
  1557→    a |= 0;
  1558→    a = (a + 0x6d2b79f5) | 0;
  1559→    let t = Math.imul(a ^ (a >>> 15), 1 | a);
  1560→    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
  1561→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  1562→  };
  1563→}
