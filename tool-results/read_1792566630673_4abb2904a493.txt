   226→   226→  private buildSky() {
   227→   227→    const geo = new THREE.SphereGeometry(3900, 32, 16);
   228→   228→    const mat = new THREE.ShaderMaterial({
   229→   229→      side: THREE.BackSide,
   230→   230→      depthWrite: false,
   231→   231→      fog: false,
   232→   232→      uniforms: {
   233→   233→        uZen: { value: new THREE.Color(PALETTE.skyZenith) },
   234→   234→        uMid: { value: new THREE.Color(PALETTE.skyMid) },
   235→   235→        uHor: { value: new THREE.Color(PALETTE.skyHorizon) },
   236→   236→        uSun: { value: SUN_DIR.clone() },
   237→   237→      },
   238→   238→      vertexShader: `
   239→   239→        varying vec3 vW;
   240→   240→        void main() {
   241→   241→          vW = (modelMatrix * vec4(position, 1.0)).xyz;
   242→   242→          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
   243→   243→        }
   244→   244→      `,
   245→   245→      fragmentShader: `
   246→   246→        uniform vec3 uZen; uniform vec3 uMid; uniform vec3 uHor; uniform vec3 uSun;
   247→   247→        varying vec3 vW;
   248→   248→        void main() {
   249→   249→          vec3 d = normalize(vW);
   250→   250→          float h = d.y;
   251→   251→          vec3 col = h > 0.16 ? mix(uMid, uZen, smoothstep(0.16, 0.6, h))
   252→   252→                              : mix(uHor, uMid, smoothstep(-0.06, 0.16, h));
   253→   253→          float sd = max(dot(d, uSun), 0.0);
   254→   254→          col += vec3(1.0, 0.88, 0.62) * (pow(sd, 900.0) * 1.2 + pow(sd, 16.0) * 0.3 + pow(sd, 4.0) * 0.1);
   255→   255→          gl_FragColor = vec4(col, 1.0);
   256→   256→        }
   257→   257→      `,
   258→   258→    });
   259→   259→    const sky = new THREE.Mesh(geo, mat);
   260→   260→    sky.frustumCulled = false;
   261→   261→    this.scene.add(sky);
   262→   262→  }
   263→   263→
   264→   264→  private buildHaze() {
   265→   265→    this.hazeRT = new THREE.WebGLRenderTarget(2, 2, {
   266→   266→      type: THREE.HalfFloatType,
   267→   267→    });
   268→   268→    this.hazeMat = new THREE.ShaderMaterial({
   269→   269→      uniforms: {
   270→   270→        tDiffuse: { value: this.hazeRT.texture },
   271→   271→        uTime: { value: 0 },
   272→   272→        uAmount: { value: 1 },
   273→   273→      },
   274→   274→      vertexShader: `
   275→   275→        varying vec2 vUv;
   276→   276→        void main() { vUv = uv; gl_Position = vec4(position.xy, 0.0, 1.0); }
   277→   277→      `,
   278→   278→      fragmentShader: `
   279→   279→        uniform sampler2D tDiffuse; uniform float uTime; uniform float uAmount;
   280→   280→        varying vec2 vUv;
   281→   281→        void main() {
   282→   282→          vec2 uv = vUv;
   283→   283→          float mask = smoothstep(0.85, 0.45, uv.y) * 0.75 + 0.1;
   284→   284→          uv.x += sin(uv.y * 110.0 + uTime * 3.3) * 0.0026 * mask * uAmount;
   285→   285→          uv.y += sin(uv.x * 70.0 + uTime * 2.1) * 0.0018 * mask * uAmount;
   286→   286→          vec3 c = texture2D(tDiffuse, uv).rgb;
   287→   287→          // subtle warm vignette
   288→   288→          vec2 q = vUv - 0.5;
   289→   289→          c *= 1.0 - dot(q, q) * 0.35;
   290→   290→          gl_FragColor = vec4(c, 1.0);
   291→   291→        }
   292→   292→      `,
   293→   293→    });
   294→   294→    this.hazeScene = new THREE.Scene();
   295→   295→    const quadGeo = new THREE.PlaneGeometry(2, 2);
   296→   296→    this.hazeScene.add(new THREE.Mesh(quadGeo, this.hazeMat));
   297→   297→  }

   441→   441→  // main loop

   443→   443→
   444→   444→  private loop = () => {
   445→   445→    if (this.disposed) return;
   446→   446→    this.raf = requestAnimationFrame(this.loop);
   447→   447→    if (this.debugSelftestActive) return; // selftest drives physics deterministically
   448→   448→    const now = performance.now();
   449→   449→    let dt = (now - this.lastT) / 1000;
   450→   450→    this.lastT = now;
   451→   451→    dt = clamp(dt, 0.0005, 1 / 30);
   452→   452→    if (this.paused) dt = 0;
   453→   453→
   454→   454→    this.time += dt;
   455→   455→
   456→   456→    const input = this.input.sample(Math.max(dt, 0.0001));
   457→   457→    this.curInput = input;
   458→   458→    // longitudinal accel estimate for body pitch
   459→   459→    const sp = this.physics.speed;
   460→   460→    if (dt > 0.0001) {
   461→   461→      this.lastAccel = damp(this.lastAccel, (sp - this.prevSpeed) / dt, 6, dt);
   462→   462→    }
   463→   463→    this.prevSpeed = sp;
   464→   464→    this.noteInput(input.steer, input.brake, this.lastAccel);
   465→   465→
   466→   466→    if (this.phase === "countdown") this.updateCountdown(dt);

   588→   588→  private checkGates(prevS: number) {
   589→   589→    const t = this.track;
   590→   590→    while (this.physics.progressS >= this.nextGateAbs) {
   591→   591→      const gate = t.gates[this.nextGateK];

   593→   593→      const latOk = Math.abs(this.physics.lateral) < t.sample(gate.idx).hw + 10;

   595→   595→      if (!latOk || !prevOk) break;
   596→   596→      if (gate.isFinish) {
   597→   597→        this.completeLap();

   604→   604→        this.physics.boostMeter = clamp(this.physics.boostMeter + 12, 0, 100);
   605→   605→        this.audio.checkpoint(delta);
   606→   606→        this.nextGateK++;

   612→   612→  private completeLap() {
   613→   613→    const lapMs = this.raceTime - this.lapStart;

   625→   625→      const bestLap = Math.min(...this.lapTimes);
   626→   626→      const isNewRecord = this.bestTotalMs == null || totalMs < this.bestTotalMs;
   627→   627→      if (isNewRecord) {
   628→   628→        this.bestTotalMs = totalMs;
   629→   629→        this.bestSplits = [...this.splits];
   630→   630→      }
   631→   631→      try {

   640→   640→        );

   644→   644→      } catch {
   645→   645→        /* ignore */
   646→   646→      }
   647→   647→      if (isNewRecord) {

   649→   649→        store.pushPopup("newrecord", "NEW RECORD!");

   665→   665→      });
   666→   666→      return;
   667→   667→    }
   668→   668→    // lap complete, more to go
   669→   669→    store.pushPopup("lap", `LAP ${this.lapsDone}/${this.totalLaps} — ${fmt(lapMs)}`);
   670→   670→    this.audio.checkpoint(-1);
   671→   671→    this.lapStart = this.raceTime;
   672→   672→    this.lineAbs += this.track.length;
   673→   673→    this.nextGateK = 0;

   748→   748→  private emitEffects(dt: number) {
   749→   749→    const p = this.physics;
   750→   750→    const fwd = this.tmpV.set(Math.sin(p.yaw), 0, Math.cos(p.yaw));
   751→   751→    const right = this.tmpV2.set(-Math.cos(p.yaw), 0, Math.sin(p.yaw));
   752→   752→
   753→   753→    const speedH = p.speed;
   754→   754→
   755→   755→    if (p.grounded) {
   756→   756→      // drift smoke + skids
   757→   757→      if (p.drifting && speedH > 8) {
   758→   758→        const q = clamp(Math.abs(p.slip) / 0.5, 0.2, 1);
   759→   759→        for (const side of [-1, 1]) {
   760→   760→          const wx = p.pos.x - fwd.x * 1.25 + right.x * side * 0.85;
   761→   761→          const wz = p.pos.z - fwd.z * 1.25 + right.z * side * 0.85;
   762→   762→          const wy = p.pos.y;
   763→   763→          this.particles.emitSmoke(wx, wy, wz, q);
   764→   764→        }
   765→   765→        this.skidTimer -= dt;
   766→   766→        if (this.skidTimer <= 0) {
   767→   767→          this.skidTimer = 0.03;
   768→   768→          const vx = p.vel.x / Math.max(1, speedH);
   769→   769→          const vz = p.vel.z / Math.max(1, speedH);
   770→   770→          for (const side of [-1, 1]) {
   771→   771→            const wx = p.pos.x - fwd.x * 1.25 + right.x * side * 0.85;
   772→   772→            const wz = p.pos.z - fwd.z * 1.25 + right.z * side * 0.85;
   773→   773→            this.particles.emitSkid(wx, p.pos.y, wz, vx, vz, clamp(Math.abs(p.slip) * 2, 0.3, 1), this.time);
   774→   774→          }
   775→   775→        }
   776→   776→      }
   777→   777→      // off-road dust
   778→   778→      if (p.offRoad && speedH > 9) {
   779→   779→        this.particles.emitDust(
   780→   780→          p.pos.x - fwd.x * 1.4, p.pos.y, p.pos.z - fwd.z * 1.4,
   781→   781→          clamp(speedH / 30, 0.3, 1.2), p.vel.x, p.vel.z
   782→   782→        );
   783→   783→      }
   784→   784→      // wall-ride sparks
   785→   785→      if (p.wallRiding) {
   786→   786→        this.sparkTimer -= dt;
   787→   787→        if (this.sparkTimer <= 0) {
   788→   788→          this.sparkTimer = 0.06;
   789→   789→          const side = p.lateral > 0 ? 1 : -1;
   790→   790→          this.particles.emitSparks(
   791→   791→            p.pos.x + right.x * side * 1.1, p.pos.y + 0.2, p.pos.z + right.z * side * 1.1,
   792→   792→            -right.x * side, -right.z * side, 3
   793→   793→          );
   794→   794→        }
   795→   795→      }
   796→   796→    }
   797→   797→    // boost trail
   798→   798→    if (p.boosting) {
   799→   799→      this.boostTrailTimer -= dt;
   800→   800→      if (this.boostTrailTimer <= 0) {
   801→   801→        this.boostTrailTimer = 0.02;
   802→   802→        this.particles.emitBoost(p.pos.x - fwd.x * 2.0, p.pos.y + 0.45, p.pos.z - fwd.z * 2.0, -fwd.x, -fwd.z);
   803→   803→      }
   804→   804→    }
   805→   805→  }

   846→   846→  private playGhost() {
   847→   847→    if (!this.ghost.length) {
   848→   848→      this.ghostCar.group.visible = false;
   849→   849→      return;
   850→   850→    }
   851→   851→    this.ghostCar.group.visible = true;
   852→   852→    const tMs = this.raceTime;
   853→   853→    const frames = this.ghost;
   854→   854→    // binary search
   855→   855→    let lo = 0;
   856→   856→    let hi = frames.length - 1;
   857→   857→    while (lo < hi) {
   858→   858→      const mid = (lo + hi) >> 1;
   859→   859→      if (frames[mid].t < tMs) lo = mid + 1;
   860→   860→      else hi = mid;
   861→   861→    }
   862→   862→    const i = Math.max(1, lo);
   863→   863→    const a = frames[i - 1];
   864→   864→    const b = frames[Math.min(frames.length - 1, i)];
   865→   865→    const span = Math.max(1e-3, b.t - a.t);
   866→   866→    const f = clamp((tMs - a.t) / span, 0, 1);
   867→   867→    const g = this.ghostCar.group;
   868→   868→    g.position.set(a.x + (b.x - a.x) * f, a.y + (b.y - a.y) * f, a.z + (b.z - a.z) * f);
   869→   869→    let dy = b.yaw - a.yaw;
   870→   870→    while (dy > Math.PI) dy -= Math.PI * 2;
   871→   871→    while (dy < -Math.PI) dy += Math.PI * 2;
   872→   872→    g.rotation.set(0, a.yaw + dy * f, 0);
   873→   873→  }

   879→   879→  private updateCamera(dt: number) {
   880→   880→    if (this.phase === "title") return; // handled in updateTitle
   881→   881→    const p = this.physics;
   882→   882→    const fwd = this.tmpV.set(Math.sin(p.yaw), 0, Math.cos(p.yaw));
   883→   883→
   884→   884→    if (this.phase === "countdown") {
   885→   885→      // cinematic sweep down to the chase position
   886→   886→      const t = clamp(1 - this.countdownT / 3, 0, 1);
   887→   887→      const e = t * t * (3 - 2 * t);
   888→   888→      const ang = 2.4 - e * 2.4;
   889→   889→      const dist = 30 - 24 * e;
   890→   890→      this.camPos.set(
   891→   891→        p.pos.x + Math.sin(ang) * dist + fwd.x * -4,
   892→   892→        p.pos.y + 14 - 11.5 * e,
   893→   893→        p.pos.z + Math.cos(ang) * dist + fwd.z * -4
   894→   894→      );
   895→   895→      this.camLook.set(p.pos.x + fwd.x * 6, p.pos.y + 1.2, p.pos.z + fwd.z * 6);
   896→   896→      this.camera.position.copy(this.camPos);
   897→   897→      this.camera.up.set(0, 1, 0);
   898→   898→      this.camera.lookAt(this.camLook);
   899→   899→      this.camera.fov = damp(this.camera.fov, 66, 3, dt);
   900→   900→      this.camera.updateProjectionMatrix();
   901→   901→      return;
   902→   902→    }
   903→   903→
   904→   904→    if (this.phase === "finished") {
   905→   905→      const a = this.finishOrbitT * 0.5;
   906→   906→      this.camPos.set(p.pos.x + Math.sin(a) * 9, p.pos.y + 3.2, p.pos.z + Math.cos(a) * 9);
   907→   907→      this.camLook.copy(p.pos).add(new THREE.Vector3(0, 1, 0));
   908→   908→      this.camera.position.copy(this.camPos);
   909→   909→      this.camera.up.set(0, 1, 0);
   910→   910→      this.camera.lookAt(this.camLook);
   911→   911→      return;
   912→   912→    }
   913→   913→
   914→   914→    // chase camera
   915→   915→    const mode = this.settings.cameraMode;
   916→   916→    const dist = mode === 1 ? 9.6 : 6.4;
   917→   917→    const height = mode === 1 ? 3.8 : 2.55;
   918→   918→    if (mode === 2) {
   919→   919→      // hood cam
   920→   920→      this.camPos.set(p.pos.x + fwd.x * 0.5, p.pos.y + 1.35, p.pos.z + fwd.z * 0.5);
   921→   921→      this.camLook.set(p.pos.x + fwd.x * 12, p.pos.y + 1.6, p.pos.z + fwd.z * 12);
   922→   922→      this.camera.position.copy(this.camPos);
   923→   923→      this.camera.up.set(0, 1, 0);
   924→   924→      this.camera.lookAt(this.camLook);
   925→   925→    } else {
   926→   926→      const ffX = clamp(p.vel.x * 0.16, -7, 7);
   927→   927→      const ffZ = clamp(p.vel.z * 0.16, -7, 7);
   928→   928→      const tx = p.pos.x - fwd.x * dist + ffX;
   929→   929→      const ty = p.pos.y + height;
   930→   930→      const tz = p.pos.z - fwd.z * dist + ffZ;
   931→   931→      const rate = 6.5;
   932→   932→      this.camPos.x = damp(this.camPos.x, tx, rate, dt);
   933→   933→      this.camPos.y = damp(this.camPos.y, ty, rate, dt);
   934→   934→      this.camPos.z = damp(this.camPos.z, tz, rate, dt);
   935→   935→      this.camLook.set(p.pos.x + fwd.x * 7, p.pos.y + 1.3, p.pos.z + fwd.z * 7);
   936→   936→      this.camera.position.copy(this.camPos);
   937→   937→      this.camera.up.set(0, 1, 0);
   938→   938→      this.camera.lookAt(this.camLook);
   939→   939→      // banking roll
   940→   940→      const targetRoll = clamp(-p.slip * 0.3 - this.lastSteer * 0.05, -0.24, 0.24);
   941→   941→      this.camRoll = damp(this.camRoll, targetRoll, 5, dt);
   942→   942→      this.camera.rotateZ(this.camRoll);
   943→   943→    }
   944→   944→
   945→   945→    // wall-ride lean
   946→   946→    this.wallRideLean = damp(this.wallRideLean, p.wallRiding ? clamp(p.lateral / 24, 0, 1) * 0.5 : 0, 4, dt);
   947→   947→    if (this.wallRideLean > 0.01) this.camera.rotateZ(this.wallRideLean);
   948→   948→
   949→   949→    // FOV kick
   950→   950→    const speedFrac = clamp(p.speed / 55, 0, 1.1);
   951→   951→    const targetFov = 64 + speedFrac * 10 + (p.boosting ? 9 : 0);
   952→   952→    this.fov = damp(this.fov, targetFov, 4, dt);
   953→   953→    this.camera.fov = this.fov;
   954→   954→    this.camera.updateProjectionMatrix();
   955→   955→
   956→   956→    // restrained screen shake
   957→   957→    this.trauma = Math.max(0, this.trauma - dt * 1.6);
   958→   958→    const sh = this.trauma * this.trauma;
   959→   959→    if (sh > 0.001) {
   960→   960→      this.camera.position.x += (Math.random() - 0.5) * sh * 0.55;
   961→   961→      this.camera.position.y += (Math.random() - 0.5) * sh * 0.4;
   962→   962→      this.camera.position.z += (Math.random() - 0.5) * sh * 0.55;
   963→   963→      this.camera.rotateZ((Math.random() - 0.5) * sh * 0.03);
   964→   964→    }
   965→   965→  }
