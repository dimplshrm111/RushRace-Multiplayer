     2→ * CANYON RUSH — real-time multiplayer racing service.
     3→ *
     4→ * Socket.IO server (port 3003) providing:
     5→ *  - room-based lobbies with 4-char codes, multiple independent rooms
     6→ *  - hard 10-player limit enforced server-side
     7→ *  - host/admin authority + safe host transfer on disconnect
     8→ *  - custom names, live renaming, ready system
     9→ *  - room-isolated real-time chat (validation + anti-spam)
    10→ *  - server-authorized synchronized race countdown (3-2-1-GO)
    11→ *  - 10Hz aggregated car-state relay with pre-GO movement lock
    12→ *  - server-validated laps / finishes / final results + rematch
    13→ *  - reconnection tokens, disconnect grace, abandoned-room cleanup
    14→ *
    15→ * All game physics run on each client; the server is authoritative for
    16→ * everything social + race lifecycle + result ordering, and validates
    17→ * lap/finish plausibility (distance + minimum-time gates) so results
    18→ * cannot be manipulated with plain client-side requests.

   131→function sanitizeText(raw: unknown): string {
   132→  if (typeof raw !== "string") return "";
   133→  // strip control chars, collapse whitespace
   134→  return raw
   135→    .replace(/[\u0000-\u001f\u007f]/g, "")
   136→    .replace(/\s+/g, " ")
   137→    .trim();
   138→}

   140→function maskProfanity(text: string): string {
   141→  let out = text;
   142→  for (const w of BAD_WORDS) {
   143→    out = out.replace(new RegExp(w, "gi"), (m) => m[0] + "*".repeat(Math.max(1, m.length - 1)));
   144→  }
   145→  return out;
   146→}
   147→
   148→function hasProfanity(text: string): boolean {
   149→  const lower = text.toLowerCase();
   150→  return BAD_WORDS.some((w) => lower.includes(w));
   151→}

   153→/** Valid display name or null. */
   154→function validName(raw: unknown): string | null {
   155→  const name = sanitizeText(raw);
   156→  if (name.length < 1 || name.length > NAME_MAX) return null;
   157→  if (hasProfanity(name)) return null;
   158→  return name;
   159→}

   161→function genCode(): string {
   162→  let code = "";
   163→  for (let i = 0; i < 4; i++) {
   164→    code += CODE_CHARS[Math.floor(Math.random() * CODE_CHARS.length)];
   165→  }
   166→  return code;
   167→}

   169→function genRoomCode(): string {
   170→  for (let tries = 0; tries < 50; tries++) {
   171→    const code = genCode();
   172→    if (!rooms.has(code)) return code;
   173→  }
   174→  // astronomically unlikely fallback
   175→  return genCode() + Date.now().toString(36).slice(-2).toUpperCase();
   176→}

   178→function num(v: unknown, fallback = 0): number {
   179→  const n = typeof v === "number" ? v : Number(v);
   180→  return Number.isFinite(n) ? n : fallback;
   181→}

   183→function sanitizeState(raw: unknown): RaceState | null {
   184→  if (typeof raw !== "object" || raw === null) return null;
   185→  const r = raw as Record<string, unknown>;
   186→  let yaw = num(r.yaw);
   187→  while (yaw > Math.PI) yaw -= Math.PI * 2;
   188→  while (yaw < -Math.PI) yaw += Math.PI * 2;
   189→  return {
   190→    x: clampAbs(num(r.x), 6000),
   191→    y: clampRange(num(r.y), -60, 400),
   192→    z: clampAbs(num(r.z), 6000),
   193→    yaw,
   194→    steer: clampRange(num(r.steer), -1, 1),
   195→    spd: clampRange(num(r.spd), 0, 250),
   196→    boost: !!r.boost,
   197→    drift: !!r.drift,
   198→    ground: !!r.ground,
   199→    lap: Math.min(50, Math.max(0, Math.round(num(r.lap)))),
   200→    prog: clampAbs(num(r.prog), 1e7),
   201→  };
   202→}
   203→
   204→function clampAbs(v: number, lim: number): number {
   205→  return Math.max(-lim, Math.min(lim, v));
   206→}
   207→
   208→function clampRange(v: number, lo: number, hi: number): number {
   209→  return Math.max(lo, Math.min(hi, v));
   210→}

   254→function pickColor(r: Room): string {

   269→/** Remove a player entirely (grace expired or explicit leave). */
   270→function removePlayer(r: Room, p: Player, reason: "left" | "timeout") {
   271→  if (!r.players.has(p.id)) return;
   272→  if (p.removeTimer) clearTimeout(p.removeTimer);
   273→  r.players.delete(p.id);
   274→  r.order = r.order.filter((pid) => pid !== p.id);
   275→  io.to(r.code).emit("mp:chat", {
   276→    msg: sysMsg(reason === "left" ? `${p.name} left the room` : `${p.name} timed out`),
   277→  });
   278→
   279→  if (r.players.size === 0) {
   280→    deleteRoom(r);
   281→    return;
   282→  }
   283→  // host transfer
   284→  if (r.hostId === p.id) {
   285→    const next = r.order
   286→      .map((pid) => r.players.get(pid)!)
   287→      .find((pl) => pl.connected);
   288→    if (next) {
   289→      r.hostId = next.id;
   290→      systemChat(r, `${next.name} is now the host`);
   291→    }
   292→  }
   293→  broadcastRoom(r);
   294→}

   296→function deleteRoom(r: Room) {
   297→  r.goTimer = clearTimer(r.goTimer);
   298→  r.raceTimer = clearTimer(r.raceTimer);
   299→  r.compileTimer = clearTimer(r.compileTimer);
   300→  r.lobbyTimer = clearTimer(r.lobbyTimer);
   301→  for (const p of r.players.values()) {
   302→    if (p.removeTimer) clearTimeout(p.removeTimer);
   303→  }
   304→  r.relayOn = false;
   305→  rooms.delete(r.code);
   306→  console.log(`[room ${r.code}] deleted`);
   307→}

   309→/** Assign fresh grid slots: connected racers in `order` sequence. */
   310→function assignGrid(r: Room) {
   311→  let g = 0;
   312→  for (const pid of r.order) {
   313→    const p = r.players.get(pid);
   314→    if (!p) continue;
   315→    p.grid = g++;
   316→  }
   317→}

   319→/** Begin the server-authorized countdown. */
   320→function beginCountdown(r: Room) {
   321→  r.phase = "countdown";
   322→  r.startAt = Date.now() + COUNTDOWN_MS;
   323→  r.finishCounter = 0;
   324→  for (const p of r.players.values()) {
   325→    // everyone present at start time races (even currently-disconnected
   326→    // players keep their slot so they can rejoin in progress)
   327→    p.racing = true;
   328→    p.lap = 0;
   329→    p.finishPos = null;
   330→    p.finishMs = null;
   331→    p.bestLapMs = null;
   332→    p.topSpeedKmh = null;
   333→    p.driftScore = null;
   334→    p.lastProgress = 0;
   335→    p.lapProgBase = null;
   336→    p.anchor = null;
   337→    p.lastState = null;
   338→    p.ready = false;
   339→  }
   340→  assignGrid(r);
   341→  r.relayOn = true;
   342→  broadcastRoom(r);
   343→  io.to(r.code).emit("mp:countdown", { startAt: r.startAt, laps: r.laps });
   344→  console.log(`[room ${r.code}] countdown → GO at ${r.startAt}`);
   345→
   346→  r.goTimer = clearTimer(r.goTimer);
   347→  r.goTimer = setTimeout(() => {
   348→    r.phase = "racing";
   349→    r.raceStartedAt = r.startAt;
   350→    io.to(r.code).emit("mp:go", { startAt: r.startAt });
   351→    broadcastRoom(r);
   352→    // hard race timeout — force results so a room can never hang
   353→    r.raceTimer = clearTimer(r.raceTimer);
   354→    r.raceTimer = setTimeout(
   355→      () => {
   356→        if (r.phase === "racing") {
   357→          console.log(`[room ${r.code}] race timeout → results`);
   358→          compileResults(r, true);
   359→        }
   360→      },
   361→      r.laps * RACE_TIMEOUT_PER_LAP_MS + 60000
   362→    );
   363→  }, COUNTDOWN_MS);
   364→}

   366→/** Build + broadcast final results and move to the results phase. */
   367→function compileResults(r: Room, fromTimeout = false) {
   368→  if (r.phase !== "racing") return;
   369→  r.compileTimer = clearTimer(r.compileTimer);
   370→  r.raceTimer = clearTimer(r.raceTimer);
   371→  const racers = r.order
   372→    .map((pid) => r.players.get(pid)!)
   373→    .filter((p) => p && p.racing);
   374→  const finished = racers
   375→    .filter((p) => p.finishPos != null)
   376→    .sort((a, b) => (a.finishPos ?? 0) - (b.finishPos ?? 0));
   377→  const dnf = racers
   378→    .filter((p) => p.finishPos == null)
   379→    .sort((a, b) => b.lastProgress - a.lastProgress);
   380→  const results = [...finished, ...dnf].map((p) => ({
   381→    pid: p.id,
   382→    name: p.name,
   383→    color: p.color,
   384→    pos: p.finishPos,
   385→    totalMs: p.finishMs,
   386→    bestLapMs: p.bestLapMs,
   387→    topSpeedKmh: p.topSpeedKmh,
   388→    driftScore: p.driftScore,
   389→    finished: p.finishPos != null,
   390→  }));
   391→  r.phase = "results";
   392→  io.to(r.code).emit("mp:results", { results });
   393→  broadcastRoom(r);
   394→  console.log(
   395→    `[room ${r.code}] results compiled (${finished.length} finished, ${dnf.length} DNF${fromTimeout ? ", timeout" : ""})`
   396→  );
   397→  // auto return to lobby so the room never gets stuck in results
   398→  r.lobbyTimer = clearTimer(r.lobbyTimer);
   399→  r.lobbyTimer = setTimeout(() => returnToLobby(r), RESULTS_TO_LOBBY_MS);
   400→}
   401→
   402→function returnToLobby(r: Room) {
   403→  if (r.phase !== "results") return;
   404→  r.lobbyTimer = clearTimer(r.lobbyTimer);
   405→  r.phase = "lobby";
   406→  for (const p of r.players.values()) {
   407→    p.ready = false;
   408→    p.racing = false;
   409→    p.lap = 0;
   410→    p.finishPos = null;
   411→    p.finishMs = null;
   412→  }
   413→  r.relayOn = false;
   414→  io.to(r.code).emit("mp:phase", { phase: "lobby" });
   415→  broadcastRoom(r);
   416→}

   418→/** Schedule result compilation once every connected racer has finished. */
   419→function maybeScheduleCompile(r: Room) {
   420→  if (r.phase !== "racing") return;
   421→  const racers = [...r.players.values()].filter((p) => p.racing);
   422→  const connectedRacers = racers.filter((p) => p.connected);
   423→  if (connectedRacers.length === 0) {
   424→    // everyone dropped — compile after the reconnect grace window
   425→    if (!r.compileTimer) {
   426→      r.compileTimer = setTimeout(() => compileResults(r), RECONNECT_GRACE_MS);
   427→    }
   428→    return;
   429→  }
   430→  const allDone = connectedRacers.every((p) => p.finishPos != null);
   431→  if (allDone) {
   432→    r.compileTimer = clearTimer(r.compileTimer);
   433→    r.compileTimer = setTimeout(() => compileResults(r), RESULTS_COMPILE_DELAY_MS);
   434→  }
   435→}

   450→interface SocketData {
   451→  roomCode?: string;
   452→  pid?: string;

   455→function ctxOf(socket: Socket): { r: Room; p: Player } | null {
   456→  const code = (socket.data as SocketData).roomCode;
   457→  const pid = (socket.data as SocketData).pid;
   458→  if (!code || !pid) return null;
   459→  const r = rooms.get(code);
   460→  if (!r) return null;
   461→  const p = r.players.get(pid);
   462→  if (!p) return null;
   463→  return { r, p };
   464→}

   466→/** Detach a socket from its current room (explicit leave / room switch). */
   467→function detachSocket(socket: Socket, reason: "left" | "timeout") {
   468→  const ctx = ctxOf(socket);
   469→  if (!ctx) return;
   470→  const { r, p } = ctx;
   471→  (socket.data as SocketData) = {};
   472→  socket.leave(r.code);
   473→  removePlayer(r, p, reason);
   474→}

   476→io.on("connection", (socket) => {
   477→  console.log(`socket connected: ${socket.id}`);
   478→
   479→  socket.on("mp:ping", (payload: unknown) => {
   480→    const t0 = (payload as { t0?: number })?.t0;
   481→    socket.emit("mp:pong", { t0: typeof t0 === "number" ? t0 : 0, serverNow: Date.now() });
   482→  });
   483→
   484→  // ----- create ------------------------------------------------------------
   485→  socket.on("mp:create", (payload: unknown, ack?: (res: unknown) => void) => {
   486→    try {
   487→      const body = (payload ?? {}) as { name?: unknown };
   488→      const name = validName(body.name);
   489→      if (!name) {
   490→        ack?.({ ok: false, error: "Enter a name (1-14 characters)" });
   491→        return;
   492→      }
   493→      detachSocket(socket, "left");
   494→      const code = genRoomCode();
   495→      const room: Room = {
   496→        code,
   497→        hostId: "p1",
   498→        phase: "lobby",
   499→        players: new Map(),
   500→        order: [],
   501→        nextPid: 1,
   502→        laps: 3,
   503→        trackLength: 2600,
   504→        startAt: 0,
   505→        raceStartedAt: 0,
   506→        finishCounter: 0,
   507→        goTimer: null,
   508→        raceTimer: null,
   509→        compileTimer: null,
   510→        lobbyTimer: null,
   511→        relayOn: false,
   512→        createdAt: Date.now(),
   513→      };
   514→      const pid = "p" + room.nextPid++;
   515→      const player: Player = {
   516→        id: pid,
   517→        token: randomUUID(),
   518→        name,
   519→        color: pickColor(room),
   520→        ready: false,
   521→        connected: true,
   522→        socketId: socket.id,
   523→        grid: 0,
   524→        racing: false,
   525→        lap: 0,
   526→        finishPos: null,
   527→        finishMs: null,
   528→        bestLapMs: null,
   529→        topSpeedKmh: null,
   530→        driftScore: null,
   531→        lastProgress: 0,
   532→        lapProgBase: null,
   533→        anchor: null,
   534→        lastState: null,
   535→        lastStateAt: 0,
   536→        chatTimes: [],
   537→        disconnectAt: null,
   538→        removeTimer: null,
   539→      };
   540→      room.players.set(pid, player);
   541→      room.order.push(pid);
   542→      room.hostId = pid;
   543→      rooms.set(code, room);
   544→      (socket.data as SocketData) = { roomCode: code, pid };
   545→      socket.join(code);
   546→      console.log(`[room ${code}] created by ${name}`);
   547→      ack?.({ ok: true, code, token: player.token, pid });
   548→      socket.emit("mp:room", { room: roomState(room) });
   549→      systemChat(room, `${name} created the room`);
   550→    } catch (err) {
   551→      console.error("mp:create error", err);
   552→      ack?.({ ok: false, error: "Server error" });
   553→    }
   554→  });
   555→
   556→  // ----- join --------------------------------------------------------------
   557→  socket.on("mp:join", (payload: unknown, ack?: (res: unknown) => void) => {
   558→    try {
   559→      const body = (payload ?? {}) as { code?: unknown; name?: unknown; token?: unknown };
   560→      const code = sanitizeText(body.code).toUpperCase();
   561→      if (code.length !== 4) {
   562→        ack?.({ ok: false, error: "Room code must be 4 characters" });
   563→        return;
   564→      }
   565→      const room = rooms.get(code);
   566→      if (!room) {
   567→        ack?.({ ok: false, error: "Room not found — check the code" });
   568→        return;
   569→      }
   570→      const token = typeof body.token === "string" ? body.token : "";
   571→
   572→      // --- reconnection path (token match, currently disconnected) ---
   573→      if (token) {
   574→        for (const p of room.players.values()) {
   575→          if (p.token === token) {
   576→            if (p.connected) {
   577→              ack?.({ ok: false, error: "This player is already active in the room" });
   578→              return;
   579→            }
   580→            detachSocket(socket, "left");
   581→            p.connected = true;
   582→            p.socketId = socket.id;
   583→            p.disconnectAt = null;
   584→            if (p.removeTimer) {
   585→              clearTimeout(p.removeTimer);
   586→              p.removeTimer = null;
   587→            }
   588→            (socket.data as SocketData) = { roomCode: code, pid: p.id };
   589→            socket.join(code);
   590→            console.log(`[room ${code}] ${p.name} reconnected`);
   591→            ack?.({ ok: true, code, token: p.token, pid: p.id });
   592→            socket.emit("mp:room", { room: roomState(room) });
   593→            systemChat(room, `${p.name} reconnected`);
   594→            broadcastRoom(room);
   595→            return;
   596→          }
   597→        }
   598→      }
   599→
   600→      // --- fresh join ---
   601→      if (room.players.size >= MAX_PLAYERS) {
   602→        ack?.({ ok: false, error: `Room is full (${room.players.size}/${MAX_PLAYERS})` });
   603→        return;
   604→      }
   605→      const name = validName(body.name);
   606→      if (!name) {
   607→        ack?.({ ok: false, error: "Enter a name (1-14 characters)" });
   608→        return;
   609→      }
   610→      detachSocket(socket, "left");
   611→      const pid = "p" + room.nextPid++;
   612→      const color = pickColor(room);
   613→      const player: Player = {
   614→        id: pid,
   615→        token: randomUUID(),
   616→        name,
   617→        color,
   618→        ready: false,
   619→        connected: true,
   620→        socketId: socket.id,
   621→        grid: room.players.size,
   622→        racing: false,
   623→        lap: 0,
   624→        finishPos: null,
   625→        finishMs: null,
   626→        bestLapMs: null,
   627→        topSpeedKmh: null,
   628→        driftScore: null,
   629→        lastProgress: 0,
   630→        lapProgBase: null,
   631→        anchor: null,
   632→        lastState: null,
   633→        lastStateAt: 0,
   634→        chatTimes: [],
   635→        disconnectAt: null,
   636→        removeTimer: null,
   637→      };
   638→      room.players.set(pid, player);
   639→      room.order.push(pid);
   640→      (socket.data as SocketData) = { roomCode: code, pid };
   641→      socket.join(code);
   642→      console.log(`[room ${code}] ${name} joined (${room.players.size}/${MAX_PLAYERS})`);
   643→      ack?.({ ok: true, code, token: player.token, pid });
   644→      socket.emit("mp:room", { room: roomState(room) });
   645→      systemChat(room, `${name} joined the room`);
   646→      broadcastRoom(room);
   647→    } catch (err) {
   648→      console.error("mp:join error", err);
   649→      ack?.({ ok: false, error: "Server error" });
   650→    }
   651→  });
   652→

   654→  socket.on("mp:leave", (_payload: unknown, ack?: (res: unknown) => void) => {

   656→      detachSocket(socket, "left");
   657→      ack?.({ ok: true });

   664→  // ----- rename ------------------------------------------------------------

   666→    try {
   667→      const ctx = ctxOf(socket);
   668→      if (!ctx) {
   669→        ack?.({ ok: false, error: "Not in a room" });
   670→        return;
   671→      }
   672→      const { r, p } = ctx;
   673→      const body = (payload ?? {}) as { name?: unknown };
   674→      const name = validName(body.name);

   676→        ack?.({ ok: false, error: "Name must be 1-14 characters (no bad words)" });
   677→        return;
   678→      }
   679→      if (name === p.name) {
   680→        ack?.({ ok: true });
   681→        return;
   682→      }

   684→      p.name = name;

   696→    try {
   697→      const ctx = ctxOf(socket);
   698→      if (!ctx) {
   699→        ack?.({ ok: false, error: "Not in a room" });
   700→        return;
   701→      }
   702→      const { r, p } = ctx;
   703→      if (r.phase !== "lobby") {
   704→        ack?.({ ok: false, error: "Can't change ready during a race" });

   708→      ack?.({ ok: true });
   709→      broadcastRoom(r);

   711→      console.error("mp:ready error", err);
   712→      ack?.({ ok: false, error: "Server error" });
   713→    }

   718→    try {
   719→      const ctx = ctxOf(socket);
   720→      if (!ctx) {
   721→        ack?.({ ok: false, error: "Not in a room" });
   722→        return;
   723→      }
   724→      const { r, p } = ctx;
   725→      if (r.hostId !== p.id) {
   726→        ack?.({ ok: false, error: "Only the host can change settings" });

   737→      }
   738→      r.laps = laps;
   739→      ack?.({ ok: true });
   740→      broadcastRoom(r);

   743→      ack?.({ ok: false, error: "Server error" });
   744→    }
   745→  });

   747→  // ----- chat --------------------------------------------------------------
   748→  socket.on("mp:chat", (payload: unknown, ack?: (res: unknown) => void) => {
   749→    try {
   750→      const ctx = ctxOf(socket);
   751→      if (!ctx) {
   752→        ack?.({ ok: false, error: "Not in a room" });
   753→        return;
   754→      }
   755→      const { r, p } = ctx;
   756→      const raw = (payload as { text?: unknown })?.text;
   757→      if (typeof raw !== "string") {
   758→        ack?.({ ok: false, error: "Invalid message" });
   759→        return;
   760→      }
   761→      const text = sanitizeText(raw).slice(0, CHAT_MAX_LEN);
   762→      if (!text) {
   763→        ack?.({ ok: false, error: "Message is empty" });
   764→        return;
   765→      }
   766→      // anti-spam: max CHAT_MAX_IN_WINDOW messages per window
   767→      const now = Date.now();
   768→      p.chatTimes = p.chatTimes.filter((t) => now - t < CHAT_WINDOW_MS);
   769→      if (p.chatTimes.length >= CHAT_MAX_IN_WINDOW) {
   770→        ack?.({ ok: false, error: "Slow down — too many messages" });
   771→        return;
   772→      }
   773→      p.chatTimes.push(now);
   774→      ack?.({ ok: true });
   775→      io.to(r.code).emit("mp:chat", { msg: chatMsg(p, maskProfanity(text)) });
   776→    } catch (err) {
   777→      console.error("mp:chat error", err);
   778→      ack?.({ ok: false, error: "Server error" });
   779→    }

   782→  // ----- host: start race ----------------------------------------------------
   783→  socket.on("mp:start", (payload: unknown, ack?: (res: unknown) => void) => {
   784→    try {
   785→      const ctx = ctxOf(socket);
   786→      if (!ctx) {
   787→        ack?.({ ok: false, error: "Not in a room" });
   788→        return;
   789→      }
   790→      const { r, p } = ctx;
   791→      if (r.hostId !== p.id) {
   792→        ack?.({ ok: false, error: "Only the host can start the race" });
   793→        return;
   794→      }
   795→      if (r.phase !== "lobby") {
   796→        ack?.({ ok: false, error: "Race already running" });
   797→        return;
   798→      }
   799→      const connected = [...r.players.values()].filter((pl) => pl.connected);
   800→      if (connected.length < 1) {
   801→        ack?.({ ok: false, error: "Not enough players" });
   802→        return;
   803→      }
   804→      const notReady = connected.filter((pl) => !pl.ready);
   805→      if (notReady.length > 0) {
   806→        ack?.({ ok: false, error: `Waiting for ${notReady.length} player(s) to ready up` });
   807→        return;
   808→      }
   809→      const body = (payload ?? {}) as { laps?: unknown; trackLength?: unknown };
   810→      const laps = Math.round(num(body.laps, r.laps));
   811→      if (laps < 1 || laps > 5) {
   812→        ack?.({ ok: false, error: "Laps must be 1-5" });
   813→        return;
   814→      }
   815→      const trackLength = num(body.trackLength, 2600);
   816→      r.laps = laps;
   817→      r.trackLength = clampRange(trackLength, 500, 20000);
   818→      ack?.({ ok: true });
   819→      beginCountdown(r);
   820→    } catch (err) {
   821→      console.error("mp:start error", err);
   822→      ack?.({ ok: false, error: "Server error" });
   823→    }
   824→  });

   826→  // ----- host: rematch -------------------------------------------------------
   827→  socket.on("mp:rematch", (_payload: unknown, ack?: (res: unknown) => void) => {
   828→    try {
   829→      const ctx = ctxOf(socket);
   830→      if (!ctx) {
   831→        ack?.({ ok: false, error: "Not in a room" });
   832→        return;
   833→      }
   834→      const { r, p } = ctx;
   835→      if (r.hostId !== p.id) {
   836→        ack?.({ ok: false, error: "Only the host can start a rematch" });
   837→        return;
   838→      }
   839→      if (r.phase !== "results") {
   840→        ack?.({ ok: false, error: "Rematch is available on the results screen" });
   841→        return;
   842→      }
   843→      // new grid: finish order first (winner on pole), DNFs by progress
   844→      r.order = r.order
   845→        .map((pid) => r.players.get(pid)!)
   846→        .filter(Boolean)
   847→        .sort((a, b) => {
   848→          const fa = a.finishPos ?? Infinity;
   849→          const fb = b.finishPos ?? Infinity;
   850→          if (fa !== fb) return fa - fb;
   851→          if (fa === Infinity) return b.lastProgress - a.lastProgress;
   852→          return 0;
   853→        })
   854→        .map((pl) => pl.id);
   855→      ack?.({ ok: true });
   856→      beginCountdown(r);
   857→    } catch (err) {
   858→      console.error("mp:rematch error", err);
   859→      ack?.({ ok: false, error: "Server error" });
   860→    }
   861→  });
   862→
   863→  // ----- race state (throttled, sanitized, relayed) --------------------------
   864→  socket.on("mp:state", (payload: unknown) => {
   865→    try {
   866→      const ctx = ctxOf(socket);
   867→      if (!ctx) return;
   868→      const { r, p } = ctx;
   869→      if (r.phase !== "countdown" && r.phase !== "racing" && r.phase !== "results") return;
   870→      if (!p.racing || !p.connected) return;
   871→      const now = Date.now();
   872→      if (now - p.lastStateAt < STATE_MIN_INTERVAL_MS) return; // drop, don't queue
   873→      const st = sanitizeState(payload);
   874→      if (!st) return;
   875→      p.lastStateAt = now;
   876→      p.lastState = st;
   877→      p.lastProgress = st.prog;
   878→      if (r.phase === "racing" && p.lapProgBase == null) p.lapProgBase = st.prog;
   879→      // pre-GO lock: the first state during countdown becomes the grid anchor
   880→      // and every relayed state until GO uses the anchor (no early movement)
   881→      if (r.phase === "countdown") {
   882→        if (!p.anchor) p.anchor = st;
   883→        p.lastState = { ...p.anchor, lap: 0, prog: p.anchor.prog };
   884→      }
   885→    } catch (err) {
   886→      console.error("mp:state error", err);
   887→    }
   888→  });
   889→
   890→  // ----- lap report (validated) ----------------------------------------------
   891→  socket.on("mp:lap", (payload: unknown) => {
   892→    try {
   893→      const ctx = ctxOf(socket);
   894→      if (!ctx) return;
   895→      const { r, p } = ctx;
   896→      if (r.phase !== "racing" || !p.racing) return;
   897→      const body = (payload ?? {}) as { lap?: unknown; lapMs?: unknown; prog?: unknown };
   898→      const lap = Math.round(num(body.lap, -1));
   899→      const lapMs = num(body.lapMs, 0);
   900→      if (lap !== p.lap + 1) return; // out of order / duplicate
   901→      if (lapMs < MIN_LAP_MS) return; // impossible lap time
   902→      if (lapMs > Date.now() - r.raceStartedAt + 3000) return; // faster than wall clock
   903→      if (p.lapProgBase == null) return;
   904→      // distance travelled this lap — the lap payload's own progress is merged
   905→      // with the state stream so a just-missed state tick can't reject a lap
   906→      const progNow = Math.max(p.lastProgress, clampAbs(num(body.prog, p.lastProgress), 1e7));
   907→      if (progNow - p.lapProgBase < r.trackLength * 0.75) return; // not enough distance
   908→      p.lastProgress = progNow;
   909→      p.lap++;
   910→      p.lapProgBase = p.lastProgress;
   911→      if (p.bestLapMs == null || lapMs < p.bestLapMs) p.bestLapMs = lapMs;
   912→      io.to(r.code).emit("mp:lap", { pid: p.id, lap: p.lap, lapMs, name: p.name });
   913→    } catch (err) {
   914→      console.error("mp:lap error", err);
   915→    }
   916→  });
   917→
   918→  // ----- finish report (validated, position assigned server-side) -------------
   919→  socket.on("mp:finish", (payload: unknown) => {
   920→    try {
   921→      const ctx = ctxOf(socket);
   922→      if (!ctx) return;
   923→      const { r, p } = ctx;
   924→      if (r.phase !== "racing" || !p.racing) return;
   925→      if (p.finishPos != null) return; // already finished
   926→      const body = (payload ?? {}) as Record<string, unknown>;
   927→      const totalMs = num(body.totalMs, -1);
   928→      if (totalMs < r.laps * MIN_LAP_MS * 0.5) return; // impossible total
   929→      const serverElapsed = Date.now() - r.raceStartedAt;
   930→      if (totalMs > serverElapsed + 3000) return; // future time = reject
   931→      if (p.lap < r.laps) return; // laps not validated yet
   932→      p.finishMs = totalMs;
   933→      p.bestLapMs = clampRange(num(body.bestLapMs, 0), 0, 36e5);
   934→      p.topSpeedKmh = clampRange(num(body.topSpeedKmh, 0), 0, 500);
   935→      p.driftScore = clampRange(num(body.driftScore, 0), 0, 1e7);
   936→      p.finishPos = ++r.finishCounter;
   937→      io.to(r.code).emit("mp:finish", {
   938→        pid: p.id,
   939→        totalMs: p.finishMs,
   940→        pos: p.finishPos,
   941→        name: p.name,
   942→      });
   943→      broadcastRoom(r);
   944→      maybeScheduleCompile(r);
   945→    } catch (err) {
   946→      console.error("mp:finish error", err);
   947→    }
   948→  });

   982→// ---------------------------------------------------------------------------
   983→// aggregated 10 Hz relay: one broadcast per room with every racer's state
   984→// ---------------------------------------------------------------------------
   985→setInterval(() => {
   986→  const now = Date.now();
   987→  for (const r of rooms.values()) {
   988→    if (!r.relayOn) continue;
   989→    const players: unknown[] = [];
   990→    for (const pid of r.order) {
   991→      const p = r.players.get(pid);
   992→      if (!p || !p.racing || !p.lastState) continue;
   993→      const st = r.phase === "countdown" && p.anchor ? p.anchor : p.lastState;
   994→      players.push({ pid, ...st });
   995→    }
   996→    if (players.length) {
   997→      io.to(r.code).emit("mp:update", { t: now, players });
   998→    }
   999→  }
  1000→}, RELAY_INTERVAL_MS);

  1006→  const now = Date.now();
  1007→  for (const r of [...rooms.values()]) {

  1009→      console.log(`[room ${r.code}] expired (max age)`);
  1010→      deleteRoom(r);
  1011→      continue;
  1012→    }
  1013→    const anyConnected = [...r.players.values()].some((p) => p.connected);
  1014→    if (!anyConnected) {
  1015→      // all sockets gone (crash / mass drop) — remove once every grace expired
  1016→      const earliest = [...r.players.values()].reduce(
  1017→        (min, p) => Math.min(min, p.disconnectAt ?? now),
  1018→        Infinity
  1019→      );
  1020→      if (r.players.size === 0 || now - earliest > RECONNECT_GRACE_MS + 10000) {
  1021→        console.log(`[room ${r.code}] abandoned — cleaning up`);
  1022→        deleteRoom(r);
  1023→      }
  1024→    }
  1025→  }
  1026→}, 30000);
